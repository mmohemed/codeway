    نظرة عامة على المشروع

    تطبيق المهام (To-Do List) هو مشروع مثالي لتعلم C# في سياق عملي حقيقي. رغم بساطته الظاهرة، فهو يغطي مفاهيم أساسية: نمذجة البيانات، فصل المسؤوليات، حفظ البيانات، وواجهة تفاعلية.

    ما ستتعلمه

    العمارة النظيفة

    فصل Model عن Repository عن Service عن UI.

    حفظ البيانات

    القراءة والكتابة في ملف JSON محلي.

    واجهة حديثة

    تصميم تفاعلي باستخدام WPF أو Blazor.

    أمان الكود

    معالجة الأخطاء، التحقق، والاختبار.

    المتطلبات الوظيفية

    • إضافة مهمة جديدة مع عنوان وأولوية.
    • عرض قائمة المهام مرتبة حسب الأولوية.
    • تحديد مهمة كمكتملة أو إعادتها كمتبقية.
    • حذف مهمة.
    • فلترة المهام (الكل / متبقية / مكتملة).
    • حفظ المهام محلياً بحيث تبقى بعد إغلاق التطبيق.

    لماذا هذا المشروع؟

    لأنه صغير بما يكفي للفهم الكامل، وكبير بما يكفي لتغطية مفاهيم هندسة البرمجيات التي ستحتاجها في أي تطبيق حقيقي.

    نموذج البيانات (Model)

    نبدأ بأبسط جزء: كلاس يمثل المهمة. نستخدم record لأنها مثالية لنماذج البيانات غير القابلة للتغيير بعد الإنشاء.

    Models/TodoItem.cs
    namespace TodoApp.Models;

public enum Priority
{
    Low,
    Medium,
    High
}

public record TodoItem
{
    public Guid Id { get; init; } = Guid.NewGuid();
    public string Title { get; init; } = string.Empty;
    public Priority Priority { get; init; } = Priority.Medium;
    public bool IsDone { get; init; }
    public DateTime CreatedAt { get; init; } = DateTime.Now;

    // نسخة معدّلة — لأن record غير قابلة للتغيير
    public TodoItem WithDone(bool done) => this with { IsDone = done };
}

    لماذا record؟

    record في C# 9+ توفر مقارنة قيمة تلقائية، وخصائص init فقط، وعبارة with لإنشاء نسخ معدّلة — مثالية لنماذج البيانات.

    التحقق من الصحة

    نضيف دالة تحقق بسيطة قبل حفظ المهمة، لمنع إضافة مهام فارغة.

    Models/TodoItem.cs (إضافة)
    public static TodoItem Create(string title, Priority priority)
{
    if (string.IsNullOrWhiteSpace(title))
        throw new ArgumentException("العنوان مطلوب");

    return new TodoItem
    {
        Title = title.Trim(),
        Priority = priority
    };
}

    المستودع (Repository)

    المستودع هو المسؤول الوحيد عن الوصول للبيانات. نبدأ بواجهة، ثم نُنفّذها بحفظ JSON. هذا يسمح لنا لاحقاً بالتبديل إلى قاعدة بيانات دون تغيير باقي الكود.

    Repositories/ITodoRepository.cs
    using TodoApp.Models;

namespace TodoApp.Repositories;

public interface ITodoRepository
{
    Task<IReadOnlyList<TodoItem>> GetAllAsync();
    Task AddAsync(TodoItem item);
    Task UpdateAsync(TodoItem item);
    Task DeleteAsync(Guid id);
}

    تنفيذ JSON

    Repositories/JsonTodoRepository.cs
    using System.Text.Json;
using TodoApp.Models;

namespace TodoApp.Repositories;

public class JsonTodoRepository : ITodoRepository
{
    private readonly string _filePath;
    private readonly SemaphoreSlim _lock = new(1, 1);
    private List<TodoItem>? _cache;

    private static readonly JsonSerializerOptions JsonOpts = new()
    {
        WriteIndented = true,
        PropertyNamingPolicy = JsonNamingPolicy.CamelCase
    };

    public JsonTodoRepository(string filePath)
    {
        _filePath = filePath;
    }

    public async Task<IReadOnlyList<TodoItem>> GetAllAsync()
    {
        await EnsureLoadedAsync();
        return _cache!;
    }

    public async Task AddAsync(TodoItem item)
    {
        await EnsureLoadedAsync();
        _cache!.Add(item);
        await SaveAsync();
    }

    public async Task UpdateAsync(TodoItem item)
    {
        await EnsureLoadedAsync();
        var index = _cache!.FindIndex(x => x.Id == item.Id);
        if (index >= 0)
        {
            _cache[index] = item;
            await SaveAsync();
        }
    }

    public async Task DeleteAsync(Guid id)
    {
        await EnsureLoadedAsync();
        _cache!.RemoveAll(x => x.Id == id);
        await SaveAsync();
    }

    private async Task EnsureLoadedAsync()
    {
        if (_cache is not null) return;

        await _lock.WaitAsync();
        try
        {
            if (_cache is not null) return;

            if (!File.Exists(_filePath))
            {
                _cache = new List<TodoItem>();
                return;
            }

            var json = await File.ReadAllTextAsync(_filePath);
            _cache = JsonSerializer.Deserialize<List<TodoItem>>(json, JsonOpts)
                     ?? new List<TodoItem>();
        }
        finally { _lock.Release(); }
    }

    private async Task SaveAsync()
    {
        var json = JsonSerializer.Serialize(_cache, JsonOpts);
        await File.WriteAllTextAsync(_filePath, json);
    }
}

    لاحظ الـ SemaphoreSlim

    نستخدمه لمنع وصول متزامن إلى الملف عند وجود عمليات متعددة في نفس الوقت. هذا مبدأ أساسي في كتابة كود آمن للخيوط (thread-safe).

    منطق الأعمال (Service)

    طبقة الخدمة تحتوي على القواعد التجارية: التحقق، الترتيب، الفلترة. لا تعرف شيئاً عن الواجهة أو التخزين — تعتمد فقط على الواجهة ITodoRepository.

    Services/TodoService.cs
    using TodoApp.Models;
using TodoApp.Repositories;

namespace TodoApp.Services;

public class TodoService
{
    private readonly ITodoRepository _repo;

    public TodoService(ITodoRepository repo) => _repo = repo;

    public async Task<IReadOnlyList<TodoItem>> GetSortedAsync()
    {
        var items = await _repo.GetAllAsync();
        return items
            .OrderBy(x => x.IsDone)
            .ThenByDescending(x => x.Priority)
            .ThenBy(x => x.CreatedAt)
            .ToList();
    }

    public async Task AddAsync(string title, Priority priority)
    {
        var item = TodoItem.Create(title, priority);
        await _repo.AddAsync(item);
    }

    public async Task ToggleAsync(TodoItem item)
    {
        var updated = item.WithDone(!item.IsDone);
        await _repo.UpdateAsync(updated);
    }

    public Task DeleteAsync(Guid id) => _repo.DeleteAsync(id);
}

    واجهة المستخدم (UI)

    نستخدم WPF + MVVM لفصل الواجهة عن المنطق. الـ ViewModel يحتوي على حالة الواجهة والأوامر، والـ View يحتوي فقط على التصميم.

    MainViewModel

    ViewModels/MainViewModel.cs
    using System.Collections.ObjectModel;
using System.Windows.Input;
using TodoApp.Models;
using TodoApp.Services;

namespace TodoApp.ViewModels;

public class MainViewModel : ObservableObject
{
    private readonly TodoService _service;

    public ObservableCollection<TodoItem> Items { get; } = new();

    private string _newTitle = "";
    public string NewTitle
    {
        get => _newTitle;
        set => SetProperty(ref _newTitle, value);
    }

    public ICommand AddCommand { get; }
    public ICommand ToggleCommand { get; }
    public ICommand DeleteCommand { get; }

    public MainViewModel(TodoService service)
    {
        _service = service;
        AddCommand = new RelayCommand(async _ => await AddAsync());
        ToggleCommand = new RelayCommand(async p => await ToggleAsync((TodoItem)p));
        DeleteCommand = new RelayCommand(async p => await DeleteAsync((TodoItem)p));
    }

    public async Task LoadAsync()
    {
        Items.Clear();
        foreach (var item in await _service.GetSortedAsync())
            Items.Add(item);
    }

    private async Task AddAsync()
    {
        if (string.IsNullOrWhiteSpace(NewTitle)) return;

        await _service.AddAsync(NewTitle, Priority.Medium);
        NewTitle = "";
        await LoadAsync();
    }

    private async Task ToggleAsync(TodoItem item)
    {
        await _service.ToggleAsync(item);
        await LoadAsync();
    }

    private async Task DeleteAsync(TodoItem item)
    {
        await _service.DeleteAsync(item.Id);
        await LoadAsync();
    }
}

    حفظ البيانات (JSON)

    نستخدم System.Text.Json (مدمج في .NET) لحفظ واسترجاع المهام. الملف يُخزّن في مجلد التطبيق المحلي لضمان الوصول الآمن.

    Program.cs
    using TodoApp.Repositories;
using TodoApp.Services;

var appData = Environment.GetFolderPath(
    Environment.SpecialFolder.LocalApplicationData);

var todoDir = Path.Combine(appData, "CodeWay", "TodoApp");
Directory.CreateDirectory(todoDir);

var filePath = Path.Combine(todoDir, "todos.json");

var repo = new JsonTodoRepository(filePath);
var service = new TodoService(repo);
var vm = new MainViewModel(service);

await vm.LoadAsync();

    أفضل الممارسات

    • فصل المسؤوليات: كل طبقة تعرف فقط ما تحتاجه. Model لا يعرف Repository، والـ UI لا يعرف JSON.
    • استخدم async/await في كل عمليات I/O — لحفظ الملفات دون تجميد الواجهة.
    • تحقق من الصحة مبكراً في طبقة الخدمة، قبل الوصول للتخزين.
    • استخدم CancellationToken للعمليات الطويلة لدعم الإلغاء.
    • اكتب اختبارات وحدة للـ Service عبر Mock Repository.
    • تعامل مع الاستثناءات بلطف وأظهر رسائل واضحة للمستخدم.

    مثال: اختبار وحدة

    Tests/TodoServiceTests.cs
    public class TodoServiceTests
{
    [Fact]
    public async Task AddAsync_ValidTitle_AddsItem()
    {
        // Arrange
        var repo = new Mock<ITodoRepository>();
        var service = new TodoService(repo.Object);

        // Act
        await service.AddAsync("مهمة جديدة", Priority.High);

        // Assert
        repo.Verify(r => r.AddAsync(
            It.Is<TodoItem>(i =>
                i.Title == "مهمة جديدة" &&
                i.Priority == Priority.High)),
            Times.Once());
    }
}
