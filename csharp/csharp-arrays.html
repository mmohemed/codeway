المصفوفات والمجاميع في C#

تعلم كيفية تخزين مجموعات من البيانات باستخدام Arrays، ثم قم بجمعها، حساب متوسطها، والبحث فيها. أساسيات لا غنى عنها لأي مبرمج.

Single-Dimensional Sum, Average, Max foreach, for, LINQ

ما هي المصفوفة (Array)؟

المصفوفة هي مجموعة من العناصر من نفس النوع، مخزنة في مواقع ذاكرة متتالية. يمكن الوصول إلى أي عنصر باستخدام الفهرس (index) الذي يبدأ من 0.

// طرق مختلفة لإنشاء مصفوفة
int[] numbers = new int[5];        // مصفوفة من 5 أعداد (كلها 0)
numbers[0] = 10;
numbers[1] = 20;

int[] scores = new int[] { 95, 82, 74, 88, 91 }; // تهيئة بالقيم
string[] names = { "أحمد", "سارة", "محمد" };     // صيغة مختصرة

// الوصول إلى العناصر
Console.WriteLine(names[1]);  // الناتج: سارة
حجم المصفوفة ثابت بعد إنشائها. يمكنك الحصول على الطول باستخدام خاصية Length.

التكرار على المصفوفات

استخدم for أو foreach لاجتياز جميع عناصر المصفوفة بسهولة.

int[] grades = { 90, 85, 77, 92, 88 };

// باستخدام for
for (int i = 0; i < grades.Length; i++)
{
    Console.WriteLine($"العنصر {i}: {grades[i]}");
}

// باستخدام foreach (أسهل للقراءة)
foreach (int grade in grades)
{
    Console.WriteLine($"الدرجة: {grade}");
}

حساب المجموع، المتوسط، أكبر وأصغر قيمة

🔢 Sum (المجموع) 📊 Average (المتوسط) 📈 Max / Min ⚡ LINQ (طريقة مختصرة) 
int[] values = { 15, 27, 33, 42, 50 };

// 1. حساب المجموع باستخدام حلقة
int sum = 0;
for (int i = 0; i < values.Length; i++)
{
    sum += values[i];
}
Console.WriteLine($"المجموع: {sum}");

// 2. المتوسط الحسابي
double average = (double)sum / values.Length;
Console.WriteLine($"المتوسط: {average:F2}");

// 3. أكبر وأصغر قيمة
int max = values[0];
int min = values[0];
foreach (int num in values)
{
    if (num > max) max = num;
    if (num < min) min = num;
}
Console.WriteLine($"الأكبر: {max}, الأصغر: {min}");

// 4. باستخدام LINQ (سهل وسريع)
using System.Linq;
int sumLinq = values.Sum();
double avgLinq = values.Average();
int maxLinq = values.Max();
Console.WriteLine($"LINQ → مجموع: {sumLinq}, متوسط: {avgLinq}");
نصيحة: استخدم using System.Linq; للاستفادة من دوال جاهزة مثل Sum() و Average() و Max().

مثال تطبيقي: درجات الطلاب

نفترض مصفوفة بدرجات 5 طلاب. نحسب المجموع والمتوسط ونطبع النتائج.

double[] studentScores = { 88.5, 92.0, 76.5, 89.0, 95.5 };
double total = 0;

for (int i = 0; i < studentScores.Length; i++)
{
    total += studentScores[i];
}

double averageScore = total / studentScores.Length;
Console.WriteLine($"مجموع الدرجات: {total}");
Console.WriteLine($"متوسط الدرجات: {averageScore:F2}");

// إيجاد أعلى درجة
double highest = studentScores.Max();
Console.WriteLine($"أعلى درجة: {highest}");

البحث في المصفوفة

البحث عن قيمة معينة داخل مصفوفة وإرجاع فهرسها أو رسالة بعدم وجودها.

string[] fruits = { "تفاح", "موز", "برتقال", "عنب" };
string searchItem = "موز";
int index = -1;

for (int i = 0; i < fruits.Length; i++)
{
    if (fruits[i] == searchItem)
    {
        index = i;
        break;
    }
}

if (index != -1)
    Console.WriteLine($"تم العثور على '{searchItem}' في الفهرس {index}");
else
    Console.WriteLine($"'{searchItem}' غير موجود");

المصفوفات ثنائية الأبعاد وجمع العناصر

المصفوفات ثنائية الأبعاد (Matrix) تشبه جدولاً له صفوف وأعمدة. يمكن حساب مجموع كل الصفوف أو كل عمود.

// مصفوفة 3x2
int[,] matrix = {
    { 2, 4 },
    { 6, 8 },
    { 1, 3 }
};

int totalSum = 0;
for (int row = 0; row < matrix.GetLength(0); row++)
{
    for (int col = 0; col < matrix.GetLength(1); col++)
    {
        totalSum += matrix[row, col];
    }
}
Console.WriteLine($"مجموع جميع العناصر: {totalSum}");  // الناتج: 24

// حساب مجموع كل صف
for (int row = 0; row < matrix.GetLength(0); row++)
{
    int rowSum = 0;
    for (int col = 0; col < matrix.GetLength(1); col++)
        rowSum += matrix[row, col];
    Console.WriteLine($"مجموع الصف {row + 1} = {rowSum}");
}

جرب بنفسك: محاكاة جمع أرقام مصفوفة

الكود التالي يحسب مجموع عناصر مصفوفة numbers. اضغط على "تشغيل المحاكاة" لترى النتيجة خطوة بخطوة.

int[] numbers = { 12, 45, 23, 67, 34, 19 };
int total = 0;

Console.WriteLine("عناصر المصفوفة:");
foreach (int num in numbers)
{
    Console.Write(num + " ");
    total += num;
}
Console.WriteLine($"\nالمجموع الكلي = {total}");
Console.WriteLine($"المتوسط الحسابي = {(double)total / numbers.Length:F2}");
// انقر على الزر لترى مخرجات حساب المجموع والمتوسط

هذه محاكاة تعليمية تُظهر تنفيذ برنامج حقيقي يجمع أعداد مصفوفة.
