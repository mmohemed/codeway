مقدمة عن المدخلات والمخرجات

في لغة C++، تعتبر عمليات الإدخال (Input) والإخراج (Output) أساسية في أي برنامج. تتيح هذه العمليات التفاعل مع المستخدم وقراءة البيانات وعرض النتائج. توفر لغة C++ مكتبة الـ iostream التي تحتوي على الأدوات اللازمة للتعامل مع المدخلات والمخرجات.

تضمين مكتبة iostream 
#include <iostream>
using namespace std;

int main() {
    // الكود الخاص بك هنا
    return 0;
}

#include <iostream> - لتضمين مكتبة الإدخال والإخراج

using namespace std; - لاستخدام العناصر في المكتبة القياسية بدون بادئة std::

int main() - الدالة الرئيسية التي يبدأ منها البرنامج

cout - إخراج البيانات

يستخدم cout (سي أوت) لعرض البيانات على الشاشة. يتم استخدامه مع عامل التشغيل << (insertion operator).

أمثلة على استخدام cout:

مثال 1: طباعة نص 
#include <iostream>
using namespace std;

int main() {
    cout << "مرحباً بكم في لغة C++" << endl;
    cout << "هذا درس المدخلات والمخرجات" << endl;
    return 0;
}

الشرح:

cout << "النص" - لطباعة النص

endl - للانتقال إلى سطر جديد

يمكنك استخدام \n بدلاً من endl

طباعة المتغيرات:

مثال 2: طباعة متغيرات 
#include <iostream>
using namespace std;

int main() {
    int age = 20;
    double salary = 5000.50;
    char grade = 'A';
    string name = "Ahmed";
    
    cout << "الاسم: " << name << endl;
    cout << "العمر: " << age << " سنة" << endl;
    cout << "الراتب: " << salary << " دولار" << endl;
    cout << "التقدير: " << grade << endl;
    
    return 0;
}

المخرجات

الاسم: Ahmed
العمر: 20 سنة
الراتب: 5000.5 دولار
التقدير: A

cin - إدخال البيانات

يستخدم cin (سي إن) لقراءة البيانات من المستخدم. يتم استخدامه مع عامل التشغيل >> (extraction operator).

مثال 3: قراءة بيانات من المستخدم 
#include <iostream>
using namespace std;

int main() {
    string name;
    int age;
    double salary;
    
    cout << "أدخل اسمك: ";
    cin >> name;
    
    cout << "أدخل عمرك: ";
    cin >> age;
    
    cout << "أدخل راتبك: ";
    cin >> salary;
    
    cout << "\nالبيانات المدخلة:\n";
    cout << "الاسم: " << name << endl;
    cout << "العمر: " << age << endl;
    cout << "الراتب: " << salary << endl;
    
    return 0;
}

ملاحظات مهمة:

1. cin يقرأ حتى أول مسافة (لا يقرأ النص بعد المسافة)

2. لقراءة سطر كامل استخدم getline(cin, variable)

3. تأكد من تطابق نوع البيانات المدخلة مع نوع المتغير

قراءة سطر كامل باستخدام getline:

مثال 4: استخدام getline 
#include <iostream>
#include <string>
using namespace std;

int main() {
    string fullName;
    string address;
    
    cout << "أدخل اسمك الكامل: ";
    getline(cin, fullName);
    
    cout << "أدخل عنوانك: ";
    getline(cin, address);
    
    cout << "\nالاسم: " << fullName << endl;
    cout << "العنوان: " << address << endl;
    
    return 0;
}

تنسيق المخرجات

توفر C++ طرق متعددة لتنسيق المخرجات وجعلها أكثر وضوحاً وجاذبية.

مثال 5: تنسيق الأرقام 
#include <iostream>
#include <iomanip> // للمؤثرات setw, setprecision
using namespace std;

int main() {
    double pi = 3.14159265359;
    int number = 42;
    
    // تحديد عدد المنازل العشرية
    cout << fixed << setprecision(2);
    cout << "Pi: " << pi << endl;
    
    // تحديد عرض الحقل
    cout << "الرقم: " << setw(10) << number << endl;
    cout << "الرقم: " << setw(10) << setfill('*') << number << endl;
    
    // عرض الأرقام بنظام hexadecimal
    cout << hex << "بالنظام الست عشري: " << number << endl;
    cout << dec << "بالنظام العشري: " << number << endl;
    
    return 0;
}
المؤثر الوظيفة مثال
setw(n) تحديد عرض الحقل cout << setw(10) << x;
setprecision(n) تحديد عدد المنازل العشرية cout << setprecision(2) << x;
setfill(c) تحديد حرف التعبئة cout << setfill('*');
fixed عرض الأرقام العشرية بشكل ثابت cout << fixed << x;

cerr و clog - أخطاء وتحذيرات

يستخدم cerr لعرض رسائل الخطأ، ويستخدم clog لعرض رسائل التحذير أو السجلات.

مثال 6: استخدام cerr و clog 
#include <iostream>
using namespace std;

int main() {
    int age;
    
    cout << "أدخل عمرك: ";
    cin >> age;
    
    if (age < 0) {
        cerr << "خطأ: العمر لا يمكن أن يكون سالباً!" << endl;
    } else if (age < 18) {
        clog << "تحذير: المستخدم تحت السن القانوني" << endl;
        cout << "عذراً، لا يمكنك الدخول" << endl;
    } else {
        cout << "مرحباً بك في الموقع" << endl;
    }
    
    return 0;
}

cerr

لرسائل الخطأ (غير مخزنة مؤقتاً)

clog

لرسائل التحذير (مخزنة مؤقتاً)

cout

للمخرجات العادية (مخزنة مؤقتاً)

تمارين وتطبيقات عملية

التمرين 1: آلة حاسبة بسيطة

اكتب برنامجاً يطلب رقمين من المستخدم ويقوم بجمعهما وطرحهما وضربهما وقسمتهما.

الحل 
#include <iostream>
using namespace std;

int main() {
    double num1, num2;
    
    cout << "أدخل الرقم الأول: ";
    cin >> num1;
    
    cout << "أدخل الرقم الثاني: ";
    cin >> num2;
    
    cout << "\nالنتائج:\n";
    cout << "الجمع: " << num1 + num2 << endl;
    cout << "الطرح: " << num1 - num2 << endl;
    cout << "الضرب: " << num1 * num2 << endl;
    
    if (num2 != 0) {
        cout << "القسمة: " << num1 / num2 << endl;
    } else {
        cerr << "خطأ: لا يمكن القسمة على صفر!" << endl;
    }
    
    return 0;
}

التمرين 2: محول درجات الحرارة

اكتب برنامجاً يحول درجة الحرارة من مئوية إلى فهرنهايت والعكس.

الحل 
#include <iostream>
#include <iomanip>
using namespace std;

int main() {
    double celsius, fahrenheit;
    int choice;
    
    cout << "اختر نوع التحويل:\n";
    cout << "1. من مئوي إلى فهرنهايت\n";
    cout << "2. من فهرنهايت إلى مئوي\n";
    cout << "اختيارك: ";
    cin >> choice;
    
    cout << fixed << setprecision(2);
    
    if (choice == 1) {
        cout << "أدخل درجة الحرارة بالمئوي: ";
        cin >> celsius;
        fahrenheit = (celsius * 9/5) + 32;
        cout << celsius << "°C = " << fahrenheit << "°F" << endl;
    } else if (choice == 2) {
        cout << "أدخل درجة الحرارة بالفهرنهايت: ";
        cin >> fahrenheit;
        celsius = (fahrenheit - 32) * 5/9;
        cout << fahrenheit << "°F = " << celsius << "°C" << endl;
    } else {
        cerr << "خطأ: اختيار غير صحيح!" << endl;
    }
    
    return 0;
}
