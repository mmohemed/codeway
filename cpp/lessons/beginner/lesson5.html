العوامل الحسابية (Arithmetic Operators)

تستخدم العوامل الحسابية لإجراء العمليات الرياضية الأساسية على الأرقام.

العامل الوصف مثال النتيجة
+ الجمع 5 + 3 8
- الطرح 10 - 4 6
* الضرب 6 * 7 42
/ القسمة 15 / 4 3.75
% الباقي (Modulus) 17 % 5 2

أمثلة عملية:

مثال 1: العمليات الحسابية الأساسية 
#include <iostream>
using namespace std;

int main() {
    int a = 15, b = 4;
    
    cout << "a = " << a << ", b = " << b << endl;
    cout << "الجمع: " << a + b << endl;
    cout << "الطرح: " << a - b << endl;
    cout << "الضرب: " << a * b << endl;
    cout << "القسمة: " << a / b << endl;  // ناتج قسمة عددين صحيحين
    cout << "الباقي: " << a % b << endl;
    
    return 0;
}

ملاحظات مهمة:

1. عند قسمة عددين صحيحين، تكون النتيجة عدداً صحيحاً (يتم تجاهل الكسور)

2. عامل الباقي % يعمل فقط مع الأعداد الصحيحة

3. للحصول على ناتج قسمة دقيق، استخدم الأعداد العشرية (double/float)

مثال 2: القسمة مع الأعداد العشرية 
#include <iostream>
using namespace std;

int main() {
    double x = 15.0, y = 4.0;
    int a = 15, b = 4;
    
    cout << "قسمة أعداد عشرية: " << x / y << endl;
    cout << "قسمة أعداد صحيحة: " << a / b << endl;
    cout << "قسمة مع تحويل النوع: " << (double)a / b << endl;
    
    return 0;
}

المخرجات:

قسمة أعداد عشرية: 3.75
قسمة أعداد صحيحة: 3
قسمة مع تحويل النوع: 3.75

عوامل الإسناد المركبة (Assignment Operators)

تستخدم لاختصار العمليات الحسابية مع الإسناد في نفس الوقت.

العامل المعنى مثال يعادل
+= إضافة وإسناد x += 5 x = x + 5
-= طرح وإسناد x -= 3 x = x - 3
*= ضرب وإسناد x *= 2 x = x * 2
/= قسمة وإسناد x /= 4 x = x / 4
%= باقي وإسناد x %= 3 x = x % 3
مثال 3: استخدام عوامل الإسناد المركبة 
#include <iostream>
using namespace std;

int main() {
    int x = 10;
    
    cout << "x = " << x << endl;
    
    x += 5;  // x = x + 5
    cout << "بعد x += 5: " << x << endl;
    
    x *= 2;  // x = x * 2
    cout << "بعد x *= 2: " << x << endl;
    
    x -= 7;  // x = x - 7
    cout << "بعد x -= 7: " << x << endl;
    
    x %= 4;  // x = x % 4
    cout << "بعد x %= 4: " << x << endl;
    
    return 0;
}

عوامل الزيادة والنقصان (Increment/Decrement)

تستخدم لزيادة أو نقصان قيمة المتغير بمقدار 1.

++ (الزيادة)

تزيد قيمة المتغير بمقدار 1

int x = 5;
x++; // x = 6
++x; // x = 7

-- (النقصان)

تنقص قيمة المتغير بمقدار 1

int y = 5;
y--; // y = 4
--y; // y = 3

الفرق بين prefix و postfix:

مثال 4: prefix vs postfix 
#include <iostream>
using namespace std;

int main() {
    int a = 5, b = 5;
    
    // Postfix: تستخدم القيمة ثم تزيد
    cout << "postfix a++: " << a++ << endl;  // تطبع 5 ثم تزيد إلى 6
    cout << "قيمة a الآن: " << a << endl;
    
    // Prefix: تزيد ثم تستخدم القيمة
    cout << "prefix ++b: " << ++b << endl;  // تزيد إلى 6 ثم تطبع 6
    cout << "قيمة b الآن: " << b << endl;
    
    return 0;
}

المخرجات:

postfix a++: 5
قيمة a الآن: 6
prefix ++b: 6
قيمة b الآن: 6

عوامل المقارنة (Relational Operators)

تستخدم لمقارنة قيمتين وتنتج قيمة منطقية (true/false).

العامل الوصف مثال النتيجة
== يساوي 5 == 5 true
!= لا يساوي 5 != 3 true
> أكبر من 7 > 3 true
< أصغر من 4 < 8 true
>= أكبر من أو يساوي 5 >= 5 true
<= أصغر من أو يساوي 3 <= 5 true
مثال 5: استخدام عوامل المقارنة 
#include <iostream>
using namespace std;

int main() {
    int x = 10, y = 5;
    
    cout << boolalpha;  // لطباعة true/false بدلاً من 1/0
    
    cout << "x == y: " << (x == y) << endl;
    cout << "x != y: " << (x != y) << endl;
    cout << "x > y: " << (x > y) << endl;
    cout << "x < y: " << (x < y) << endl;
    cout << "x >= y: " << (x >= y) << endl;
    cout << "x <= y: " << (x <= y) << endl;
    
    return 0;
}

العوامل المنطقية (Logical Operators)

تستخدم لربط الشروط المنطقية وإنتاج نتائج منطقية.

العامل الوصف مثال النتيجة
&& و (AND) (5 > 3) && (2 < 4) true
|| أو (OR) (5 < 3) || (2 < 4) true
! نفي (NOT) !(5 < 3) true

جدول الحقيقة:

A B A && B A || B !A
true true true true false
true false false true false
false true false true true
false false false false true
مثال 6: استخدام العوامل المنطقية 
#include <iostream>
using namespace std;

int main() {
    int age = 25;
    bool hasLicense = true;
    bool isStudent = false;
    
    cout << boolalpha;
    
    // التحقق من شروط قيادة السيارة
    bool canDrive = (age >= 18) && hasLicense;
    cout << "يمكنه القيادة: " << canDrive << endl;
    
    // التحقق من خصم الطلاب
    bool hasDiscount = isStudent || (age >= 60);
    cout << "له خصم: " << hasDiscount << endl;
    
    // نفي الشرط
    bool isMinor = !(age >= 18);
    cout << "قاصر: " << isMinor << endl;
    
    // شروط مركبة
    bool condition = (age > 20) && (age < 30) && hasLicense;
    cout << "شرط مركب: " << condition << endl;
    
    return 0;
}

أولويات العمليات (Operator Precedence)

تحدد أولويات العمليات الترتيب الذي يتم به تقييم التعبيرات الحسابية.

الأولوية العامل الوصف
1 (أعلى) () الأقواس
2 ++ -- ! الزيادة/النقصان والنفي
3 * / % الضرب والقسمة والباقي
4 + - الجمع والطرح
5 < <= > >= عوامل المقارنة
6 == != المساواة وعدم المساواة
7 && الوصل المنطقي AND
8 || الوصل المنطقي OR
9 (أدنى) = += -= *= /= %= الإسناد
مثال 7: تأثير أولويات العمليات 
#include <iostream>
using namespace std;

int main() {
    int a = 5, b = 3, c = 2;
    
    // بدون أقواس
    int result1 = a + b * c;
    cout << "a + b * c = " << result1 << " (الضرب أولاً)" << endl;
    
    // مع أقواس
    int result2 = (a + b) * c;
    cout << "(a + b) * c = " << result2 << " (الجمع أولاً)" << endl;
    
    // مثال معقد
    int result3 = a++ * b + --c;
    cout << "a++ * b + --c = " << result3 << endl;
    
    // شرط منطقي
    bool result4 = (a > b) && (b < c) || (a == 5);
    cout << boolalpha << "نتيجة الشرط: " << result4 << endl;
    
    return 0;
}

نصيحة: استخدم الأقواس لتوضيح أولويات العمليات وجعل الكود أكثر قابلية للقراءة، حتى عندما تكون الأولويات واضحة.

تمارين وتطبيقات عملية

التمرين 1: آلة حاسبة متقدمة

اكتب برنامجاً يطلب رقمين من المستخدم ويقوم بجميع العمليات الحسابية عليهم.

الحل 
#include <iostream>
#include <iomanip>
using namespace std;

int main() {
    double num1, num2;
    
    cout << "أدخل الرقم الأول: ";
    cin >> num1;
    
    cout << "أدخل الرقم الثاني: ";
    cin >> num2;
    
    cout << fixed << setprecision(2);
    cout << "\n=== النتائج ===\n";
    cout << num1 << " + " << num2 << " = " << num1 + num2 << endl;
    cout << num1 << " - " << num2 << " = " << num1 - num2 << endl;
    cout << num1 << " * " << num2 << " = " << num1 * num2 << endl;
    
    if (num2 != 0) {
        cout << num1 << " / " << num2 << " = " << num1 / num2 << endl;
        
        // باقي القسمة للأعداد الصحيحة فقط
        int n1 = (int)num1, n2 = (int)num2;
        if (n2 != 0) {
            cout << n1 << " % " << n2 << " = " << n1 % n2 << endl;
        }
    } else {
        cout << "لا يمكن القسمة على صفر!" << endl;
    }
    
    return 0;
}

التمرين 2: التحقق من الشروط

اكتب برنامجاً يتحقق من شروط معينة في رقم مدخل.

الحل 
#include <iostream>
using namespace std;

int main() {
    int number;
    
    cout << "أدخل رقماً: ";
    cin >> number;
    
    cout << boolalpha;
    cout << "\n=== نتائج التحقق ===\n";
    cout << "الرقم موجب: " << (number > 0) << endl;
    cout << "الرقم زوجي: " << (number % 2 == 0) << endl;
    cout << "الرقم يقبل القسمة على 3: " << (number % 3 == 0) << endl;
    cout << "الرقم بين 10 و 20: " << (number > 10 && number < 20) << endl;
    cout << "الرقم أقل من 0 أو أكبر من 100: " << (number < 0 || number > 100) << endl;
    
    return 0;
}
