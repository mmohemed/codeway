١ مقدمة عن المتغيرات

المتغيرات هي بمثابة وعاء لتخزين البيانات في الذاكرة. تخيل أنك تمتلك خزانة بها أدراج متعددة، كل درج له اسم وقيمة بداخله. هذا هو بالضبط مفهوم المتغيرات في C++!

C++ 
// مفهوم المتغيرات
#include <iostream>
using namespace std;

int main() {
    // تعريف متغيرات بأنواع مختلفة
    int age = 25;              // متغير لحفظ العمر
    double salary = 5000.50;   // متغير لحفظ الراتب
    char grade = 'A';          // متغير لحفظ الدرجة
    string name = "Ahmed";     // متغير لحفظ الاسم
    
    cout << "Name: " << name << endl;
    cout << "Age: " << age << endl;
    cout << "Salary: " << salary << endl;
    cout << "Grade: " << grade << endl;
    
    return 0;
}

٢ قواعد تسمية المتغيرات

✅ صحيح

int age;
double salary_2024;
string firstName;
int _counter;
int number1;

❌ غير صحيح

int 123age;     // يبدأ برقم
double salary$;  // رمز خاص
string first name; // مسافة
int void;        // كلمة محجوزة

📌 أهم القواعد:

  • يجب أن يبدأ الاسم بحرف أو شرطة سفلية (_)
  • يمكن أن يحتوي على حروف وأرقام وشرطة سفلية
  • حساس لحالة الأحرف (age و Age متغيران مختلفان)
  • لا يمكن استخدام الكلمات المحجوزة (int, return, etc.)

٣ أنواع البيانات الأساسية

النوع الوصف الحجم المدى مثال
int أعداد صحيحة 4 بايت -2B to 2B int x = 10;
float أعداد عشرية (دقة مفردة) 4 بايت ±3.4e±38 float y = 3.14f;
double أعداد عشرية (دقة مزدوجة) 8 بايت ±1.7e±308 double z = 3.14159;
char حرف واحد 1 بايت -128 to 127 char c = 'A';
bool قيمة منطقية 1 بايت true/false bool flag = true;
string نص متغير - string s = "Hello";
C++ 
// أمثلة على أنواع البيانات
#include <iostream>
#include <string>
using namespace std;

int main() {
    // 1. Integer (أعداد صحيحة)
    int age = 25;
    int count = -10;
    int hexValue = 0xFF; // 255 بالنظام الست عشري
    
    // 2. Floating Point (أعداد عشرية)
    float price = 99.99f;
    float scientific = 1.5e4f; // 15000
    
    // 3. Double (دقة أعلى)
    double pi = 3.141592653589793;
    double large = 1.7e308;
    
    // 4. Character (حروف)
    char grade = 'A';
    char newline = '\n';
    char digit = '7';
    
    // 5. Boolean (منطقي)
    bool isStudent = true;
    bool isPassed = false;
    
    // 6. String (نصوص)
    string name = "Mohammed";
    string message = "Hello, C++!";
    
    // طباعة القيم
    cout << "Integer: " << age << endl;
    cout << "Float: " << price << endl;
    cout << "Double: " << pi << endl;
    cout << "Char: " << grade << endl;
    cout << "Bool: " << isStudent << endl;
    cout << "String: " << name << endl;
    
    return 0;
}

٤ معدلات الأنواع (Type Modifiers)

المعدل الوصف مثال
short يقلل حجم العدد الصحيح short int x = 100;
long يزيد حجم العدد الصحيح long int y = 100000;
long long يزيد الحجم أكثر long long z = 1e12;
unsigned أعداد موجبة فقط unsigned int w = 500;
signed أعداد موجبة وسالبة signed int v = -50; 
// أمثلة على المعدلات
#include <iostream>
using namespace std;

int main() {
    short int small = 32000;          // 2 بايت
    long int big = 1000000;            // 4 أو 8 بايت
    long long veryBig = 1000000000000; // 8 بايت
    
    unsigned int positive = 4294967295; // فقط أرقام موجبة
    signed int negative = -100;         // أرقام موجبة وسالبة
    
    cout << "Size of short: " << sizeof(short) << " bytes" << endl;
    cout << "Size of long: " << sizeof(long) << " bytes" << endl;
    cout << "Size of long long: " << sizeof(long long) << " bytes" << endl;
    
    return 0;
}
