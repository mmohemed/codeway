الجمل الشرطية في لغة ++C

شرح أكاديمي متكامل – التحكم في تدفق البرنامج

اتخاذ القرار

تسمح الجمل الشرطية للبرنامج باختيار مسار معين بناءً على شرط منطقي، تمامًا مثل القرارات في الحياة الواقعية.

قيم منطقية

أساس الشروط هو التعبيرات التي تكون قيمتها true (صحيح) أو false (خاطئ).

هيكلة الأكواد

تنظيم العمليات المعقدة وجعل البرنامج أكثر ذكاءً واستجابة للمدخلات المختلفة.

1. الجملة الشرطية if

أبسط صورة للشرط: إذا تحقق الشرط ننفذ كتلة معينة، وإلا نتجاوزها.

// صيغة if الأساسية
if (الشرط) {
    // الكود الذي ينفذ إذا كان الشرط true
}

// مثال عملي
int درجة = 85;
if (درجة >= 50) {
    cout << "نجحت! تهانينا." << endl;
}

2. if – else (إذا ... وإلا)

نستخدمها عندما نريد تنفيذ كود في حال تحقق الشرط، وكود آخر في حال عدم تحققه.

if (الشرط) {
    // إذا كان الشرط صحيحاً
} else {
    // إذا كان الشرط خاطئاً
}

// مثال: تحديد زوجي أو فردي
int عدد = 7;
if (عدد % 2 == 0) {
    cout << "العدد زوجي" << endl;
} else {
    cout << "العدد فردي" << endl;
}

نصيحة: يمكن حذف الأقواس {} إذا كان هناك جملة واحدة فقط، لكن يُفضل دائماً استخدامها للوضوح وتجنب الأخطاء.

3. if – else if – else (شروط متعددة)

عندما يكون لدينا أكثر من احتمال، نستخدم else if لتحديد شروط إضافية.

if (الشرط1) {
    // إذا تحقق الشرط1
} else if (الشرط2) {
    // إذا لم يتحقق الشرط1 وتحقق الشرط2
} else if (الشرط3) {
    // إذا لم يتحقق الشرط1 والشرط2 وتحقق الشرط3
} else {
    // إذا لم يتحقق أي من الشروط السابقة
}

// مثال: تقدير الدرجات
if (درجة >= 90) {
    cout << "امتياز";
} else if (درجة >= 75) {
    cout << "جيد جداً";
} else if (درجة >= 60) {
    cout << "جيد";
} else {
    cout << "راسب";
}

4. switch (الاختيار المتعدد المنظم)

تستخدم عندما نريد مقارنة متغير بعدد ثابت من القيم (أعداد صحيحة أو محارف). أنيقة وفعالة.

switch (التعبير) {
    case قيمة1:
        // كود
        break;
    case قيمة2:
        // كود
        break;
    default:
        // إذا لم يتطابق أي case
}

// مثال: أيام الأسبوع
int يوم = 3;
switch (يوم) {
    case 1: cout << "السبت"; break;
    case 2: cout << "الأحد"; break;
    case 3: cout << "الاثنين"; break;
    case 4: cout << "الثلاثاء"; break;
    case 5: cout << "الأربعاء"; break;
    case 6: cout << "الخميس"; break;
    case 7: cout << "الجمعة"; break;
    default: cout << "يوم غير صحيح";
}
تنبيه: لا تنسَ كتابة break بعد كل حالة، وإلا ستستمر الجملة في تنفيذ الحالات التالية (fall-through).

5. العامل الشرطي (?:) – Ternary Operator

طريقة مختصرة لكتابة if بسيط من سطر واحد. مفيد للتعيينات السريعة.

// الصيغة: (شرط) ? (قيمة إذا true) : (قيمة إذا false);

int عمر = 20;
string نتيجة = (عمر >= 18) ? "بالغ" : "قاصر";
cout << نتيجة; // سيطبع "بالغ"

// يمكن استخدامه مع العمليات الحسابية أيضاً
int x = 10, y = 20;
int الأكبر = (x > y) ? x : y; // الأكبر = 20

6. الشروط المتداخلة (Nested)

يمكن وضع جملة شرطية داخل أخرى لفحص أدق. لكن احذر من التعقيد الزائد.

int رقم = 15;
if (رقم > 0) {
    cout << "موجب ";
    if (رقم % 2 == 0) {
        cout << "وزوجي";
    } else {
        cout << "وفردي";
    }
} else {
    cout << "سالِب أو صفر";
}
// الناتج: "موجب وفردي"

أفضل الممارسات: اجعل شروطك واضحة، تجنب التداخل العميق أكثر من 3 مستويات، استخدم الأقواس حتى مع جملة واحدة، وعلّق على الشروط المعقدة.

ملصق تذكاري: أنواع الشروط بلغة ++C

  • if (condition) { }
  • if () { } else { }
  • if () { } else if () { } else { }
  • switch (x) { case ...: }
  • (cond) ? expr1 : expr2
  • Nested statements
