مقدمة عن STL (Standard Template Library)

مكتبة القوالب المعيارية STL هي مجموعة قوية من هياكل البيانات والخوارزميات في C++، مقسمة إلى ثلاثة أجزاء رئيسية:

الحاويات (Containers)

هياكل بيانات لتخزين المجموعات مثل Vector, List, Map

الخوارزميات (Algorithms)

دوال جاهزة للبحث والترتيب والمعالجة مثل sort, find

المكررات (Iterators)

أدوات للتنقل بين عناصر الحاويات

تضمين مكتبات STL 
#include <iostream>
#include <vector>
#include <list>
#include <map>
#include <set>
#include <algorithm>
#include <string>

using namespace std;

int main() {
    cout << "مرحباً في عالم STL!" << endl;
    return 0;
}

الحاويات التسلسلية (Sequence Containers)

تقوم بتخزين العناصر بترتيب خطي. يمكن الوصول للعناصر حسب موقعها.

Vector

مصفوفة ديناميكية

وصول عشوائي سريع O(1)، إضافة في النهاية O(1) amortized

List

قائمة مزدوجة

إضافة/حذف سريع في أي مكان O(1)، بطيء في الوصول العشوائي

Deque

طابور مزدوج

إضافة/حذف سريع من البداية والنهاية O(1)

1. Vector - المصفوفة الديناميكية

10
20
30
40
50
مثال Vector 
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    // إنشاء vector
    vector<int> numbers;
    
    // إضافة عناصر
    numbers.push_back(10);
    numbers.push_back(30);
    numbers.push_back(20);
    numbers.push_back(50);
    numbers.push_back(40);
    
    // الوصول للعناصر
    cout << "العنصر الأول: " << numbers[0] << endl;
    cout << "العنصر الأخير: " << numbers.back() << endl;
    
    // الحجم والسعة
    cout << "الحجم: " << numbers.size() << endl;
    cout << "السعة: " << numbers.capacity() << endl;
    
    // الطباعة باستخدام iterator
    cout << "العناصر: ";
    for(auto it = numbers.begin(); it != numbers.end(); ++it) {
        cout << *it << " ";
    }
    cout << endl;
    
    // الترتيب
    sort(numbers.begin(), numbers.end());
    
    cout << "بعد الترتيب: ";
    for(int num : numbers) {
        cout << num << " ";
    }
    cout << endl;
    
    return 0;
}

المخرجات:

العنصر الأول: 10
العنصر الأخير: 40
الحجم: 5
السعة: 8
العناصر: 10 30 20 50 40
بعد الترتيب: 10 20 30 40 50

مثال List 
#include <iostream>
#include <list>
using namespace std;

int main() {
    list<string> names;
    
    // إضافة عناصر
    names.push_back("Ahmed");
    names.push_front("Sara");
    names.push_back("Mohamed");
    names.push_front("Ali");
    
    // إدراج في منتصف القائمة
    auto it = names.begin();
    advance(it, 2);
    names.insert(it, "Omar");
    
    // طباعة القائمة
    cout << "الأسماء: ";
    for(const string& name : names) {
        cout << name << " ";
    }
    cout << endl;
    
    // حذف عنصر
    names.remove("Ali");
    cout << "بعد حذف Ali: ";
    for(const string& name : names) {
        cout << name << " ";
    }
    cout << endl;
    
    return 0;
}

الحاويات الترابطية (Associative Containers)

تقوم بتخزين العناصر بطريقة مرتبة باستخدام مفاتيح للوصول السريع.

Set

شجرة حمراء-سوداء

مجموعة من العناصر الفريدة مرتبة O(log n)

Map

قاموس key-value

أزواج مفتاح-قيمة مرتبة O(log n)

Multiset/Multimap

تسمح بالتكرار

مجموعات تسمح بتكرار العناصر

Set - المجموعة

مثال Set 
#include <iostream>
#include <set>
using namespace std;

int main() {
    set<int> numbers;
    
    // إضافة عناصر (تلقائياً مرتبة وفريدة)
    numbers.insert(50);
    numbers.insert(10);
    numbers.insert(30);
    numbers.insert(20);
    numbers.insert(10); // لن يتم إضافتها (مكررة)
    
    cout << "عناصر set: ";
    for(int num : numbers) {
        cout << num << " ";
    }
    cout << endl;
    
    // البحث
    if(numbers.find(30) != numbers.end()) {
        cout << "تم العثور على 30" << endl;
    }
    
    return 0;
}

Map - الخريطة

"أحمد" → 25
•
"سارة" → 22
•
"محمد" → 30
مثال Map 
#include <iostream>
#include <map>
#include <string>
using namespace std;

int main() {
    map<string, int> ages;
    
    // إضافة عناصر
    ages["Ahmed"] = 25;
    ages["Sara"] = 22;
    ages["Mohamed"] = 30;
    ages.insert(pair<string, int>("Ali", 28));
    
    // طباعة جميع العناصر
    cout << "الأعمار:" << endl;
    for(auto& pair : ages) {
        cout << pair.first << ": " << pair.second << " سنة" << endl;
    }
    
    // البحث
    string name = "Sara";
    if(ages.find(name) != ages.end()) {
        cout << "عمر " << name << " هو " << ages[name] << endl;
    }
    
    return 0;
}

الحاويات غير المرتبة (Unordered Containers)

تستخدم جداول التجزئة (Hash Tables) لتوفير وصول أسرع.

unordered_set

جدول تجزئة

مجموعة غير مرتبة O(1) average

unordered_map

جدول تجزئة

قاموس غير مرتب O(1) average

مثال unordered_map 
#include <iostream>
#include <unordered_map>
#include <string>
using namespace std;

int main() {
    unordered_map<string, double> products;
    
    // إضافة منتجات
    products["Laptop"] = 2500.0;
    products["Mouse"] = 50.0;
    products["Keyboard"] = 150.0;
    
    // طباعة
    cout << "المنتجات:" << endl;
    for(auto& p : products) {
        cout << p.first << ": $" << p.second << endl;
    }
    
    // البحث السريع
    string item = "Laptop";
    if(products.count(item) > 0) {
        cout << "سعر " << item << " هو $" << products[item] << endl;
    }
    
    return 0;
}

محولات الحاويات (Container Adapters)

توفر واجهات محددة للتعامل مع الحاويات الأساسية.

Stack

LIFO

آخر دخول أول خروج

Queue

FIFO

أول دخول أول خروج

Priority Queue

أولوية

عنصر ذو الأولوية القصوى أولاً

مثال Stack و Queue 
#include <iostream>
#include <stack>
#include <queue>
using namespace std;

int main() {
    // Stack - LIFO
    stack<int> st;
    st.push(10);
    st.push(20);
    st.push(30);
    
    cout << "Stack (LIFO): ";
    while(!st.empty()) {
        cout << st.top() << " ";
        st.pop();
    }
    cout << endl;
    
    // Queue - FIFO
    queue<int> q;
    q.push(10);
    q.push(20);
    q.push(30);
    
    cout << "Queue (FIFO): ";
    while(!q.empty()) {
        cout << q.front() << " ";
        q.pop();
    }
    cout << endl;
    
    // Priority Queue
    priority_queue<int> pq;
    pq.push(30);
    pq.push(10);
    pq.push(50);
    pq.push(20);
    
    cout << "Priority Queue: ";
    while(!pq.empty()) {
        cout << pq.top() << " ";
        pq.pop();
    }
    cout << endl;
    
    return 0;
}

المخرجات:

Stack (LIFO): 30 20 10
Queue (FIFO): 10 20 30
Priority Queue: 50 30 20 10

مقارنة أداء الحاويات

الحاوية الوصول إضافة/حذف في النهاية إضافة/حذف في البداية إضافة/حذف في المنتصف البحث
vector O(1) O(1)* O(n) O(n) O(n)
list O(n) O(1) O(1) O(1)** O(n)
set/map - O(log n) O(log n) O(log n) O(log n)
unordered_set/map - O(1)*** O(1)*** O(1)*** O(1)***

ملاحظات:

* vector: O(1) amortized، قد تحتاج لإعادة تخصيص الذاكرة

** list: O(1) إذا كان لديك iterator للموقع

*** unordered containers: متوسط الحالة O(1)، أسوأ حالة O(n)

تمارين وتطبيقات عملية

التمرين 1: عداد الكلمات

استخدم map لحساب تكرار الكلمات في نص:

الحل 
#include <iostream>
#include <map>
#include <string>
#include <sstream>
using namespace std;

int main() {
    string text = "hello world hello C++ world hello programming";
    map<string, int> wordCount;
    
    stringstream ss(text);
    string word;
    
    while(ss >> word) {
        wordCount[word]++;
    }
    
    cout << "تكرار الكلمات:" << endl;
    for(auto& pair : wordCount) {
        cout << pair.first << ": " << pair.second << endl;
    }
    
    return 0;
}

التمرين 2: إدارة قائمة طلاب

استخدم vector و set لإدارة قائمة الطلاب:

الحل 
#include <iostream>
#include <vector>
#include <set>
#include <string>
#include <algorithm>
using namespace std;

class StudentManager {
private:
    vector<string> students;
    set<string> uniqueStudents;
    
public:
    void addStudent(string name) {
        students.push_back(name);
        uniqueStudents.insert(name);
        cout << name << " added successfully" << endl;
    }
    
    void removeStudent(string name) {
        auto it = find(students.begin(), students.end(), name);
        if(it != students.end()) {
            students.erase(it);
            uniqueStudents.erase(name);
            cout << name << " removed" << endl;
        }
    }
    
    void displayAll() {
        cout << "All students: ";
        for(const string& s : students) {
            cout << s << " ";
        }
        cout << endl;
    }
    
    void displayUnique() {
        cout << "Unique students: ";
        for(const string& s : uniqueStudents) {
            cout << s << " ";
        }
        cout << endl;
    }
};

int main() {
    StudentManager mgr;
    
    mgr.addStudent("Ahmed");
    mgr.addStudent("Sara");
    mgr.addStudent("Ahmed");
    mgr.addStudent("Mohamed");
    mgr.addStudent("Sara");
    
    mgr.displayAll();
    mgr.displayUnique();
    
    mgr.removeStudent("Ahmed");
    mgr.displayAll();
    
    return 0;
}
