القالبات (Templates)

البرمجة العامة (Generic Programming) – اكتب مرة واحدة .. استخدم مع أي نوع

القالبات (Templates) هي أساس البرمجة العامة (Generic Programming) في ++C. تتيح لك كتابة دوال وأصناف تعمل مع أي نوع بيانات دون تكرار الكود. القالبات هي السبب وراء المكتبة القياسية (STL) بكاملها.

قالب دالة

دالة واحدة تعمل مع أنواع متعددة (int, double, string, ...) دون إعادة كتابتها.

قالب صنف

صنف واحد يمكن استخدامه لإنشاء كائنات من أي نوع (مثل vector<int>، vector<string>).

metaprogramming

القالبات تمكن من البرمجة أثناء الترجمة (Template Metaprogramming) لحسابات فائقة الكفاءة.

أولاً: قوالب الدوال (Function Templates)

قالب الدالة يسمح لك بكتابة دالة واحدة تعمل مع أي نوع. يتم تحديد النوع عبر معامل القالب (template parameter).

مثال: دالة swap عامة

#include <iostream>
using namespace std;

// تعريف قالب دالة
template <typename T>
void mySwap(T& a, T& b) {
    T temp = a;
    a = b;
    b = temp;
}

int main() {
    int x = 5, y = 10;
    mySwap(x, y);  // T يُستنتج تلقائياً كـ int
    cout << "x = " << x << ", y = " << y << endl; // x=10, y=5

    double d1 = 1.5, d2 = 3.7;
    mySwap(d1, d2);  // T يُستنتج كـ double
    cout << "d1 = " << d1 << ", d2 = " << d2 << endl;

    string s1 = "Ahmed", s2 = "Sara";
    mySwap(s1, s2);  // T يُستنتج كـ string
    cout << "s1 = " << s1 << ", s2 = " << s2 << endl;
    return 0;
}

استخدام عدة معاملات قالب

#include <iostream>
using namespace std;

// قالب مع نوعين مختلفين
template <typename T1, typename T2>
void printPair(T1 a, T2 b) {
    cout << "(" << a << ", " << b << ")" << endl;
}

int main() {
    printPair(5, 3.14);          // T1=int, T2=double
    printPair("Ahmed", 25);       // T1=const char*, T2=int
    printPair(true, "نعم");       // T1=bool, T2=const char*
    return 0;
}

استنتاج الأنواع: المترجم يستنتج أنواع معاملات القالب من المعاملات الفعلية التي ترسلها للدالة. يمكن أيضاً تحديد الأنواع صراحةً: mySwap<int>(x, y);

ثانياً: قوالب الأصناف (Class Templates)

قالب الصنف يسمح لك بإنشاء أصناف عامة تعمل مع أي نوع. مثال كلاسيكي: حاوية بسيطة (Box) تخزن قيمة من أي نوع.

مثال: صنف Box<T>

#include <iostream>
using namespace std;

// تعريف قالب صنف
template <typename T>
class Box {
private:
    T content;
public:
    Box(T value) : content(value) {}

    void set(T value) {
        content = value;
    }

    T get() const {
        return content;
    }

    void print() const {
        cout << "الصندوق يحوي: " << content << endl;
    }
};

int main() {
    Box<int> intBox(123);          // صندوق للأعداد
    intBox.print();

    Box<string> stringBox("مرحبا"); // صندوق للنصوص
    stringBox.print();

    Box<double> doubleBox(3.14);
    doubleBox.print();

    return 0;
}

صنف مع عدة أنواع

template <typename T1, typename T2>
class Pair {
private:
    T1 first;
    T2 second;
public:
    Pair(T1 f, T2 s) : first(f), second(s) {}

    T1 getFirst() const { return first; }
    T2 getSecond() const { return second; }

    void display() const {
        cout << "[" << first << ", " << second << "]" << endl;
    }
};

int main() {
    Pair<int, string> student(12345, "أحمد");
    student.display();

    Pair<string, double> product("حاسوب", 2500.75);
    product.display();

    return 0;
}

قيم افتراضية لمعاملات القالب

#include <iostream>
using namespace std;

template <typename T = int>  // نوع افتراضي: int
class Container {
private:
    T value;
public:
    Container(T v) : value(v) {}
    T get() const { return value; }
};

int main() {
    Container<> defaultContainer(100);     // يستخدم int
    Container<double> doubleContainer(3.14); // يستخدم double

    cout << defaultContainer.get() << endl;   // 100
    cout << doubleContainer.get() << endl;    // 3.14
    return 0;
}

معاملات قالب غير نوعية (Non-type Template Parameters)

يمكن أن تكون معاملات القالب قيماً ثابتة (أعداد، مؤشرات، مراجع) وليس فقط أنواعاً. هذا يسمح بإنشاء أصناف ودوال تعتمد على قيم ثابتة وقت الترجمة.

#include <iostream>
using namespace std;

// قالب صفيف حجمه ثابت
template <typename T, int Size>
class FixedArray {
private:
    T arr[Size];
public:
    T& operator[](int index) {
        return arr[index];
    }

    int size() const { return Size; }
};

int main() {
    FixedArray<int, 5> numbers;  // مصفوفة من 5 أعداد
    for (int i = 0; i < numbers.size(); i++) {
        numbers[i] = i * 10;
    }

    FixedArray<string, 3> names;
    names[0] = "أحمد";
    names[1] = "فاطمة";
    names[2] = "سعيد";

    for (int i = 0; i < 3; i++) {
        cout << names[i] << " ";
    }
    return 0;
}

تخصص القوالب (Template Specialization)

أحياناً نريد سلوكاً مختلفاً لنوع معين. هنا يأتي دور التخصص (specialization). يمكننا تخصيص القالب لأنواع محددة.

تخصص كامل (Full Specialization)

#include <iostream>
using namespace std;

// قالب عام
template <typename T>
class Printer {
public:
    void print(T value) {
        cout << "القيمة: " << value << endl;
    }
};

// تخصص كامل للنوع bool
template <>
class Printer<bool> {
public:
    void print(bool value) {
        cout << "القيمة المنطقية: " << (value ? "صحيح" : "خاطئ") << endl;
    }
};

// تخصص كامل للنوع char
template <>
class Printer<char> {
public:
    void print(char value) {
        cout << "الحرف: '" << value << "' - رمز ASCII: " << (int)value << endl;
    }
};

int main() {
    Printer<int> p1;
    p1.print(100);

    Printer<bool> p2;
    p2.print(true);

    Printer<char> p3;
    p3.print('A');

    return 0;
}

مفاهيم متقدمة

المفهومالوصف
تخصص جزئي (Partial Specialization)تخصيص قالب لأنماط معينة من الأنواع (مثلاً: T* للمؤشرات).
قالب القوالب (Template Template Parameters)معامل قالب يكون هو نفسه قالباً.
Variadic Templatesقوالب تقبل عدداً متغيراً من المعاملات (C++11).
SFINAEآلية تسمح باختيار أفضل قالب مناسب دون أخطاء.
Concepts (C++20)تقييد الأنواع التي يمكن استخدامها مع القالب.

تطبيقات عملية

دالة max عامة

template <typename T>
T max(T a, T b) {
    return (a > b) ? a : b;
}

// استخدام
int imax = max(10, 20);
double dmax = max(3.14, 2.71);

قائمة مرتبطة عامة

template <typename T>
class Node {
public:
    T data;
    Node* next;
    Node(T val) : data(val), next(nullptr) {}
};

// يمكن إنشاء Node<int>، Node<string>...

تمرين: اكتب قالب دالة findMax تأخذ مصفوفة (array) وحجمها وتعيد أكبر عنصر. ثم اختبرها مع أنواع مختلفة (int, double, string).

عرض الحل 
#include <iostream>
#include <string>
using namespace std;

template <typename T>
T findMax(T arr[], int size) {
    T maxVal = arr[0];
    for (int i = 1; i < size; i++) {
        if (arr[i] > maxVal) {
            maxVal = arr[i];
        }
    }
    return maxVal;
}

int main() {
    int intArr[] = {5, 2, 9, 1, 7};
    double dblArr[] = {3.14, 2.71, 1.618, 0.99};
    string strArr[] = {"Ahmed", "Sara", "Mohamed", "Fatima"};

    cout << "أكبر int: " << findMax(intArr, 5) << endl;
    cout << "أكبر double: " << findMax(dblArr, 4) << endl;
    cout << "أكبر string: " << findMax(strArr, 4) << endl;

    return 0;
}

أفضل الممارسات مع القوالب

  • ضع تعريف القوالب بالكامل في ملفات الرأس (.h أو .hpp).
  • استخدم typename أو class بالتبادل لمعاملات النوع.
  • استخدم const T& بدلاً من T للأنواع الكبيرة تجنباً للنسخ.
  • افهم أخطاء القوالب (قد تكون طويلة ومعقدة).
  • استخدم Concepts (C++20) لتقييد الأنواع إذا أمكن.
  • القوالب تولد كوداً مستقلاً لكل نوع، لذا استخدمها بحكمة لتجنب تضخم الكود.
