مقدمة في البرمجة المتزامنة

البرمجة المتزامنة (Concurrency) هي قدرة البرنامج على تنفيذ عمليات متعددة في نفس الوقت، مما يزيد من كفاءة استخدام المعالج ويحسن أداء التطبيقات [citation:8].

إنشاء وإدارة الخيوط (Threads)

توفر C++11 مكتبة std::thread لإدارة الخيوط بطريقة آمنة ومنفصلة عن نظام التشغيل [citation:8].

مثال 1: إنشاء خيط بسيط 
#include <iostream>
#include <thread>
using namespace std;

void helloFunction() {
    cout << "مرحباً من الخيط المنفصل!" << endl;
}

int main() {
    // إنشاء خيط جديد
    thread t1(helloFunction);
    
    cout << "مرحباً من الخيط الرئيسي!" << endl;
    
    // انتظار انتهاء الخيط
    t1.join();
    
    return 0;
}

المخرجات:

مرحباً من الخيط الرئيسي!
مرحباً من الخيط المنفصل!

طرق مختلفة لإنشاء الخيوط

مثال 2: طرق إنشاء الخيوط 
#include <iostream>
#include <thread>
using namespace std;

// 1. باستخدام دالة عادية
void printMessage(int id) {
    cout << "خيط " << id << " قيد التشغيل" << endl;
}

// 2. كلاس مع ()operator
class Functor {
public:
    void operator()(int id) {
        cout << "خيط " << id << " من functor" << endl;
    }
};

int main() {
    // باستخدام دالة عادية
    thread t1(printMessage, 1);
    
    // باستخدام لامبدا
    thread t2([](int id) {
        cout << "خيط " << id << " من lambda" << endl;
    }, 2);
    
    // باستخدام functor
    Functor f;
    thread t3(f, 3);
    
    t1.join();
    t2.join();
    t3.join();
    
    return 0;
}

إدارة الخيوط

الدالة الوصف مثال
join() انتظار انتهاء الخيط [citation:10] thread1.join();
detach() فصل الخيط ليعمل بشكل مستقل [citation:10] thread1.detach();
joinable() التحقق إمكانية انتظار الخيط [citation:10] if(t.joinable()) t.join();
get_id() الحصول على معرف الخيط auto id = t.get_id();
مثال 3: إدارة الخيوط 
#include <iostream>
#include <thread>
#include <chrono>
#include <vector>
using namespace std;

void worker(int id) {
    cout << "خيط " << id << " بدأ العمل" << endl;
    this_thread::sleep_for(chrono::milliseconds(100 * id));
    cout << "خيط " << id << " انتهى" << endl;
}

int main() {
    vector<thread> threads;
    
    // إنشاء 5 خيوط
    for(int i = 1; i <= 5; i++) {
        threads.emplace_back(worker, i);
    }
    
    cout << "عدد الخيوط النشطة: " << thread::hardware_concurrency() << endl;
    
    // انتظار جميع الخيوط
    for(auto& t : threads) {
        if(t.joinable()) {
            t.join();
        }
    }
    
    cout << "جميع الخيوط انتهت" << endl;
    
    return 0;
}

سباق البيانات (Data Race)

سباق البيانات يحدث عندما يحاول أكثر من خيط الوصول لنفس المتغير في نفس الوقت، مع وجود عملية كتابة واحدة على الأقل [citation:1][citation:10].

مثال 4: مشكلة سباق البيانات 
#include <iostream>
#include <thread>
#include <vector>
using namespace std;

int counter = 0; // متغير مشترك

void increment() {
    for(int i = 0; i < 100000; i++) {
        counter++; // ليست عملية ذرية!
    }
}

int main() {
    vector<thread> threads;
    
    // إنشاء 10 خيوط تزيد العداد
    for(int i = 0; i < 10; i++) {
        threads.emplace_back(increment);
    }
    
    for(auto& t : threads) {
        t.join();
    }
    
    cout << "القيمة النهائية: " << counter << endl;
    cout << "القيمة المتوقعة: " << 10 * 100000 << endl;
    
    return 0;
}

المخرجات:

القيمة النهائية: 584392 (تختلف في كل مرة)
القيمة المتوقعة: 1000000

سبب المشكلة:

العملية counter++ ليست ذرية. تتكون من: قراءة القيمة، زيادتها، كتابتها. يمكن أن يحدث تداخل بين الخيوط في هذه الخطوات [citation:1].

المزامنة باستخدام Mutex

الـ Mutex (Mutual Exclusion) يضمن أن خيطاً واحداً فقط يمكنه الوصول إلى مورد مشترك في نفس الوقت [citation:9].

الخيط 1:
محاولة قفل
قفل
تعديل
فتح
الخيط 2:
محاولة قفل
انتظار...
انتظار...
قفل
مثال 5: حل المشكلة باستخدام Mutex 
#include <iostream>
#include <thread>
#include <mutex>
#include <vector>
using namespace std;

int counter = 0;
mutex mtx; // كائن المزامنة

void safeIncrement() {
    for(int i = 0; i < 100000; i++) {
        mtx.lock(); // قفل المورد
        counter++;  // منطقة حرجة
        mtx.unlock(); // فتح المورد
    }
}

int main() {
    vector<thread> threads;
    
    for(int i = 0; i < 10; i++) {
        threads.emplace_back(safeIncrement);
    }
    
    for(auto& t : threads) {
        t.join();
    }
    
    cout << "القيمة النهائية: " << counter << endl;
    
    return 0;
}

المخرجات:

القيمة النهائية: 1000000

أنواع الأقفال (Locks)

توفر C++ عدة أنواع من الأقفال لإدارة المزامنة بشكل آمن [citation:9][citation:10].

1. lock_guard - الأبسط والأكثر أماناً

مثال 6: استخدام lock_guard 
#include <iostream>
#include <thread>
#include <mutex>
#include <vector>
using namespace std;

mutex mtx;
int safeCounter = 0;

void incrementWithGuard() {
    for(int i = 0; i < 100000; i++) {
        lock_guard<mutex> guard(mtx); // قفل تلقائي
        safeCounter++; // آمن حتى لو حدث استثناء
        // يتم الفتح تلقائياً عند خروج guard من النطاق
    }
}

int main() {
    vector<thread> threads;
    
    for(int i = 0; i < 10; i++) {
        threads.emplace_back(incrementWithGuard);
    }
    
    for(auto& t : threads) t.join();
    
    cout << "القيمة: " << safeCounter << endl;
    
    return 0;
}

2. unique_lock - مرونة أكبر

مثال 7: استخدام unique_lock 
#include <iostream>
#include <thread>
#include <mutex>
#include <chrono>
using namespace std;

mutex mtx;

void task() {
    unique_lock<mutex> lock(mtx, defer_lock); // إنشاء بدون قفل
    
    // القيام ببعض العمل خارج القفل
    this_thread::sleep_for(chrono::milliseconds(100));
    
    lock.lock(); // قفل يدوياً عند الحاجة
    cout << "منطقة حرجة للخيط " << this_thread::get_id() << endl;
    lock.unlock(); // فتح يدوياً
    
    // عمل آخر بدون قفل
    this_thread::sleep_for(chrono::milliseconds(100));
    
    lock.lock(); // قفل مرة أخرى
    cout << "منطقة حرجة ثانية" << endl;
    // يتم الفتح تلقائياً في النهاية
}

int main() {
    thread t1(task);
    thread t2(task);
    
    t1.join();
    t2.join();
    
    return 0;
}

مقارنة أنواع الأقفال

النوع الوصف متى تستخدم
lock_guard قفل بسيط مع فتح تلقائي عندما تحتاج قفل طوال النطاق
unique_lock مرن، يمكن قفل وفتح يدوياً عند الحاجة لفتح/قفل متعدد
scoped_lock (C++17) لقفل عدة mutex معاً [citation:10] لتجنب deadlock مع عدة موارد

تجنب Deadlock

مثال 8: تجنب deadlock باستخدام scoped_lock 
#include <iostream>
#include <thread>
#include <mutex>
using namespace std;

mutex mtx1, mtx2;

void badTransfer() {
    // قد يسبب deadlock!
    lock_guard<mutex> lock1(mtx1);
    this_thread::sleep_for(chrono::milliseconds(100));
    lock_guard<mutex> lock2(mtx2); // قد ينتظر للأبد
}

void goodTransfer() {
    // C++17: قفل عدة mutex معاً
    scoped_lock lock(mtx1, mtx2);
    // آمن من deadlock
}

int main() {
    thread t1(badTransfer);
    thread t2(badTransfer);
    
    t1.join();
    t2.join(); // قد لا يصل هنا!
    
    return 0;
}

المتغيرات الشرطية (Condition Variables)

تستخدم لتزامن الخيوط بناءً على حدوث شرط معين، مثل نموذج المنتج-المستهلك [citation:9].

مثال 9: نموذج المنتج-المستهلك 
#include <iostream>
#include <thread>
#include <mutex>
#include <condition_variable>
#include <queue>
using namespace std;

queue<int> buffer;
mutex mtx;
condition_variable cv;
const int MAX_BUFFER = 5;

void producer() {
    for(int i = 1; i <= 10; i++) {
        unique_lock<mutex> lock(mtx);
        
        // انتظار حتى يصبح هناك مساحة
        cv.wait(lock, []{ return buffer.size() < MAX_BUFFER; });
        
        buffer.push(i);
        cout << "منتج: تم إنتاج " << i << endl;
        
        lock.unlock();
        cv.notify_one(); // إشعار المستهلك
        
        this_thread::sleep_for(chrono::milliseconds(100));
    }
}

void consumer() {
    for(int i = 1; i <= 10; i++) {
        unique_lock<mutex> lock(mtx);
        
        // انتظار حتى يصبح هناك منتجات
        cv.wait(lock, []{ return !buffer.empty(); });
        
        int item = buffer.front();
        buffer.pop();
        cout << "مستهلك: تم استهلاك " << item << endl;
        
        lock.unlock();
        cv.notify_one(); // إشعار المنتج
        
        this_thread::sleep_for(chrono::milliseconds(150));
    }
}

int main() {
    thread prod(producer);
    thread cons(consumer);
    
    prod.join();
    cons.join();
    
    return 0;
}

شرح المتغيرات الشرطية:

• wait(): ينتظر حتى يتحقق الشرط [citation:9]

• notify_one(): إيقاظ خيط واحد منتظر

• notify_all(): إيقاظ جميع الخيوط المنتظرة

• تستخدم مع unique_lock للمرونة

العمليات الذرية (Atomic Operations)

الأنواع الذرية توفر عمليات غير قابلة للمقاطعة، مما يلغي الحاجة للأقفال في العمليات البسيطة [citation:10].

مثال 10: استخدام atomic 
#include <iostream>
#include <thread>
#include <atomic>
#include <vector>
using namespace std;

atomic<int> atomicCounter(0); // عداد ذري
int normalCounter = 0;

void incrementAtomic() {
    for(int i = 0; i < 100000; i++) {
        atomicCounter++; // عملية ذرية
    }
}

void incrementNormal() {
    for(int i = 0; i < 100000; i++) {
        normalCounter++; // غير ذرية - غير آمن
    }
}

int main() {
    vector<thread> threads;
    
    // تجربة مع atomic
    for(int i = 0; i < 10; i++) {
        threads.emplace_back(incrementAtomic);
    }
    for(auto& t : threads) t.join();
    cout << "Atomic: " << atomicCounter << endl;
    
    threads.clear();
    atomicCounter = 0;
    
    // تجربة مع normal
    for(int i = 0; i < 10; i++) {
        threads.emplace_back(incrementNormal);
    }
    for(auto& t : threads) t.join();
    cout << "Normal: " << normalCounter << endl;
    
    return 0;
}

المخرجات:

Atomic: 1000000
Normal: 584392 (قيمة غير متوقعة)
النوع الذري الوصف
atomic<bool> قيمة منطقية ذرية
atomic<int> عدد صحيح ذري
atomic<long long> عدد كبير ذري
atomic_flag أبسط نوع ذري - لعمليات القفل

تمارين وتطبيقات عملية

التمرين 1: عداد آمن متعدد الخيوط

قم بإنشاء عداد آمن للاستخدام مع عدة خيوط:

الحل 
#include <iostream>
#include <thread>
#include <mutex>
#include <vector>
using namespace std;

class SafeCounter {
private:
    int value;
    mutex mtx;
    
public:
    SafeCounter() : value(0) {}
    
    void increment() {
        lock_guard<mutex> lock(mtx);
        value++;
    }
    
    void decrement() {
        lock_guard<mutex> lock(mtx);
        value--;
    }
    
    int getValue() {
        lock_guard<mutex> lock(mtx);
        return value;
    }
};

void worker(SafeCounter& counter, int id) {
    for(int i = 0; i < 1000; i++) {
        if(i % 2 == 0)
            counter.increment();
        else
            counter.decrement();
    }
    cout << "خيط " << id << " انتهى" << endl;
}

int main() {
    SafeCounter counter;
    vector<thread> threads;
    
    for(int i = 0; i < 10; i++) {
        threads.emplace_back(worker, ref(counter), i);
    }
    
    for(auto& t : threads) {
        t.join();
    }
    
    cout << "القيمة النهائية: " << counter.getValue() << endl;
    
    return 0;
}

التمرين 2: تجمع خيوط بسيط

قم بإنشاء تجمع خيوط بسيط لتنفيذ المهام:

الحل 
#include <iostream>
#include <thread>
#include <mutex>
#include <condition_variable>
#include <queue>
#include <functional>
#include <vector>
using namespace std;

class ThreadPool {
private:
    vector<thread> workers;
    queue<function<void()>> tasks;
    mutex queueMutex;
    condition_variable condition;
    bool stop;
    
public:
    ThreadPool(size_t numThreads) : stop(false) {
        for(size_t i = 0; i < numThreads; i++) {
            workers.emplace_back([this] {
                while(true) {
                    function<void()> task;
                    {
                        unique_lock<mutex> lock(queueMutex);
                        condition.wait(lock, [this] {
                            return stop || !tasks.empty();
                        });
                        
                        if(stop && tasks.empty()) return;
                        
                        task = move(tasks.front());
                        tasks.pop();
                    }
                    task();
                }
            });
        }
    }
    
    void enqueue(function<void()> task) {
        {
            unique_lock<mutex> lock(queueMutex);
            tasks.push(task);
        }
        condition.notify_one();
    }
    
    ~ThreadPool() {
        {
            unique_lock<mutex> lock(queueMutex);
            stop = true;
        }
        condition.notify_all();
        for(auto& worker : workers) {
            worker.join();
        }
    }
};

int main() {
    ThreadPool pool(4);
    
    for(int i = 0; i < 8; i++) {
        pool.enqueue([i] {
            cout << "تنفيذ المهمة " << i 
                 << " في الخيط " << this_thread::get_id() << endl;
            this_thread::sleep_for(chrono::milliseconds(100));
        });
    }
    
    this_thread::sleep_for(chrono::seconds(1));
    return 0;
}
