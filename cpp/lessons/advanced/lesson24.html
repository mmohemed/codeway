معالجة الأخطاء (Exception Handling)

try / throw / catch – برامج قوية وآمنة

معالجة الأخطاء (Exception Handling) هي آلية تسمح للبرنامج بالتعامل مع الأخطاء أو الحالات الاستثنائية (مثل قسمة على صفر، فشل في فتح ملف، نفاد الذاكرة) بطريقة منظمة، دون انهيار البرنامج فجأة.

try

كتلة الكود التي قد تسبب خطأ. يتم وضعها داخل try لمراقبة الاستثناءات.

throw

يستخدم لإلقاء (رمي) استثناء عند حدوث خطأ. يمكن رمي أي نوع (int, string, class).

catch

كتلة تلتقط الاستثناء وتعالجه. يمكن أن يكون عدة catch لأنواع مختلفة.

1. البنية الأساسية (try / throw / catch)

#include <iostream>
using namespace std;

int main() {
    try {
        // كود قد يسبب خطأ
        int age;
        cout << "أدخل عمرك: ";
        cin >> age;

        if (age < 0) {
            throw "العمر لا يمكن أن يكون سالباً!";  // رمي استثناء (نص)
        }

        if (age > 150) {
            throw 150;  // رمي استثناء (رقم)
        }

        cout << "عمرك: " << age << endl;
    }
    catch (const char* msg) {  // التقاط الاستثناء النصي
        cout << "خطأ: " << msg << endl;
    }
    catch (int maxAge) {  // التقاط الاستثناء الرقمي
        cout << "خطأ: العمر لا يمكن أن يتجاوز " << maxAge << endl;
    }

    cout << "البرنامج مستمر..." << endl;
    return 0;
}

2. التقاط أنواع متعددة من الاستثناءات

يمكن أن يكون لديك عدة catch blocks للتعامل مع أنواع مختلفة من الاستثناءات. يتم تنفيذ أول catch يطابق نوع الاستثناء.

#include <iostream>
#include <stdexcept>  // للاستثناءات القياسية
using namespace std;

double divide(double a, double b) {
    if (b == 0) {
        throw runtime_error("خطأ: قسمة على صفر!");
    }
    if (b < 0) {
        throw "لا يمكن القسمة على سالب";  // نص
    }
    return a / b;
}

int main() {
    try {
        double result = divide(10, 0);
        cout << "النتيجة: " << result << endl;
    }
    catch (const runtime_error& e) {
        cout << "استثناء قياسي: " << e.what() << endl;
    }
    catch (const char* msg) {
        cout << "استثناء نصي: " << msg << endl;
    }
    catch (...) {  // يلتقط أي استثناء آخر
        cout << "حدث خطأ غير معروف!" << endl;
    }
    return 0;
}

catch (...): يستخدم لالتقاط أي استثناء بغض النظر عن نوعه. يفضل وضعه في النهاية كملاذ أخير.

3. الاستثناءات القياسية (Standard Exceptions)

توفر ++C مكتبة <stdexcept> تحتوي على أصناف استثناءات جاهزة يمكن استخدامها أو وراثتها.

الاستثناءالوصف
exceptionالصنف الأساسي لجميع الاستثناءات القياسية.
runtime_errorخطأ يحدث أثناء تنفيذ البرنامج.
logic_errorخطأ في المنطق البرمجي (مثل انتهاك precondition).
invalid_argumentتم تمرير معامل غير صحيح لدالة.
out_of_rangeمحاولة الوصول إلى عنصر خارج حدود المصفوفة.
bad_allocفشل في تخصيص الذاكرة (من new).
#include <iostream>
#include <stdexcept>
#include <vector>
using namespace std;

int main() {
    try {
        vector<int> vec(5);
        vec.at(10) = 100;  // at() تتحقق من الحدود وترمي out_of_range
    }
    catch (const out_of_range& e) {
        cout << "خطأ نطاق: " << e.what() << endl;
    }

    try {
        int* p = new int[100000000000];  // قد يفشل
    }
    catch (const bad_alloc& e) {
        cout << "فشل تخصيص ذاكرة: " << e.what() << endl;
    }
    return 0;
}

4. إنشاء استثناءات مخصصة (Custom Exceptions)

يمكنك إنشاء أصناف استثناء خاصة بك بوراثة من std::exception أو runtime_error.

#include <iostream>
#include <stdexcept>
#include <string>
using namespace std;

// استثناء مخصص
class InsufficientFundsException : public runtime_error {
public:
    InsufficientFundsException(double balance, double amount)
        : runtime_error("رصيد غير كافٍ"), m_balance(balance), m_amount(amount) {}

    double getBalance() const { return m_balance; }
    double getAmount() const { return m_amount; }

private:
    double m_balance;
    double m_amount;
};

class BankAccount {
private:
    double balance;
public:
    BankAccount(double b) : balance(b) {}

    void withdraw(double amount) {
        if (amount > balance) {
            throw InsufficientFundsException(balance, amount);
        }
        balance -= amount;
        cout << "تم السحب. الرصيد الجديد: " << balance << endl;
    }
};

int main() {
    BankAccount acc(500);
    try {
        acc.withdraw(600);
    }
    catch (const InsufficientFundsException& e) {
        cout << "خطأ: " << e.what() << endl;
        cout << "الرصيد الحالي: " << e.getBalance()
             << ", المبلغ المطلوب: " << e.getAmount() << endl;
    }
    return 0;
}

5. ضمانات الاستثناء (Exception Safety)

عند كتابة كود قد يرمي استثناءات، يجب أن نضمن أن البرنامج يبقى في حالة سليمة. هناك ثلاثة مستويات:

المستوىالوصف
Basic guaranteeلا يوجد تسرب للموارد، وكل الكائنات في حالة سليمة (قد تكون غير محددة).
Strong guaranteeإذا فشلت العملية، يعود البرنامج إلى الحالة التي كان عليها قبل العملية (commit or rollback).
No-throw guaranteeالدالة لا ترمي استثناءات أبداً (تستخدم noexcept).

6. المواصفة noexcept (C++11)

تستخدم لتحديد أن الدالة لا ترمي استثناءات. يساعد المترجم في التحسينات ويزيد الأمان.

#include <iostream>
using namespace std;

// هذه الدالة تعد بعدم رمي استثناءات
int add(int a, int b) noexcept {
    return a + b;
}

// دالة قد ترمي استثناء
int divide(int a, int b) {
    if (b == 0) throw "division by zero";
    return a / b;
}

int main() {
    cout << "add is noexcept? " << noexcept(add(1, 2)) << endl;  // 1 (true)
    cout << "divide is noexcept? " << noexcept(divide(1, 2)) << endl; // 0 (false)
    return 0;
}

7. إعادة رمي الاستثناء (Rethrow)

أحياناً نريد التقاط استثناء، القيام ببعض الإجراءات (مثل تسجيل الخطأ)، ثم إعادة رميه لمستوى أعلى ليعالجه كتلة أخرى.

#include <iostream>
#include <fstream>
using namespace std;

void processFile(const string& filename) {
    try {
        ifstream file(filename);
        if (!file) {
            throw runtime_error("لا يمكن فتح الملف");
        }
        // معالجة الملف...
    }
    catch (const runtime_error& e) {
        cout << "تسجيل الخطأ: " << e.what() << endl;
        throw;  // إعادة رمي نفس الاستثناء
    }
}

int main() {
    try {
        processFile("non_existent.txt");
    }
    catch (const runtime_error& e) {
        cout << "تم التعامل مع الخطأ في main: " << e.what() << endl;
    }
    return 0;
}

تطبيقات عملية

آلة حاسبة آمنة

double safeDivide(double a, double b) {
    if (b == 0) 
        throw invalid_argument("القسمة على صفر");
    return a / b;
}

try {
    cout << safeDivide(5, 0);
} catch (const exception& e) {
    cout << "خطأ: " << e.what();
}

التحقق من المدخلات

int readPositiveInt() {
    int x;
    cin >> x;
    if (cin.fail()) {
        cin.clear();
        throw runtime_error("إدخال غير صحيح");
    }
    if (x <= 0)

تمرين: اكتب برنامجاً لإدارة حساب بنكي. أنشئ صنف BankAccount برصيد، وأضف دوال الإيداع والسحب. تأكد من أن السحب لا يتجاوز الرصيد باستخدام استثناء مخصص.

عرض الحل 
#include <iostream>
#include <stdexcept>
using namespace std;

class InsufficientBalanceException : public runtime_error {
public:
    InsufficientBalanceException(double balance, double amount)
        : runtime_error("رصيد غير كافٍ"), balance(balance), amount(amount) {}
    double getBalance() const { return balance; }
    double getAmount() const { return amount; }
private:
    double balance, amount;
};

class BankAccount {
private:
    double balance;
public:
    BankAccount(double initial) : balance(initial) {
        if (initial < 0) throw invalid_argument("الرصيد الأولي لا يمكن أن يكون سالباً");
    }

    void deposit(double amount) {
        if (amount <= 0) throw invalid_argument("مبلغ الإيداع يجب أن يكون موجباً");
        balance += amount;
        cout << "تم الإيداع. الرصيد الجديد: " << balance << endl;
    }

    void withdraw(double amount) {
        if (amount <= 0) throw invalid_argument("مبلغ السحب يجب أن يكون موجباً");
        if (amount > balance) throw InsufficientBalanceException(balance, amount);
        balance -= amount;
        cout << "تم السحب. الرصيد الجديد: " << balance << endl;
    }

    double getBalance() const { return balance; }
};

int main() {
    try {
        BankAccount acc(1000);
        acc.deposit(500);
        acc.withdraw(2000);
    }
    catch (const InsufficientBalanceException& e) {
        cout << "خطأ: " << e.what() << endl;
        cout << "الرصيد: " << e.getBalance() << ", المطلوب: " << e.getAmount() << endl;
    }
    catch (const exception& e) {
        cout << "خطأ عام: " << e.what() << endl;
    }
    return 0;
}

أفضل الممارسات في معالجة الأخطاء

  • استخدم الاستثناءات للأخطاء الاستثنائية فقط، ليس للتحكم العادي بالتدفق.
  • التقط الاستثناءات بالمرجع (const exception&) لتجنب النسخ والقطع (slicing).
  • نظم كتل catch من الأكثر تخصصاً إلى الأكثر عمومية.
  • استخدم الاستثناءات القياسية أو اشتق منها لاستثناءاتك المخصصة.
  • تأكد من أن المدمرات لا ترمي استثناءات (استخدم noexcept).
  • وثق الاستثناءات التي قد ترميها دوالك.
