مقدمة عن خوارزميات STL

توفر STL مجموعة غنية من الخوارزميات الجاهزة للاستخدام مع الحاويات المختلفة. هذه الخوارزميات مصممة للعمل مع المكررات (iterators) وتتميز بالكفاءة العالية.

كفاءة عالية

O(n) to O(log n)

خوارزميات محسّنة لأقصى أداء

قابلة لإعادة الاستخدام

تعمل مع أي حاوية تدعم المكررات

مختبرة ومضمونة

خوارزميات موثوقة من المكتبة القياسية

تضمين مكتبة الخوارزميات 
#include <iostream>
#include <algorithm>
#include <vector>
#include <numeric>
using namespace std;

int main() {
    vector<int> numbers = {5, 2, 8, 1, 9, 3};
    
    // استخدام الخوارزميات
    sort(numbers.begin(), numbers.end());
    
    for(int num : numbers) {
        cout << num << " ";
    }
    
    return 0;
}

sort() - الترتيب التصاعدي 
#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    vector<int> numbers = {50, 20, 60, 10, 30, 40};
    
    cout << "قبل الترتيب: ";
    for(int n : numbers) cout << n << " ";
    cout << endl;
    
    // ترتيب تصاعدي
    sort(numbers.begin(), numbers.end());
    
    cout << "بعد الترتيب التصاعدي: ";
    for(int n : numbers) cout << n << " ";
    cout << endl;
    
    // ترتيب تنازلي
    sort(numbers.begin(), numbers.end(), greater<int>());
    
    cout << "بعد الترتيب التنازلي: ";
    for(int n : numbers) cout << n << " ";
    cout << endl;
    
    return 0;
}

partial_sort() - ترتيب جزئي 
#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    vector<int> numbers = {50, 20, 60, 10, 30, 40, 70, 80};
    
    cout << "المصفوفة الأصلية: ";
    for(int n : numbers) cout << n << " ";
    cout << endl;
    
    // ترتيب أول 4 عناصر فقط
    partial_sort(numbers.begin(), numbers.begin() + 4, numbers.end());
    
    cout << "بعد الترتيب الجزئي (أول 4): ";
    for(int n : numbers) cout << n << " ";
    cout << endl;
    
    return 0;
}

العثور على 40 في مصفوفة مرتبة

find() - بحث خطي 
#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    vector<int> numbers = {10, 20, 30, 40, 50, 30, 60};
    
    // البحث عن 30
    auto it = find(numbers.begin(), numbers.end(), 30);
    
    if(it != numbers.end()) {
        cout << "تم العثور على 30 في الموقع: " 
             << (it - numbers.begin()) << endl;
    }
    
    // البحث عن 100
    it = find(numbers.begin(), numbers.end(), 100);
    
    if(it == numbers.end()) {
        cout << "لم يتم العثور على 100" << endl;
    }
    
    return 0;
}
binary_search() - بحث ثنائي 
#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    vector<int> numbers = {10, 20, 30, 40, 50, 60, 70};
    
    // البحث الثنائي يتطلب مصفوفة مرتبة
    bool found = binary_search(numbers.begin(), numbers.end(), 40);
    
    cout << "هل 40 موجود؟ " << (found ? "نعم" : "لا") << endl;
    
    found = binary_search(numbers.begin(), numbers.end(), 45);
    cout << "هل 45 موجود؟ " << (found ? "نعم" : "لا") << endl;
    
    // lower_bound و upper_bound
    auto lower = lower_bound(numbers.begin(), numbers.end(), 35);
    auto upper = upper_bound(numbers.begin(), numbers.end(), 45);
    
    cout << "أول عنصر >= 35: " << *lower << endl;
    cout << "أول عنصر > 45: " << *upper << endl;
    
    return 0;
}

خوارزميات التعديل (Modifying Algorithms)

خوارزميات لتعديل محتويات المجموعات.

reverse() - عكس الترتيب 
#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    vector<int> numbers = {1, 2, 3, 4, 5, 6};
    
    cout << "قبل العكس: ";
    for(int n : numbers) cout << n << " ";
    cout << endl;

    cout << "بعد العكس: ";
    for(int n : numbers) cout << n << " ";
    cout << endl;
    
    return 0;
}
replace() - استبدال القيم 
#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    vector<int> numbers = {10, 20, 30, 20, 40, 20, 50};
    
    cout << "قبل الاستبدال: ";
    for(int n : numbers) cout << n << " ";
    cout << endl;
    
    // استبدال كل 20 بـ 99
    replace(numbers.begin(), numbers.end(), 20, 99);
    
    cout << "بعد الاستبدال: ";
    for(int n : numbers) cout << n << " ";
    cout << endl;
    
    return 0;
}
rotate() - تدوير العناصر 
#include <iostream>

#include <vector>
using namespace std;

int main() {

الخوارزميات العددية (Numeric Algorithms)

خوارزميات للعمليات الحسابية على المجموعات.

accumulate() - حساب المجموع 
#include <iostream>
#include <vector>
#include <numeric>
using namespace std;

int main() {
    vector<int> numbers = {10, 20, 30, 40, 50};
    
    // حساب مجموع العناصر
    int sum = accumulate(numbers.begin(), numbers.end(), 0);
    cout << "مجموع الأرقام: " << sum << endl;
    
    // حساب مجموع العناصر مع قيمة بداية 100
    sum = accumulate(numbers.begin(), numbers.end(), 100);
    cout << "المجموع + 100: " << sum << endl;
    
    // حساب حاصل ضرب العناصر
    int product = accumulate(numbers.begin(), numbers.end(), 1, 
                             multiplies<int>());
    cout << "حاصل الضرب: " << product << endl;
    
    return 0;
}
inner_product() - الضرب القياسي 
#include <iostream>
#include <vector>
#include <numeric>
using namespace std;

int main() {
    vector<int> v1 = {1, 2, 3, 4};
    vector<int> v2 = {5, 6, 7, 8};
    
    // حساب الضرب القياسي: 1*5 + 2*6 + 3*7 + 4*8
    int result = inner_product(v1.begin(), v1.end(), v2.begin(), 0);
    
    cout << "الضرب القياسي: " << result << endl;
    
    return 0;
}
iota() - تعبئة بأرقام متتالية 
#include <iostream>

int main() {

خوارزميات المجموعات (Set Algorithms)

خوارزميات للتعامل مع المجموعات المرتبة.

عمليات المجموعات 
#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    vector<int> set1 = {1, 2, 3, 4, 5};
    vector<int> set2 = {3, 4, 5, 6, 7};
    vector<int> result(10);
    
    // اتحاد المجموعتين
    auto it = set_union(set1.begin(), set1.end(),
                        set2.begin(), set2.end(),
                        result.begin());
    result.resize(it - result.begin());
    
    cout << "الاتحاد: ";
    for(int n : result) cout << n << " ";
    cout << endl;
    
    // تقاطع المجموعتين
    it = set_intersection(set1.begin(), set1.end(),
                          set2.begin(), set2.end(),
                          result.begin());
    result.resize(it - result.begin());
    
    cout << "التقاطع: ";
    for(int n : result) cout << n << " ";
    cout << endl;
    
    // الفرق بين المجموعتين
    it = set_difference(set1.begin(), set1.end(),
                        set2.begin(), set2.end(),
                        result.begin());
    result.resize(it - result.begin());
    
    cout << "الفرق (set1 - set2): ";
    for(int n : result) cout << n << " ";
    cout << endl;
    
    return 0;
}
includes() - تضمين المجموعة 
#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    vector<int> superset = {1, 2, 3, 4, 5, 6, 7};
    vector<int> subset = {3, 5, 7};

                          subset.begin(), subset.end());

         << (result ? "نعم" : "لا") << endl;
    
    return 0;
}

تمارين وتطبيقات عملية

التمرين 1: تحليل بيانات الطلاب

استخدم الخوارزميات لتحليل درجات الطلاب:

الحل 
#include <iostream>
#include <algorithm>
#include <vector>
#include <numeric>
using namespace std;

int main() {
    vector<int> grades = {85, 90, 78, 92, 88, 76, 95, 89};
    
    // الترتيب
    sort(grades.begin(), grades.end());
    
    cout << "الدرجات مرتبة: ";
    for(int g : grades) cout << g << " ";
    cout << endl;
    
    // أعلى درجة
    cout << "أعلى درجة: " << grades.back() << endl;
    
    // أدنى درجة
    cout << "أدنى درجة: " << grades.front() << endl;
    
    // متوسط الدرجات
    double avg = accumulate(grades.begin(), grades.end(), 0.0) / grades.size();
    cout << "المتوسط: " << avg << endl;
    
    // عدد الطلاب فوق 90
    int count90 = count_if(grades.begin(), grades.end(),
                           [](int g) { return g >= 90; });
    cout << "عدد الطلاب فوق 90: " << count90 << endl;
    
    return 0;
}

التمرين 2: معالجة النصوص

استخدم الخوارزميات لمعالجة قائمة كلمات:

الحل 
#include <iostream>
#include <algorithm>
#include <vector>
#include <string>
using namespace std;

int main() {
    vector<string> words = {"apple", "banana", "cherry", "date", 
                           "elderberry", "fig", "grape"};
    
    // ترتيب أبجدي
    sort(words.begin(), words.end());
    
    cout << "الكلمات مرتبة: ";
    for(const string& w : words) cout << w << " ";
    cout << endl;
    
    // البحث عن كلمة
    string searchWord = "cherry";
    auto it = find(words.begin(), words.end(), searchWord);
    
    if(it != words.end()) {
        cout << "تم العثور على " << searchWord << endl;
    }
    
    // عكس ترتيب الكلمات
    reverse(words.begin(), words.end());
    
    cout << "الكلمات بعد العكس: ";
    for(const string& w : words) cout << w << " ";
    cout << endl;
    
    return 0;
}
