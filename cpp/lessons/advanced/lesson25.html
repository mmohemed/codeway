ملفات الإدخال/الإخراج (File I/O)

قراءة وكتابة الملفات – تخزين دائم للبيانات

ملفات الإدخال/الإخراج (File I/O) تسمح للبرنامج بالتفاعل مع الملفات على القرص الصلب. يمكن قراءة البيانات من ملفات (إدخال) أو كتابتها إلى ملفات (إخراج) للتخزين الدائم.

ifstream

Input File Stream:用于 قراءة البيانات من الملفات (إدخال).

ofstream

Output File Stream:用于 كتابة البيانات إلى الملفات (إخراج).

fstream

File Stream:用于 القراءة والكتابة معاً (إدخال/إخراج).

1. تضمين مكتبة الملفات

للتعامل مع الملفات في ++C، يجب تضمين المكتبة <fstream>.

#include <iostream>
#include <fstream>   // مكتبة ملفات الإدخال/الإخراج
#include <string>
using namespace std;

2. فتح وإغلاق الملفات

قبل التعامل مع ملف، يجب فتحه. بعد الانتهاء، يجب إغلاقه لتحرير الموارد.

مثال: كتابة إلى ملف (ofstream)

#include <iostream>
#include <fstream>
using namespace std;

int main() {
    // إنشاء كائن ofstream وفتح ملف للكتابة
    ofstream outFile("example.txt");

    // التحقق من نجاح الفتح
    if (!outFile) {
        cerr << "خطأ في فتح الملف للكتابة!" << endl;
        return 1;
    }

    // كتابة بيانات إلى الملف
    outFile << "مرحباً بالعالم!" << endl;
    outFile << "هذا سطر ثاني." << endl;
    outFile << "رقم: " << 42 << endl;

    // إغلاق الملف
    outFile.close();
    cout << "تمت الكتابة بنجاح." << endl;
    return 0;
}

مثال: قراءة من ملف (ifstream)

#include <iostream>
#include <fstream>
#include <string>
using namespace std;

int main() {
    // فتح ملف للقراءة
    ifstream inFile("example.txt");

    if (!inFile) {
        cerr << "خطأ في فتح الملف للقراءة!" << endl;
        return 1;
    }

    string line;
    // قراءة سطر سطر حتى نهاية الملف
    while (getline(inFile, line)) {
        cout << "اقرأ: " << line << endl;
    }

    inFile.close();
    return 0;
}

3. أوضاع فتح الملفات (File Modes)

يمكن تحديد كيفية فتح الملف باستخدام flags من ios.

الوضعالوصف
ios::inفتح للقراءة (افتراضي لـ ifstream).
ios::outفتح للكتابة (افتراضي لـ ofstream). يحذف المحتوى السابق.
ios::appفتح للإضافة (append) - الكتابة في نهاية الملف.
ios::ateفتح والانتقال إلى نهاية الملف.
ios::truncحذف محتوى الملف إذا كان موجوداً.
ios::binaryفتح الملف كملف ثنائي (بدلاً من نصي).
ios::in ios::out ios::app ios::ate ios::trunc ios::binary

مثال: استخدام أوضاع مختلفة

#include <fstream>
using namespace std;

int main() {
    // الكتابة مع مسح المحتوى السابق (الوضع الافتراضي)
    ofstream fout1("test.txt", ios::out);
    fout1 << "سطر جديد" << endl;
    fout1.close();

    // الإضافة إلى نهاية الملف (بدون مسح)
    ofstream fout2("test.txt", ios::app);
    fout2 << "سطر مضاف في النهاية" << endl;
    fout2.close();

    // فتح للقراءة والكتابة معاً
    fstream file("data.txt", ios::in | ios::out | ios::trunc);
    if (file.is_open()) {
        file << "بيانات جديدة" << endl;
        file.seekg(0);  // العودة للبداية للقراءة
        string content;
        getline(file, content);
        cout << content << endl;
        file.close();
    }
    return 0;
}

4. قراءة وكتابة أنواع مختلفة

قراءة كلمة بكلمة

ifstream inFile("words.txt");
string word;
while (inFile >> word) {  // تقرأ كلمة (حتى أول مسافة)
    cout << "كلمة: " << word << endl;
}

قراءة سطر بسطر

ifstream inFile("lines.txt");
string line;
while (getline(inFile, line)) {
    cout << "سطر: " << line << endl;
}

قراءة بيانات منظمة (مثل CSV)

#include <sstream>  // لـ stringstream

struct Student {
    string name;
    int age;
    double gpa;
};

int main() {
    ifstream inFile("students.txt");
    vector<Student> students;
    string line;

    while (getline(inFile, line)) {
        stringstream ss(line);
        Student s;
        string token;

        getline(ss, s.name, ',');  // قراءة حتى الفاصلة
        getline(ss, token, ',');
        s.age = stoi(token);
        getline(ss, token, ',');
        s.gpa = stod(token);

        students.push_back(s);
    }

    for (const auto& s : students) {
        cout << s.name << ", " << s.age << ", " << s.gpa << endl;
    }
    return 0;
}

5. الملفات الثنائية (Binary Files)

الملفات الثنائية تخزن البيانات بنفس تمثيلها في الذاكرة، مما يوفر مساحة وسرعة أكبر. تستخدم الدوال read() و write().

#include <fstream>
using namespace std;

struct Person {
    char name[50];
    int age;
    double salary;
};

int main() {
    // كتابة هيكل إلى ملف ثنائي
    Person p1 = {"Ahmed", 30, 5000.0};
    ofstream outFile("person.dat", ios::binary);
    outFile.write(reinterpret_cast<char*>(&p1), sizeof(Person));
    outFile.close();

    // قراءة هيكل من ملف ثنائي
    Person p2;
    ifstream inFile("person.dat", ios::binary);
    inFile.read(reinterpret_cast<char*>(&p2), sizeof(Person));
    inFile.close();

    cout << "الاسم: " << p2.name << ", العمر: " << p2.age << ", الراتب: " << p2.salary << endl;
    return 0;
}
تحذير: استخدام reinterpret_cast يتطلب حذراً. تأكد من أن الهيكل لا يحتوي على مؤشرات أو أنواع معقدة.

6. التحقق من حالة الملف

توفر كائنات الملف دوالاً للتحقق من حالتها:

الدالةالوصف
is_open()ترجع true إذا كان الملف مفتوحاً بنجاح.
good()ترجع true إذا كانت العملية السابقة ناجحة ولا يوجد خطأ.
fail()ترجع true إذا فشلت العملية السابقة (خطأ قابل للاسترجاع).
bad()ترجع true إذا حدث خطأ جسيم (غير قابل للاسترجاع).
eof()ترجع true إذا تم الوصول إلى نهاية الملف.
clear()تعيد ضبط حالة الخطأ.
ifstream file("test.txt");
if (!file.is_open()) {
    cerr << "فشل فتح الملف" << endl;
    return 1;
}

int x;
file >> x;
if (file.fail()) {
    cout << "فشل في القراءة (خطأ في التنسيق)" << endl;
    file.clear();  // إعادة ضبط حالة الخطأ
}

if (file.eof()) {
    cout << "وصلنا إلى نهاية الملف" << endl;
}

7. التحرك داخل الملف (Seeking)

يمكنك التحرك إلى مواقع محددة في الملف باستخدام seekg() (للقراءة) و seekp() (للكتابة).

#include <fstream>
using namespace std;

int main() {
    fstream file("data.txt", ios::in | ios::out | ios::trunc);
    file << "0123456789";

    // التحرك إلى الموقع 5 للقراءة
    file.seekg(5, ios::beg);  // من بداية الملف
    char ch;
    file.get(ch);
    cout << "الحرف في الموقع 5: " << ch << endl;  // '5'

    // التحرك من الموضع الحالي
    file.seekg(-2, ios::cur);  // رجوع خطوتين
    file.get(ch);
    cout << "الحرف بعد الرجوع: " << ch << endl;  // '4'

    // التحرك من نهاية الملف
    file.seekg(-3, ios::end);
    file.get(ch);
    cout << "الحرف الثالث من النهاية: " << ch << endl;  // '7'

    // معرفة الموقع الحالي
    streampos pos = file.tellg();
    cout << "الموقع الحالي: " << pos << endl;

    file.close();
    return 0;
}

تطبيقات عملية

نسخ ملف

ifstream src("source.txt");
ofstream dst("dest.txt");
dst << src.rdbuf();  // نسخ كامل

إحصائيات ملف

ifstream file("text.txt");
int lines = 0, words = 0, chars = 0;
string line;
while (getline(file, line)) {
    lines++;
    chars += line.length();
    // عد الكلمات...
}

تمرين: اكتب برنامجاً يقرأ ملفاً نصياً (input.txt)، ويقوم بتحويل جميع الأحرف إلى uppercase، ويكتب النتيجة في ملف آخر (output.txt).

عرض الحل 
#include <iostream>
#include <fstream>
#include <cctype>  // for toupper
using namespace std;

int main() {
    ifstream inFile("input.txt");
    ofstream outFile("output.txt");

    if (!inFile) {
        cerr << "خطأ في فتح ملف الإدخال" << endl;
        return 1;
    }
    if (!outFile) {
        cerr << "خطأ في فتح ملف الإخراج" << endl;
        return 1;
    }

    char ch;
    while (inFile.get(ch)) {
        outFile.put(toupper(ch));
    }

    cout << "تمت المعالجة بنجاح." << endl;
    inFile.close();
    outFile.close();
    return 0;
}

أفضل الممارسات في التعامل مع الملفات

  • تحقق دائماً من نجاح فتح الملف قبل استخدامه.
  • أغلق الملف بعد الانتهاء (أو استخدم النطاق التلقائي).
  • استخدم أوضاع الفتح المناسبة (ios::app للإضافة).
  • تعامل مع الأخطاء بشكل مناسب (تحقق من fail(), bad(), eof()).
  • للبيانات المنظمة، فكر في استخدام مكتبات مثل JSON أو XML.
  • تذكر أن الملفات النصية قابلة للقراءة البشرية، بينما الثنائية أكثر كفاءة.
