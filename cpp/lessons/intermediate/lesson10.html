ما هي المؤشرات؟

المؤشرات (Pointers) هي متغيرات تخزن عناوين الذاكرة لمتغيرات أخرى. تعتبر المؤشرات من أقوى ميزات لغة C++ وتسمح بالتحكم المباشر في الذاكرة.

عنوان الذاكرة

كل متغير في الذاكرة له عنوان فريد يمكن الوصول إليه باستخدام المؤشرات

الوصول غير المباشر

تسمح المؤشرات بالوصول إلى المتغيرات وتعديلها بشكل غير مباشر

كفاءة عالية

تساعد في تحسين أداء البرامج وإدارة الذاكرة بكفاءة

تمثيل المؤشرات في الذاكرة

42 قيمة x
0x7ffd5 عنوان x
0x7ffd5 مؤشر ptr

أساسيات المؤشرات

لتعريف مؤشر، نستخدم النجمة * قبل اسم المتغير. المؤشر يخزن عنوان متغير من نفس النوع.

مثال 1: تعريف واستخدام المؤشرات 
#include <iostream>
using namespace std;

int main() {
    int x = 10;           // متغير عادي
    int *ptr;             // تعريف مؤشر لعدد صحيح
    ptr = &x;              // تخزين عنوان x في المؤشر
    
    cout << "قيمة x: " << x << endl;
    cout << "عنوان x: " << &x << endl;
    cout << "قيمة ptr (العنوان المخزن): " << ptr << endl;
    cout << "القيمة التي يشير إليها ptr: " << *ptr << endl;  // Dereferencing
    
    // تغيير قيمة x باستخدام المؤشر
    *ptr = 20;
    cout << "\nبعد التعديل باستخدام المؤشر:" << endl;
    cout << "قيمة x الجديدة: " << x << endl;
    cout << "القيمة من خلال المؤشر: " << *ptr << endl;
    
    return 0;
}

شرح الرموز:

&: عامل أخذ العنوان (Address-of operator)

*: عامل الوصول للقيمة (Dereferencing operator) - يستخدم مرتين:

1. في تعريف المؤشر: int *ptr;

2. للوصول للقيمة: *ptr = 20;

المخرجات:

قيمة x: 10
عنوان x: 0x7ffd5e3a4 (عنوان افتراضي)
قيمة ptr (العنوان المخزن): 0x7ffd5e3a4
القيمة التي يشير إليها ptr: 10

بعد التعديل باستخدام المؤشر:
قيمة x الجديدة: 20
القيمة من خلال المؤشر: 20

عمليات المؤشرات (Pointer Arithmetic)

يمكن إجراء عمليات حسابية على المؤشرات مثل الجمع والطرح، ولكنها تعمل بشكل مختلف عن الأعداد العادية.

العملية المعنى مثال
ptr + n ينتقل n عنصر للأمام (حسب حجم نوع البيانات) int *ptr; ptr + 1 يزيد بمقدار 4 بايت
ptr - n ينتقل n عنصر للخلف char *ptr; ptr - 2 يقل بمقدار 2 بايت
ptr1 - ptr2 يحسب عدد العناصر بين مؤشرين عدد العناصر = (ptr1 - ptr2)
مثال 2: عمليات المؤشرات 
#include <iostream>
using namespace std;

int main() {
    int arr[] = {10, 20, 30, 40, 50};
    int *ptr = arr;  // المؤشر يشير إلى أول عنصر في المصفوفة
    
    cout << "المصفوفة الأصلية: ";
    for(int i = 0; i < 5; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;
    
    cout << "الوصول للمصفوفة باستخدام المؤشر:" << endl;
    cout << "ptr[0] = " << ptr[0] << endl;
    cout << "*(ptr + 1) = " << *(ptr + 1) << endl;
    cout << "*(ptr + 2) = " << *(ptr + 2) << endl;
    
    // التقدم بالمؤشر
    ptr += 2;
    cout << "\nبعد ptr += 2، القيمة: " << *ptr << endl;
    
    // الفرق بين مؤشرين
    int *ptr2 = &arr[4];
    cout << "عدد العناصر بين ptr و ptr2: " << ptr2 - ptr << endl;
    
    return 0;
}

المؤشرات والمصفوفات

اسم المصفوفة هو في الواقع مؤشر ثابت يشير إلى أول عنصر في المصفوفة.

مثال 3: العلاقة بين المؤشرات والمصفوفات 
#include <iostream>
using namespace std;

int main() {
    int numbers[] = {100, 200, 300, 400, 500};
    int *ptr = numbers;  // ptr يشير إلى numbers[0]
    
    // طرق مختلفة للوصول لعناصر المصفوفة
    cout << "الطريقة الأولى: numbers[2] = " << numbers[2] << endl;
    cout << "الطريقة الثانية: *(numbers + 2) = " << *(numbers + 2) << endl;
    cout << "الطريقة الثالثة: ptr[2] = " << ptr[2] << endl;
    cout << "الطريقة الرابعة: *(ptr + 2) = " << *(ptr + 2) << endl;
    
    cout << "\nعرض جميع العناصر باستخدام المؤشر:" << endl;
    for(int i = 0; i < 5; i++) {
        cout << "*(ptr + " << i << ") = " << *(ptr + i) << endl;
    }
    
    return 0;
}

الذاكرة الديناميكية (Dynamic Memory)

تستخدم المؤشرات لتخصيص وإدارة الذاكرة الديناميكية باستخدام new و delete.

مثال 4: تخصيص ذاكرة ديناميكية 
#include <iostream>
using namespace std;

int main() {
    // تخصيص ذاكرة لمتغير واحد
    int *p = new int;
    *p = 42;
    cout << "قيمة المتغير الديناميكي: " << *p << endl;
    delete p;  // تحرير الذاكرة
    
    // تخصيص ذاكرة لمصفوفة
    int size = 5;
    int *arr = new int[size];
    
    // إدخال قيم للمصفوفة
    for(int i = 0; i < size; i++) {
        arr[i] = (i + 1) * 10;
    }
    
    // عرض القيم
    cout << "قيم المصفوفة الديناميكية: ";
    for(int i = 0; i < size; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;
    
    delete[] arr;  // تحرير الذاكرة (لاحظ استخدام [] مع delete)
    
    return 0;
}

ملاحظات مهمة:

1. استخدم new لتخصيص الذاكرة الديناميكية

2. استخدم delete لتحرير الذاكرة المخصصة لمتغير واحد

3. استخدم delete[] لتحرير الذاكرة المخصصة لمصفوفة

4. عدم تحرير الذاكرة يؤدي إلى تسرب الذاكرة (Memory Leak)

المؤشرات والدوال

يمكن تمرير المؤشرات إلى الدوال لتعديل المتغيرات الأصلية أو للتعامل مع المصفوفات بكفاءة.

مثال 5: تمرير مؤشرات للدوال 
#include <iostream>
using namespace std;

// دالة تستقبل مؤشر وتعدل القيمة
void swap(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

// دالة تستقبل مصفوفة كمؤشر
void doubleArray(int *arr, int size) {
    for(int i = 0; i < size; i++) {
        arr[i] *= 2;
    }
}

// دالة تُرجع مؤشر
int* createArray(int size) {
    int *arr = new int[size];
    for(int i = 0; i < size; i++) {
        arr[i] = i * 10;
    }
    return arr;
}

int main() {
    int x = 5, y = 10;
    
    cout << "قبل التبديل: x = " << x << ", y = " << y << endl;
    swap(&x, &y);
    cout << "بعد التبديل: x = " << x << ", y = " << y << endl;
    
    int nums[] = {1, 2, 3, 4, 5};
    doubleArray(nums, 5);
    cout << "المصفوفة بعد المضاعفة: ";
    for(int i = 0; i < 5; i++) {
        cout << nums[i] << " ";
    }
    cout << endl;
    
    // استخدام دالة تُرجع مؤشر
    int *arr = createArray(5);
    cout << "المصفوفة المنشأة ديناميكياً: ";
    for(int i = 0; i < 5; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;
    delete[] arr;  // تذكر دائماً تحرير الذاكرة
    
    return 0;
}

مفاهيم متقدمة في المؤشرات

المؤشر لمؤشر (Pointer to Pointer)

مثال 6: مؤشر لمؤشر 
#include <iostream>
using namespace std;

int main() {
    int x = 100;
    int *ptr = &x;     // مؤشر لعدد صحيح
    int **ptr2 = &ptr;  // مؤشر لمؤشر لعدد صحيح
    
    cout << "قيمة x: " << x << endl;
    cout << "قيمة x من خلال ptr: " << *ptr << endl;
    cout << "قيمة x من خلال ptr2: " << **ptr2 << endl;
    
    **ptr2 = 200;  // تغيير القيمة من خلال ptr2
    
    cout << "قيمة x بعد التعديل: " << x << endl;
    
    return 0;
}

المؤشرات الثابتة والمؤشرات للثوابت

النوع التركيب الوصف
مؤشر لثابت const int *ptr لا يمكن تغيير القيمة التي يشير إليها
مؤشر ثابت int *const ptr لا يمكن تغيير العنوان الذي يشير إليه
مؤشر ثابت لثابت const int *const ptr لا يمكن تغيير العنوان ولا القيمة
مثال 7: const مع المؤشرات 
#include <iostream>
using namespace std;

int main() {
    int x = 10, y = 20;
    
    // مؤشر لثابت - لا يمكن تغيير القيمة
    const int *ptr1 = &x;
    // *ptr1 = 30;  // خطأ! لا يمكن تعديل القيمة
    ptr1 = &y;      // يمكن تغيير العنوان
    
    // مؤشر ثابت - لا يمكن تغيير العنوان
    int *const ptr2 = &x;
    *ptr2 = 30;     // يمكن تعديل القيمة
    // ptr2 = &y;   // خطأ! لا يمكن تغيير العنوان
    
    // مؤشر ثابت لثابت
    const int *const ptr3 = &x;
    // *ptr3 = 40;  // خطأ! لا يمكن تعديل القيمة
    // ptr3 = &y;   // خطأ! لا يمكن تغيير العنوان
    
    return 0;
}

تمارين وتطبيقات عملية

التمرين 1: عكس مصفوفة باستخدام المؤشرات

قم بكتابة دالة تعكس مصفوفة باستخدام المؤشرات فقط.

الحل 
#include <iostream>
using namespace std;

void reverseArray(int *arr, int size) {
    int *start = arr;
    int *end = arr + size - 1;
    
    while (start < end) {
        // تبديل القيم باستخدام المؤشرات
        int temp = *start;
        *start = *end;
        *end = temp;
        
        start++;
        end--;
    }
}

int main() {
    int numbers[] = {1, 2, 3, 4, 5, 6, 7};
    int size = sizeof(numbers) / sizeof(numbers[0]);
    
    cout << "المصفوفة الأصلية: ";
    for(int i = 0; i < size; i++) {
        cout << numbers[i] << " ";
    }
    cout << endl;
    
    reverseArray(numbers, size);
    
    cout << "المصفوفة بعد العكس: ";
    for(int i = 0; i < size; i++) {
        cout << numbers[i] << " ";
    }
    cout << endl;
    
    return 0;
}

التمرين 2: البحث عن أكبر عنصر

قم بكتابة دالة تجد أكبر عنصر في مصفوفة وتعيد مؤشراً إليه.

الحل 
#include <iostream>
using namespace std;

int* findMax(int *arr, int size) {
    if (size == 0) return nullptr;
    
    int *maxPtr = arr;  // نفترض أن أول عنصر هو الأكبر
    
    for(int i = 1; i < size; i++) {
        if (*(arr + i) > *maxPtr) {
            maxPtr = arr + i;
        }
    }
    
    return maxPtr;
}

int main() {
    int numbers[] = {45, 23, 89, 12, 67, 34, 91, 56};
    int size = sizeof(numbers) / sizeof(numbers[0]);
    
    int *maxPtr = findMax(numbers, size);
    
    cout << "عناصر المصفوفة: ";
    for(int i = 0; i < size; i++) {
        cout << numbers[i] << " ";
    }
    cout << endl;
    
    cout << "أكبر عنصر هو: " << *maxPtr << endl;
    cout << "موقعه في الذاكرة: " << maxPtr << endl;
    cout << "رقم الفهرس: " << (maxPtr - numbers) << endl;
    
    return 0;
}
