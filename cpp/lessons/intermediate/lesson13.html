الهياكل (Structures) في ++C

تجميع البيانات – بناء أنواع مخصصة

تجميع البيانات

الهيكل (struct) يسمح بتجميع عناصر مختلفة (متغيرات) تحت اسم واحد، لتكوين نوع بيانات مركب.

نوع مخصص

يمكن تعريف أنواع جديدة تناسب المشكلة، مثل: طالب، موظف، نقطة في المستوى، إلخ.

قابلية التوسع

الهياكل أساس بناء البرامج الكبيرة؛ يمكن إضافة دوال (طرق) إليها (في ++C).

1. تعريف الهيكل (struct)

يتم تعريف الهيكل باستخدام الكلمة المفتاحية struct متبوعة باسم الهيكل وأعضاءه (members) بين قوسين متعرجين.

// تعريف هيكل "نقطة" في المستوى الديكارتي
struct Point {
    int x;   // الإحداثي x
    int y;   // الإحداثي y
};  // انتبه للفاصلة المنقوطة بعد القوس }

// تعريف هيكل "طالب"
struct Student {
    string name;
    int age;
    double gpa;
};

// يمكن وضع الهيكل داخل أو خارج الدالة main
ملاحظة: لا تنسَ كتابة الفاصلة المنقوطة ; بعد القوس المغلق للهيكل، لأنها جزء من الجملة.

2. إنشاء متغيرات والوصول للأعضاء

بعد تعريف الهيكل، يمكن إنشاء متغيرات منه والوصول إلى أعضائها باستخدام النقطة ..

int main() {
    // إنشاء متغير من نوع Point
    Point p1;          // قيم غير محددة (تحتوي على بيانات عشوائية)
    p1.x = 10;         // تعيين قيمة للعضو x
    p1.y = -5;         // تعيين قيمة للعضو y

    // إنشاء وتهيئة مباشرة (Aggregate initialization)
    Point p2 = {20, 30};   // p2.x = 20, p2.y = 30

    // هيكل Student
    Student s1;
    s1.name = "أحمد";
    s1.age = 20;
    s1.gpa = 3.75;

    Student s2 = {"فاطمة", 22, 3.9};

    // طباعة القيم
    cout << s1.name << " عمره: " << s1.age;
    return 0;
}

3. تهيئة الهياكل

توفر ++C عدة طرق لتهيئة الهياكل، خاصة مع المعايير الحديثة.

struct Product {
    string name;
    double price;
    int quantity;
};

int main() {
    // 1. تهيئة تقليدية (حسب ترتيب الأعضاء)
    Product p1 = {"حاسوب", 2500.0, 10};

    // 2. تهيئة باستخدام قوسين (Uniform initialization) C++11
    Product p2 {"هاتف", 1200.0, 25};

    // 3. تهيئة مع حذف بعض الأعضاء (الباقي يُصفر)
    Product p3 = {"كتاب"};  // price = 0, quantity = 0

    // 4. Designated initializers (C++20) : تسمية العضو
    // Product p4 = {.name = "قلم", .price = 2.5, .quantity = 100};

    // 5. بعد التعريف يمكن تعيين قيم للأعضاء بشكل منفصل
    Product p5;
    p5.name = "ممحاة";
    p5.price = 1.0;
    p5.quantity = 50;
}

struct مقابل class في ++C: الفرق الوحيد هو أن أعضاء struct عامة (public) بشكل افتراضي، بينما class خاصة (private). ما عدا ذلك، يمكن للـ struct أن يحتوي على دوال ومُنشئات وكل ميزات class.

4. تمرير الهياكل للدوال

يمكن تمرير الهياكل للدوال بالقيمة (يتم النسخ) أو بالمرجع (لتجنب النسخ والتعديل).

struct Rectangle {
    double width;
    double height;
};

// تمرير بالقيمة (يتم نسخ الهيكل) – مناسبة للقراءة فقط
double areaByValue(Rectangle rect) {
    return rect.width * rect.height;
}

// تمرير بالمرجع (لا نسخ، أسرع للهياكل الكبيرة)
double areaByRef(const Rectangle &rect) {
    return rect.width * rect.height; // const يضمن عدم التعديل
}

// تمرير بالمرجع للتعديل على الهيكل الأصلي
void scale(Rectangle &rect, double factor) {
    rect.width *= factor;
    rect.height *= factor;
}

int main() {
    Rectangle r = {5.0, 3.0};
    cout << "المساحة: " << areaByRef(r);
    scale(r, 2.0);  // الآن r.width = 10.0, r.height = 6.0
    return 0;
}

5. المؤشرات إلى الهياكل

يمكن استخدام المؤشرات مع الهياكل، ويتم الوصول للأعضاء باستخدام ->.

struct Book {
    string title;
    string author;
    int pages;
};

int main() {
    Book b1 = {"C++ Primer", "Lippman", 900};
    Book *ptr = &b1;

    // الوصول باستخدام ->
    cout << ptr->title;   // تعادل (*ptr).title
    ptr->pages = 950;     // تعديل قيمة pages

    // إنشاء هيكل ديناميكياً
    Book *pBook = new Book{"Clean Code", "Robert Martin", 400};
    cout << pBook->author;
    delete pBook;
    return 0;
}

6. الهياكل المتداخلة (Nested Structures)

يمكن أن يحتوي الهيكل على أعضاء من نوع هيكل آخر.

struct Address {
    string city;
    string street;
    int building;
};

struct Person {
    string name;
    int age;
    Address addr;   // هيكل متداخل
};

int main() {
    Person p = {"محمد", 30, {"القاهرة", "التحرير", 15}};
    cout << p.name << " يسكن في " << p.addr.city;
    p.addr.building = 20; // تعديل العنوان
    return 0;
}

تمرين: أنشئ هيكل "Student" يحتوي على الاسم، العمر، والدرجة. اكتب دالة تقبل مرجعاً إلى Student وتزيد عمره بمقدار 1.

عرض الحل 
#include <iostream>
#include <string>
using namespace std;

struct Student {
    string name;
    int age;
    double grade;
};

void increaseAge(Student &s) {
    s.age++;
}

int main() {
    Student s = {"سارة", 20, 88.5};
    increaseAge(s);
    cout << s.name << " عمرها الآن: " << s.age; // 21
    return 0;
}

أفضل الممارسات

  • اجعل أسماء الأعضاء واضحة ومعبرة.
  • استخدم التهيئة الموحدة {} لتجنب الأخطاء.
  • مرر الهياكل الكبيرة بالمرجع (const) لتحسين الأداء.
  • استخدم struct عندما تكون جميع الأعضاء عامة (public).
  • استخدم class إذا كنت بحاجة إلى تغليف (encapsulation).
