مقدمة عن السلاسل النصية

السلاسل النصية (Strings) هي تسلسل من الأحرف تستخدم لتخزين ومعالجة النصوص. في C++، هناك طريقتان رئيسيتان للتعامل مع النصوص:

C-Strings

مصفوفات من الأحرف تنتهي بحرف null '\0'، موروثة من لغة C

string Class

كلاس متكامل من مكتبة C++ القياسية يوفر دوالاً سهلة للتعامل مع النصوص

C-Strings (أسلوب C)

في أسلوب C، يتم تمثيل النصوص كمصفوفات من الأحرف تنتهي بالحرف الفارغ '\0'.

تمثيل C-String في الذاكرة

H
e
l
l
o
\0
كلمة "Hello" في الذاكرة + حرف null
مثال 1: تعريف واستخدام C-Strings 
#include <iostream>
#include <cstring>  // لدوال النصوص في C
using namespace std;

int main() {
    // طرق مختلفة لتعريف C-Strings
    char str1[] = "Hello";           // مصفوفة أحرف
    char str2[20] = "World";         // مصفوفة بحجم محدد
    char str3[] = {'C', '+', '+', '\0'};  // تعريف يدوي
    
    cout << "str1: " << str1 << endl;
    cout << "str2: " << str2 << endl;
    cout << "str3: " << str3 << endl;
    
    // حساب طول النص
    cout << "طول str1: " << strlen(str1) << endl;
    cout << "حجم str1 في الذاكرة: " << sizeof(str1) << " بايت" << endl;
    
    return 0;
}

دوال مهمة للتعامل مع C-Strings:

الدالة الوصف مثال
strlen() حساب طول النص strlen("Hello") → 5
strcpy() نسخ نص strcpy(dest, src)
strcat() دمج نصين strcat(str1, str2)
strcmp() مقارنة نصين strcmp(str1, str2)
strchr() البحث عن حرف strchr(str, 'a')
مثال 2: دوال C-Strings 
#include <iostream>
#include <cstring>
using namespace std;

int main() {
    char name[50] = "Ahmed";
    char lastName[] = " Mohamed";
    char copy[50];
    
    // نسخ نص
    strcpy(copy, name);
    cout << "نسخة من الاسم: " << copy << endl;
    
    // دمج النصوص
    strcat(name, lastName);
    cout << "الاسم الكامل: " << name << endl;
    
    // مقارنة النصوص
    char str1[] = "apple";
    char str2[] = "banana";
    int result = strcmp(str1, str2);
    
    if(result < 0)
        cout << str1 << " أصغر من " << str2 << endl;
    else if(result > 0)
        cout << str1 << " أكبر من " << str2 << endl;
    else
        cout << "النصان متساويان" << endl;
    
    return 0;
}

string Class (أسلوب C++)

توفر مكتبة C++ القياسية كلاس string الذي يسهل التعامل مع النصوص بشكل كبير.

مثال 3: تعريف واستخدام string 
#include <iostream>
#include <string>  // مكتبة string
using namespace std;

int main() {
    // طرق مختلفة لتعريف string
    string s1;                    // نص فارغ
    string s2 = "Hello";          // تهيئة بنص
    string s3("World");           // تهيئة باستخدام constructor
    string s4(s2);                // نسخ من s2
    string s5(5, '*');            // تكرار حرف 5 مرات "*****"
    
    cout << "s1: '" << s1 << "'" << endl;
    cout << "s2: " << s2 << endl;
    cout << "s3: " << s3 << endl;
    cout << "s4: " << s4 << endl;
    cout << "s5: " << s5 << endl;
    
    // إدخال نص من المستخدم
    string name;
    cout << "\nأدخل اسمك: ";
    getline(cin, name);  // لقراءة سطر كامل
    
    cout << "مرحباً " << name << "!" << endl;
    
    return 0;
}

ملاحظة مهمة: استخدم getline(cin, str) لقراءة سطر كامل يتضمن مسافات، بينما cin >> str يقرأ حتى أول مسافة فقط.

العمليات الأساسية على النصوص

الإسناد والدمج

مثال 4: إسناد ودمج النصوص 
#include <iostream>
#include <string>
using namespace std;

int main() {
    string firstName = "Ahmed";
    string lastName = "Ali";
    
    // الإسناد
    string fullName = firstName;  // نسخ
    cout << "fullName = " << fullName << endl;
    
    // الدمج باستخدام +
    fullName = firstName + " " + lastName;
    cout << "الاسم الكامل: " << fullName << endl;
    
    // الدمج باستخدام +=
    fullName += " Mohamed";
    cout << "بعد الإضافة: " << fullName << endl;
    
    // إضافة حرف
    fullName.push_back('!');
    cout << "بعد إضافة !: " << fullName << endl;
    
    return 0;
}

الوصول للأحرف

مثال 5: الوصول للأحرف 
#include <iostream>
#include <string>
using namespace std;

int main() {
    string text = "Programming";
    
    // الوصول للأحرف باستخدام []
    cout << "الحرف الأول: " << text[0] << endl;
    cout << "الحرف الثالث: " << text[2] << endl;
    cout << "آخر حرف: " << text[text.length() - 1] << endl;
    
    // استخدام at() (أكثر أماناً)
    cout << "الحرف الرابع: " << text.at(3) << endl;
    
    // تعديل حرف
    text[0] = 'p';
    cout << "بعد التعديل: " << text << endl;
    
    // عرض جميع الأحرف
    cout << "جميع الأحرف: ";
    for(char c : text) {
        cout << c << " ";
    }
    cout << endl;
    
    return 0;
}

حجم النص وفحصه

مثال 6: دوال الحجم 
#include <iostream>
#include <string>
using namespace std;

int main() {
    string str = "Hello, World!";
    
    cout << "النص: " << str << endl;
    cout << "الطول: " << str.length() << endl;
    cout << "الحجم: " << str.size() << endl;  // نفس length()
    cout << "السعة: " << str.capacity() << endl;
    
    // فحص إذا كان النص فارغاً
    string empty;
    if(empty.empty()) {
        cout << "النص فارغ" << endl;
    }
    
    // تغيير الحجم
    str.resize(5);
    cout << "بعد تصغير الحجم: " << str << endl;
    
    return 0;
}

دوال البحث والتعديل المتقدمة

الدالة الوصف مثال
find() البحث عن نص فرعي str.find("world")
substr() استخراج جزء من النص str.substr(0, 5)
insert() إدراج نص في موضع محدد str.insert(5, "xxx")
erase() حذف جزء من النص str.erase(0, 3)
replace() استبدال جزء من النص str.replace(0, 3, "new")
مثال 7: البحث والتعديل 
#include <iostream>
#include <string>
using namespace std;

int main() {
    string text = "The quick brown fox jumps over the lazy dog";
    
    // البحث عن نص فرعي
    string word = "fox";
    size_t found = text.find(word);
    
    if(found != string::npos) {
        cout << "تم العثور على '" << word << "' في الموضع " << found << endl;
    }
    
    // استخراج جزء من النص
    string sub = text.substr(4, 5);  // 5 أحرف من الموضع 4
    cout << "نص فرعي: " << sub << endl;
    
    // إدراج نص
    text.insert(10, "very ");
    cout << "بعد الإدراج: " << text << endl;
    
    // حذف جزء
    text.erase(0, 4);  // احذف أول 4 أحرف
    cout << "بعد الحذف: " << text << endl;
    
    // استبدال جزء
    text.replace(0, 5, "A");  // استبدل 5 أحرف بـ "A"
    cout << "بعد الاستبدال: " << text << endl;
    
    return 0;
}

التحويل بين الأنواع

مثال 8: تحويل النصوص والأرقام 
#include <iostream>
#include <string>
#include <sstream>  // stringstream
using namespace std;

int main() {
    // تحويل رقم إلى نص (C++11)
    int num = 123;
    string strNum = to_string(num);
    cout << "رقم كنص: " << strNum << endl;
    
    // تحويل نص إلى رقم
    string numStr = "456";
    int value = stoi(numStr);  // string to int
    double dValue = stod("78.9");  // string to double
    
    cout << "قيمة رقمية: " << value << endl;
    cout << "قيمة عشرية: " << dValue << endl;
    
    // استخدام stringstream للتحويلات المعقدة

    ss << 123 << " " << 45.67;
    
    int x;
    double y;
    ss >> x >> y;
    
    cout << "x = " << x << ", y = " << y << endl;
    
    // تحويل C-string إلى string
    const char* cstr = "C string";
    string cppstr = cstr;
    
    // تحويل string إلى C-string
    const char* backToC = cppstr.c_str();
    
    return 0;
}

تطبيقات عملية متقدمة

مثال 9: تحليل نص (String Parsing) 
#include <iostream>
#include <string>
#include <sstream>
#include <vector>
using namespace std;

int main() {
    string data = "Ahmed,25,Cairo;Sara,22,Alex;Mohamed,30,Giza";
    vector<string> people;
    
    // تقسيم النص باستخدام ;
    stringstream ss(data);
    string person;
    
    while(getline(ss, person, ';')) {
        people.push_back(person);
    }
    
    cout << "بيانات الأشخاص:\n";
    for(const string& p : people) {
        stringstream ps(p);
        string name, age, city;
        
        getline(ps, name, ',');
        getline(ps, age, ',');
        getline(ps, city, ',');
        
        cout << "الاسم: " << name << ", العمر: " << age 
             << ", المدينة: " << city << endl;
    }
    
    return 0;
}
مثال 10: عكس النص 
#include <iostream>
#include <string>
#include <algorithm>
using namespace std;

string reverseString(string str) {
    string reversed = str;
    reverse(reversed.begin(), reversed.end());
    return reversed;
}

bool isPalindrome(string str) {
    string temp = str;
    // تحويل إلى أحرف صغيرة وتجاهل المسافات
    temp.erase(remove_if(temp.begin(), temp.end(), ::isspace), temp.end());
    transform(temp.begin(), temp.end(), temp.begin(), ::tolower);
    
    string rev = temp;
    reverse(rev.begin(), rev.end());
    return temp == rev;
}

int main() {
    string text = "Hello, World!";
    cout << "النص الأصلي: " << text << endl;
    cout << "النص المعكوس: " << reverseString(text) << endl;

تمارين وتطبيقات عملية

التمرين 1: عداد الكلمات

قم بكتابة دالة تحسب عدد الكلمات في نص معين.

الحل 
#include <iostream>
#include <string>
#include <sstream>
using namespace std;

int countWords(const string& text) {
    stringstream ss(text);
    string word;
    int count = 0;
    
    while(ss >> word) {
        count++;
    }
    
    return count;
}

int main() {
    string text = "C++ is a powerful programming language";
    
    cout << "النص: " << text << endl;
    cout << "عدد الكلمات: " << countWords(text) << endl;
    
    return 0;
}

التمرين 2: استبدال الكلمات

قم بكتابة دالة تستبدل كل كلمة معينة بكلمة أخرى في النص.

الحل 
#include <iostream>
#include <string>
using namespace std;

string replaceWord(string text, const string& oldWord, const string& newWord) {
    size_t pos = 0;
    while((pos = text.find(oldWord, pos)) != string::npos) {
        text.replace(pos, oldWord.length(), newWord);
        pos += newWord.length();
    }
    return text;
}

int main() {
    string text = "I like C++, C++ is great";
    cout << "النص الأصلي: " << text << endl;
    
    string replaced = replaceWord(text, "C++", "Python");
    cout << "بعد الاستبدال: " << replaced << endl;
    
    return 0;
}
