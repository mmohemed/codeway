ما هي الدوال؟

الدوال (Functions) هي كتل برمجية مستقلة تقوم بمهمة محددة. تساعد الدوال في تنظيم الكود، إعادة استخدامه، وتقسيم البرنامج إلى أجزاء صغيرة يسهل إدارتها وصيانتها.

تركيب الدالة (Function Syntax)

تتكون الدالة في C++ من عدة أجزاء أساسية:

نوع_البيانات اسم_الدالة(المعاملات) {
    // جسم الدالة
    return قيمة;
}
مثال 1: دالة بسيطة 
#include <iostream>
using namespace std;

// تعريف دالة تطبع رسالة ترحيب
void printWelcome() {
    cout << "مرحباً بك في برنامج تعلم الدوال!" << endl;
    cout << "هذه دالة بسيطة لا تأخذ معاملات ولا ترجع قيمة" << endl;
}

int main() {
    // استدعاء الدالة
    printWelcome();
    
    return 0;
}

شرح الأجزاء:

void: نوع البيانات (لا ترجع قيمة)

printWelcome: اسم الدالة

(): لا تأخذ معاملات

{}: جسم الدالة يحتوي على الأوامر

المعاملات (Parameters)

المعاملات هي قيم نمررها للدالة لتستخدمها في عملياتها.

أنواع تمرير المعاملات:

النوع الوصف مثال
تمرير بالقيمة (Pass by Value) يتم نسخ القيمة، والتغييرات لا تؤثر على الأصل void func(int x)
تمرير بمرجع (Pass by Reference) يتم تمرير العنوان، والتغييرات تؤثر على الأصل void func(int &x)
تمرير بمرجع ثابت (Const Reference) للقراءة فقط، لا يمكن التعديل void func(const int &x)
مثال 2: دوال بمعاملات مختلفة 
#include <iostream>
using namespace std;

// دالة تجمع رقمين (تمرير بالقيمة)
int add(int a, int b) {
    return a + b;
}

// دالة تضاعف القيمة (تمرير بمرجع)
void doubleValue(int &x) {
    x *= 2;
}

// دالة تطبع قيمة (تمرير بمرجع ثابت)
void printValue(const int &x) {
    cout << "القيمة: " << x << endl;
    // x += 5; // هذا سيعطي خطأ لأن x ثابت
}

int main() {
    int num1 = 10, num2 = 20;
    
    // استدعاء دالة الجمع
    int sum = add(num1, num2);
    cout << "مجموع " << num1 << " و " << num2 << " = " << sum << endl;
    
    // استدعاء دالة المضاعفة
    cout << "قبل المضاعفة: " << num1 << endl;
    doubleValue(num1);
    cout << "بعد المضاعفة: " << num1 << endl;
    
    // استدعاء دالة الطباعة
    printValue(num2);
    
    return 0;
}

المخرجات:

مجموع 10 و 20 = 30
قبل المضاعفة: 10
بعد المضاعفة: 20
القيمة: 20

المعاملات الافتراضية (Default Parameters)

مثال 3: معاملات افتراضية 
#include <iostream>
using namespace std;

// دالة بمعاملات افتراضية
void displayInfo(string name, int age = 0, string country = "غير محدد") {
    cout << "الاسم: " << name << endl;
    cout << "العمر: " << age << endl;
    cout << "البلد: " << country << endl;
    cout << "-------------------" << endl;
}

int main() {
    displayInfo("أحمد", 25, "مصر");
    displayInfo("سارة", 22);        // ستستخدم القيمة الافتراضية للبلد
    displayInfo("محمد");             // ستستخدم القيم الافتراضية للعمر والبلد
    
    return 0;
}

قيمة الإرجاع (Return Value)

يمكن للدوال إرجاع قيم باستخدام الكلمة المفتاحية return.

مثال 4: دوال تُرجع قيماً مختلفة 
#include <iostream>
using namespace std;

// دالة تُرجع مجموع رقمين
int getSum(int a, int b) {
    return a + b;
}

// دالة تُرجع أكبر رقمين
int getMax(int a, int b) {
    if (a > b)
        return a;
    else
        return b;
}

// دالة تُرجع قيمة مطلقة
double absolute(double x) {
    if (x < 0)
        return -x;
    return x;
}

// دالة تُرجع true إذا كان الرقم زوجياً
bool isEven(int num) {
    return num % 2 == 0;
}

int main() {
    int x = 15, y = 20;
    
    cout << "مجموع " << x << " و " << y << " = " << getSum(x, y) << endl;
    cout << "أكبر رقم: " << getMax(x, y) << endl;
    cout << "القيمة المطلقة لـ -7.5 = " << absolute(-7.5) << endl;
    cout << "هل الرقم 8 زوجي؟ " << boolalpha << isEven(8) << endl;
    
    return 0;
}

تحميل الدوال (Function Overloading)

يمكن تعريف عدة دوال بنفس الاسم ولكن بمعاملات مختلفة (عدداً أو نوعاً).

مثال 5: تحميل الدوال 
#include <iostream>
using namespace std;

// دالة لجمع رقمين صحيحين
int add(int a, int b) {
    cout << "جمع أعداد صحيحة: ";
    return a + b;
}

// دالة لجمع ثلاثة أرقام صحيحة
int add(int a, int b, int c) {
    cout << "جمع ثلاثة أعداد: ";
    return a + b + c;
}

// دالة لجمع رقمين عشريين
double add(double a, double b) {
    cout << "جمع أعداد عشرية: ";
    return a + b;
}

// دالة لدمج سلسلتين نصيتين
string add(string a, string b) {
    cout << "دمج نصوص: ";
    return a + " " + b;
}

int main() {
    cout << add(5, 3) << endl;
    cout << add(5, 3, 2) << endl;
    cout << add(5.5, 3.7) << endl;
    cout << add("Hello", "World") << endl;
    
    return 0;
}

المخرجات:

جمع أعداد صحيحة: 8
جمع ثلاثة أعداد: 10
جمع أعداد عشرية: 9.2
دمج نصوص: Hello World

النطاق (Scope)

النطاق يحدد أين يمكن الوصول إلى المتغيرات في البرنامج.

النوع الوصف النطاق
متغير محلي (Local) معرف داخل دالة داخل الدالة فقط
متغير عام (Global) معرف خارج جميع الدوال جميع أنحاء البرنامج
متغير ثابت (Static) يحتفظ بقيمته بين استدعاءات الدالة داخل الدالة ولكن قيمته محفوظة
مثال 6: النطاق في الدوال 
#include <iostream>
using namespace std;

// متغير عام
int globalVar = 100;

void demonstrateScope() {
    // متغير محلي
    int localVar = 50;
    
    // متغير ثابت - يحتفظ بقيمته
    static int staticVar = 0;
    staticVar++;
    
    cout << "المتغير العام: " << globalVar << endl;
    cout << "المتغير المحلي: " << localVar << endl;
    cout << "المتغير الثابت: " << staticVar << endl;
    cout << "-------------------" << endl;
}

int main() {
    demonstrateScope();  // staticVar = 1
    demonstrateScope();  // staticVar = 2
    demonstrateScope();  // staticVar = 3
    
    // cout << localVar; // خطأ - لا يمكن الوصول لمتغير محلي خارج الدالة
    
    return 0;
}

تمارين وتطبيقات عملية

التمرين 1: آلة حاسبة متقدمة

قم بإنشاء دوال لعملية آلة حاسبة متكاملة:

الحل 
#include <iostream>
#include <iomanip>
using namespace std;

// دوال العمليات الحسابية
double add(double a, double b) { return a + b; }
double subtract(double a, double b) { return a - b; }
double multiply(double a, double b) { return a * b; }
double divide(double a, double b) { 
    if (b != 0) return a / b;
    else return 0;
}
int modulus(int a, int b) {
    if (b != 0) return a % b;
    else return 0;
}

// دالة عرض القائمة
void showMenu() {
    cout << "\n=== الآلة الحاسبة ===\n";
    cout << "1. جمع\n";
    cout << "2. طرح\n";
    cout << "3. ضرب\n";
    cout << "4. قسمة\n";
    cout << "5. باقي القسمة\n";
    cout << "0. خروج\n";
    cout << "اختيارك: ";
}

int main() {
    int choice;
    double num1, num2;
    
    do {
        showMenu();
        cin >> choice;
        
        if (choice == 0) break;
        
        cout << "أدخل الرقم الأول: ";
        cin >> num1;
        cout << "أدخل الرقم الثاني: ";
        cin >> num2;
        
        cout << fixed << setprecision(2);
        cout << "\nالنتيجة: ";
        
        switch(choice) {
            case 1:
                cout << add(num1, num2) << endl;
                break;
            case 2:
                cout << subtract(num1, num2) << endl;
                break;
            case 3:
                cout << multiply(num1, num2) << endl;
                break;
            case 4:
                if (num2 != 0)
                    cout << divide(num1, num2) << endl;
                else
                    cout << "خطأ: القسمة على صفر" << endl;
                break;
            case 5:
                if ((int)num2 != 0)
                    cout << modulus((int)num1, (int)num2) << endl;
                else
                    cout << "خطأ: القسمة على صفر" << endl;
                break;
            default:
                cout << "اختيار غير صحيح" << endl;
        }
        cout << "-------------------" << endl;
        
    } while (choice != 0);
    
    cout << "شكراً لاستخدام الآلة الحاسبة!" << endl;
    
    return 0;
}

التمرين 2: دوال رياضية

قم بإنشاء دوال لحساب العوامل الرياضية المختلفة:

الحل 
#include <iostream>
#include <cmath>
using namespace std;

// دالة حساب المضروب (Factorial)
long long factorial(int n) {
    if (n <= 1) return 1;
    return n * factorial(n - 1);
}

// دالة التحقق من أولية الرقم
bool isPrime(int n) {
    if (n <= 1) return false;
    for (int i = 2; i <= sqrt(n); i++) {
        if (n % i == 0) return false;
    }
    return true;
}

// دالة حساب الأس
double power(double base, int exponent) {
    double result = 1;
    for (int i = 0; i < abs(exponent); i++) {
        result *= base;
    }
    if (exponent < 0) return 1 / result;
    return result;
}

// دالة مجموع الأرقام
int sumOfDigits(int n) {
    int sum = 0;
    n = abs(n);
    while (n > 0) {
        sum += n % 10;
        n /= 10;
    }
    return sum;
}

int main() {
    int num;
    
    cout << "أدخل رقماً: ";
    cin >> num;
    
    cout << "المضروب: " << factorial(num) << endl;
    cout << "هل الرقم أولي؟ " << boolalpha << isPrime(num) << endl;
    cout << "2^" << num << " = " << power(2, num) << endl;
    cout << "مجموع أرقام " << num << " = " << sumOfDigits(num) << endl;
    
    return 0;
}
