الصديقات (Friend Functions)

Friend Functions & Friend Classes – كسر التغليف بحذر

في البرمجة كائنية التوجه، التغليف (Encapsulation) يمنع الوصول المباشر للبيانات الخاصة. ولكن أحياناً نحتاج لدالة خارجية أو صنف آخر للوصول إلى الأعضاء الخاصة. هنا يأتي دور الصديقات (Friends).

دالة صديقة

دالة غير عضو في الصنف، ولكنها مصرح لها بالوصول إلى جميع أعضائه (حتى الخاصة والمحمية).

صنف صديق

صنف كامل يصبح صديقاً، فتصبح جميع دوال الأعضاء فيه قادرة على الوصول للبيانات الخاصة للصنف الآخر.

استثناء للتغليف

الصديقات تكسر التغليف بشكل مقصود ومنضبط. يجب استخدامها بحذر وفقط عند الضرورة.

أولاً: الدوال الصديقة (Friend Functions)

الدالة الصديقة هي دالة عادية (ليست عضواً في الصنف) يتم التصريح عنها داخل الصنف باستخدام الكلمة المفتاحية friend. يمكن أن تكون:

  • دالة عادية (غير تابعة لأي صنف).
  • دالة عضو في صنف آخر.

مثال: دالة صديقة عادية

#include <iostream>
using namespace std;

class Rectangle {
private:
    double width, height;

public:
    Rectangle(double w, double h) : width(w), height(h) {}

    // التصريح بدالة صديقة
    friend double calculateArea(const Rectangle& rect);
};

// تعريف الدالة الصديقة (يمكنها الوصول للأعضاء الخاصة)
double calculateArea(const Rectangle& rect) {
    return rect.width * rect.height;  // وصول مباشر للخاصية
}

int main() {
    Rectangle r(5, 3);
    cout << "مساحة المستطيل: " << calculateArea(r) << endl; // 15
    return 0;
}
ملاحظة: الدالة الصديقة ليست عضواً في الصنف، لذلك تُستدعى بدون استخدام عامل النقطة (.)، ويجب تمرير كائن من الصنف كمعامل إذا أردت الوصول لأعضائه.

مثال: دالة عضو من صنف آخر كصديقة

class B;  // تعريف مسبق

class A {
private:
    int secretA;
public:
    A(int val) : secretA(val) {}

    // التصريح بأن دالة displayB من الصنف B هي صديقة
    friend void B::displayB(const A& a); // يحتاج تعريف كامل لـ B
};

class B {
private:
    int secretB;
public:
    B(int val) : secretB(val) {}

    void displayB(const A& a) {
        cout << "secretA = " << a.secretA << endl; // وصول للخاصية الخاصة
        cout << "secretB = " << secretB << endl;
    }
};

int main() {
    A objA(42);
    B objB(99);
    objB.displayB(objA); // تعمل بنجاح
    return 0;
}

ثانياً: الأصناف الصديقة (Friend Classes)

عندما يكون صنف كامل صديقاً لصنف آخر، فإن جميع دوال الأعضاء في الصنف الصديق تصبح قادرة على الوصول إلى الأعضاء الخاصة والمحمية للصنف الآخر.

class Employee {
private:
    string name;
    double salary;

public:
    Employee(string n, double s) : name(n), salary(s) {}

    // جعل صنف Manager صديقاً كاملاً
    friend class Manager;
};

class Manager {
public:
    void showEmployeeData(const Employee& emp) {
        cout << "اسم الموظف: " << emp.name << endl;
        cout << "راتب الموظف: " << emp.salary << endl; // وصول للخاص
    }

    void giveRaise(Employee& emp, double amount) {
        emp.salary += amount; // تعديل الخاصية الخاصة
    }
};

int main() {
    Employee emp("أحمد", 5000);
    Manager mgr;
    mgr.showEmployeeData(emp);
    mgr.giveRaise(emp, 1000);
    mgr.showEmployeeData(emp);
    return 0;
}

تذكير: الصداقة ليست متبادلة. إذا جعلت A صديقاً لـ B، فليس بالضرورة أن B صديق لـ A. الصداقة لا تُورث.

خصائص ومميزات الصديقات

الخاصيةالوصف
ليست عضواًالدالة الصديقة ليست عضواً في الصنف، ولا تملك مؤشر this.
لا تتأثر بمعدلات الوصوليمكن وضع التصريح friend في أي قسم (public, private, protected) والنتيجة واحدة.
غير متبادلةإذا كانت A صديقة لـ B، فليس بالضرورة أن B صديقة لـ A.
غير متوارثةالصداقة لا تورث. إذا كان الصنف الأساسي صديقاً لصنف آخر، فالأصناف المشتقة ليست أصدقاء تلقائياً.
لا يمكن إلغاؤهابمجرد تعريف صديق، يبقى صديقاً. لا توجد آلية لإلغاء الصداقة.

حالات استخدام شائعة للصديقات

1. تحميل العوامل الثنائية (مثل << و >>

عند تحميل عامل الإخراج << أو الإدخال >>، غالباً ما تكون هناك حاجة لجعلها دوالاً صديقة لأن المعامل الأيسر هو كائن من نوع ostream أو istream وليس كائن من الصنف.

class Point {
private:
    int x, y;
public:
    Point(int xv = 0, int yv = 0) : x(xv), y(yv) {}

    // جعل الدالة operator<< صديقة
    friend ostream& operator<<(ostream& os, const Point& p);
};

ostream& operator<<(ostream& os, const Point& p) {
    os << "(" << p.x << ", " << p.y << ")";
    return os;
}

int main() {
    Point p1(3, 4);
    cout << "نقطة: " << p1 << endl; // نقطة: (3, 4)
    return 0;
}

2. دوال مساعدة تحتاج لبيانات داخلية

مثل دوال المقارنة المعقدة، أو دوال الرسم، أو دوال التسلسل (serialization).

class Matrix {
private:
    int data[3][3];
public:
    // دالة صديقة لطباعة المصفوفة بشكل جميل
    friend void prettyPrint(const Matrix& m);
};

void prettyPrint(const Matrix& m) {
    for (int i = 0; i < 3; i++) {
        for (int j = 0; j < 3; j++) {
            cout << m.data[i][j] << "\t";
        }
        cout << endl;
    }
}

متى نستخدم دالة صديقة بدلاً من دالة عضو؟

الموقفالأفضلية
الدالة تحتاج للوصول للبيانات الخاصة ولكن المعامل الأول ليس كائن من الصنفدالة صديقة (مثل operator<<)
الدالة تحتاج للعمل مع كائنين من نوعين مختلفيندالة صديقة غالباً
الدالة تعمل بشكل أساسي على كائن واحد من الصنفدالة عضو
الدالة تحتاج لتعدد الأشكال (polymorphism)دالة عضو افتراضية

تطبيقات عملية

مسافة بين نقطتين

class Point {
private:
    double x, y;
public:
    Point(double xv, double yv) : x(xv), y(yv) {}
    friend double distance(const Point& a, const Point& b);
};

double distance(const Point& a, const Point& b) {
    double dx = a.x - b.x;
    double dy = a.y - b.y;
    return sqrt(dx*dx + dy*dy);
}

مقارنة معقدة

class Student {
private:
    string name;
    double gpa;
public:
    Student(string n, double g) : name(n), gpa(g) {}
    friend bool isExcellent(const Student& s);
};

bool isExcellent(const Student& s) {
    return s.gpa >= 3.5;
}

تمرين: صمم صنف Complex (عدد مركب) يحتوي على real و imag. اجعل الدالة operator+ (لجمع عددين مركبين) صديقة للصنف، واكتب برنامجاً يختبر عملها.

عرض الحل 
#include <iostream>
using namespace std;

class Complex {
private:
    double real, imag;
public:
    Complex(double r = 0, double i = 0) : real(r), imag(i) {}

    // إعلان الأصدقاء
    friend Complex operator+(const Complex& c1, const Complex& c2);
    friend ostream& operator<<(ostream& os, const Complex& c);
};

// تعريف operator+ كدالة صديقة
Complex operator+(const Complex& c1, const Complex& c2) {
    return Complex(c1.real + c2.real, c1.imag + c2.imag);
}

ostream& operator<<(ostream& os, const Complex& c) {
    os << c.real << (c.imag >= 0 ? " + " : " - ") << abs(c.imag) << "i";
    return os;
}

int main() {
    Complex c1(3, 4), c2(1, 2);
    Complex c3 = c1 + c2;
    cout << c1 << " + " << c2 << " = " << c3 << endl;
    return 0;
}

أفضل الممارسات مع الصديقات

  • استخدم الصديقات فقط عند الضرورة القصوى.
  • لا تجعل صنفاً كاملاً صديقاً إلا إذا كان عدد الدوال التي تحتاج للوصول كبيراً.
  • تذكر أن الصداقة غير متبادلة وغير متوارثة.
  • استخدم الصديقات بشكل مكثف مع تحميل العوامل (خاصة << و >>).
  • حافظ على التغليف قدر الإمكان، والصديقات هي استثناء.
  • وثّق استخدام الصديقات في الكود لتوضيح سبب كسر التغليف.
