مقدمة عن الوراثة (Inheritance)

الوراثة هي أحد المبادئ الأساسية في البرمجة الكائنية (OOP) والتي تسمح لفئة (Class) بوراثة خصائص ودوال من فئة أخرى. هذا يساعد في إعادة استخدام الكود وتنظيم العلاقات بين الفئات.

الفئة الأساسية (Base Class)

الفئة الأم التي يتم الوراثة منها. تحتوي على الخصائص والدوال المشتركة.

الفئة المشتقة (Derived Class)

الفئة التي ترث من الفئة الأساسية. يمكنها إضافة خصائص ودوال جديدة.

إعادة الاستخدام

تسمح الوراثة بإعادة استخدام الكود المكتوب وتوسيعه دون تعديله.

الفئة الأساسية (Animal)

+ name: string
+ age: int
+ eat()
+ sleep()

كلب (Dog)

+ breed: string
+ bark()
+ (يرث من Animal)

قطة (Cat)

+ color: string
+ meow()
+ (يرث من Animal)

الوراثة الأساسية (Basic Inheritance)

لإنشاء فئة ترث من فئة أخرى، نستخدم النقطتين : متبوعين بمعدل الوصول واسم الفئة الأساسية.

مثال 1: وراثة بسيطة 
#include <iostream>
#include <string>
using namespace std;

// الفئة الأساسية (Parent Class)
class Animal {
protected:
    string name;
    int age;
    
public:
    void setName(string n) {
        name = n;
    }
    
    void setAge(int a) {
        age = a;
    }
    
    void eat() {
        cout << name << " is eating" << endl;
    }
    
    void sleep() {
        cout << name << " is sleeping" << endl;
    }
};

// الفئة المشتقة (Child Class) - ترث من Animal
class Dog : public Animal {
private:
    string breed;
    
public:
    void setBreed(string b) {
        breed = b;
    }
    
    void bark() {
        cout << name << " is barking: Woof! Woof!" << endl;
    }
    
    void displayInfo() {
        cout << "Name: " << name << endl;
        cout << "Age: " << age << " years" << endl;
        cout << "Breed: " << breed << endl;
    }
};

int main() {
    // إنشاء كائن من الفئة المشتقة
    Dog myDog;
    
    // استخدام دوال من الفئة الأساسية
    myDog.setName("Max");
    myDog.setAge(3);
    
    // استخدام دوال من الفئة المشتقة
    myDog.setBreed("Golden Retriever");
    
    // عرض المعلومات
    myDog.displayInfo();
    myDog.eat();
    myDog.sleep();
    myDog.bark();
    
    return 0;
}

المخرجات:

Name: Max
Age: 3 years
Breed: Golden Retriever
Max is eating
Max is sleeping
Max is barking: Woof! Woof!

ملاحظات مهمة:

protected: تسمح للفئات المشتقة بالوصول إلى الأعضاء، ولكن ليس من خارج الفئة

public Animal: تعني أننا نرث من Animal بشكل عام (public inheritance)

الفئة المشتقة يمكنها استخدام كل الأعضاء public و protected من الفئة الأساسية

أنواع الوراثة (Types of Inheritance)

يمكن التحكم في كيفية وراثة أعضاء الفئة الأساسية باستخدام معدّلات الوصول المختلفة.

نوع الوراثة الوصف public في الأساسي protected في الأساسي private في الأساسي
public inheritance تحافظ على مستويات الوصول الأصلية public protected لا يمكن الوصول
protected inheritance تحول public إلى protected protected protected لا يمكن الوصول
private inheritance تحول public و protected إلى private private private لا يمكن الوصول
مثال 2: أنواع الوراثة 
#include <iostream>
using namespace std;

class Base {
public:
    int publicVar;
protected:
    int protectedVar;
private:
    int privateVar;
};

// public inheritance
class PublicDerived : public Base {
public:
    void access() {
        publicVar = 10;     // public → public
        protectedVar = 20;  // protected → protected
        // privateVar = 30; // خطأ! لا يمكن الوصول
    }
};

// protected inheritance
class ProtectedDerived : protected Base {
public:
    void access() {
        publicVar = 10;     // public → protected
        protectedVar = 20;  // protected → protected
        // privateVar = 30; // خطأ! لا يمكن الوصول
    }
};

// private inheritance
class PrivateDerived : private Base {
public:
    void access() {
        publicVar = 10;     // public → private
        protectedVar = 20;  // protected → private
        // privateVar = 30; // خطأ! لا يمكن الوصول
    }
};

int main() {
    PublicDerived pub;
    pub.publicVar = 50;  // صحيح! public في الفئة المشتقة
    
    ProtectedDerived pro;
    // pro.publicVar = 50; // خطأ! أصبح protected
    
    PrivateDerived pri;
    // pri.publicVar = 50; // خطأ! أصبح private
    
    return 0;
}

المُنشئات في الوراثة (Constructors in Inheritance)

عند إنشاء كائن من فئة مشتقة، يتم استدعاء مُنشئ الفئة الأساسية أولاً، ثم مُنشئ الفئة المشتقة.

مثال 3: المُنشئات في الوراثة 
#include <iostream>
#include <string>
using namespace std;

class Person {
protected:
    string name;
    int age;
    
public:
    // مُنشئ الفئة الأساسية
    Person(string n, int a) {
        name = n;
        age = a;
        cout << "Person constructor called for " << name << endl;
    }
    
    void display() {
        cout << "Name: " << name << ", Age: " << age << endl;
    }
};

class Student : public Person {
private:
    int studentId;
    double gpa;
    
public:
    // مُنشئ الفئة المشتقة - يستدعي مُنشئ الفئة الأساسية
    Student(string n, int a, int id, double g) : Person(n, a) {
        studentId = id;
        gpa = g;
        cout << "Student constructor called for " << name << endl;
    }
    
    void displayStudent() {
        Person::display();  // استدعاء دالة الفئة الأساسية
        cout << "Student ID: " << studentId << ", GPA: " << gpa << endl;
    }
};

int main() {
    Student s1("Ahmed", 20, 2021001, 3.8);
    cout << "\nStudent Information:" << endl;
    s1.displayStudent();
    
    return 0;
}

المخرجات:

Person constructor called for Ahmed
Student constructor called for Ahmed

Student Information:
Name: Ahmed, Age: 20
Student ID: 2021001, GPA: 3.8

الوراثة المتعددة (Multiple Inheritance)

يمكن للفئة أن ترث من أكثر من فئة أساسية في نفس الوقت.

Printable

Serializable

Document

مثال 4: الوراثة المتعددة 
#include <iostream>
#include <string>
using namespace std;

// الفئة الأساسية الأولى
class Printable {
public:
    void print() {
        cout << "Printing document..." << endl;
    }
};

// الفئة الأساسية الثانية
class Serializable {
public:
    void serialize() {
        cout << "Serializing document..." << endl;
    }
};

// الفئة المشتقة - ترث من كلا الفئتين
class Document : public Printable, public Serializable {
private:
    string title;
    string content;
    
public:
    Document(string t, string c) : title(t), content(c) {}
    
    void display() {
        cout << "Title: " << title << endl;
        cout << "Content: " << content << endl;
    }
};

int main() {
    Document doc("Report", "This is the annual report...");
    
    doc.display();
    doc.print();      // من Printable
    doc.serialize();  // من Serializable
    
    return 0;
}

تمارين وتطبيقات عملية

التمرين 1: نظام إدارة الموظفين

قم بإنشاء نظام إدارة موظفين باستخدام الوراثة:

الحل 
#include <iostream>
#include <string>
#include <vector>
using namespace std;

// الفئة الأساسية
class Employee {
protected:
    string name;
    int id;
    double baseSalary;
    
public:
    Employee(string n, int i, double salary) {
        name = n;
        id = i;
        baseSalary = salary;
    }
    
    virtual double calculateSalary() {
        return baseSalary;
    }
    
    virtual void display() {
        cout << "ID: " << id << ", Name: " << name;
        cout << ", Salary: $" << calculateSalary() << endl;
    }
    
    virtual ~Employee() {}
};

// موظف عادي
class RegularEmployee : public Employee {
public:
    RegularEmployee(string n, int i, double salary) 
        : Employee(n, i, salary) {}
};

// مدير - له مكافأة إضافية
class Manager : public Employee {
private:
    double bonus;
    
public:
    Manager(string n, int i, double salary, double b) 
        : Employee(n, i, salary), bonus(b) {}
    
    double calculateSalary() override {
        return baseSalary + bonus;
    }
    
    void display() override {
        Employee::display();
        cout << "  (Manager Bonus: $" << bonus << ")" << endl;
    }
};

// مبرمج - له overtime
class Programmer : public Employee {
private:
    int overtimeHours;
    double overtimeRate;
    
public:
    Programmer(string n, int i, double salary, int hours, double rate) 
        : Employee(n, i, salary), overtimeHours(hours), overtimeRate(rate) {}
    
    double calculateSalary() override {
        return baseSalary + (overtimeHours * overtimeRate);
    }
    
    void display() override {
        Employee::display();
        cout << "  (Overtime: " << overtimeHours << " hours)" << endl;
    }
};

int main() {
    vector<Employee*> employees;
    
    employees.push_back(new RegularEmployee("Ahmed", 1001, 5000));
    employees.push_back(new Manager("Sara", 1002, 7000, 1500));
    employees.push_back(new Programmer("Mohamed", 1003, 6000, 20, 50));
    
    cout << "Employee Information:" << endl;
    for(auto emp : employees) {
        emp->display();
    }
    
    // تنظيف الذاكرة
    for(auto emp : employees) {
        delete emp;
    }
    
    return 0;
}

التمرين 2: أشكال هندسية

قم بإنشاء نظام للأشكال الهندسية باستخدام الوراثة:

الحل 
#include <iostream>
#include <cmath>
#include <vector>
using namespace std;

// الفئة الأساسية المجردة
class Shape {
protected:
    string color;
    
public:
    Shape(string c) : color(c) {}
    
    virtual double getArea() = 0;  // دالة افتراضية خالصة
    virtual double getPerimeter() = 0;
    
    virtual void display() {
        cout << "Shape color: " << color << endl;
        cout << "Area: " << getArea() << endl;
        cout << "Perimeter: " << getPerimeter() << endl;
    }
    
    virtual ~Shape() {}
};

class Circle : public Shape {
private:
    double radius;
    
public:
    Circle(string c, double r) : Shape(c), radius(r) {}
    
    double getArea() override {
        return M_PI * radius * radius;
    }
    
    double getPerimeter() override {
        return 2 * M_PI * radius;
    }
    
    void display() override {
        cout << "\nCircle:" << endl;
        Shape::display();
        cout << "Radius: " << radius << endl;
    }
};

class Rectangle : public Shape {
private:
    double width;
    double height;
    
public:
    Rectangle(string c, double w, double h) : Shape(c), width(w), height(h) {}
    
    double getArea() override {
        return width * height;
    }
    
    double getPerimeter() override {
        return 2 * (width + height);
    }
    
    void display() override {
        cout << "\nRectangle:" << endl;
        Shape::display();
        cout << "Width: " << width << ", Height: " << height << endl;
    }
};

int main() {
    vector<Shape*> shapes;
    
    shapes.push_back(new Circle("Red", 5));
    shapes.push_back(new Rectangle("Blue", 4, 6));
    
    for(auto shape : shapes) {
        shape->display();
    }
    
    for(auto shape : shapes) {
        delete shape;
    }
    
    return 0;
}
