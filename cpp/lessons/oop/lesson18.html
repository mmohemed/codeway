تعدد الأشكال (Polymorphism)

Polymorphism in C++ – واجهة واحدة .. سلوكيات متعددة

تعدد الأشكال (Polymorphism) هو أحد الركائز الأساسية للبرمجة كائنية التوجه. يعني القدرة على استخدام واجهة واحدة لتمثيل أنواع مختلفة من الكائنات، بحيث يستجيب كل كائن بطريقته الخاصة عند استدعاء نفس الدالة.

نوعا تعدد الأشكال في ++C

تعدد الأشكال الثابت

يحدث أثناء وقت الترجمة (Compile-time). يشمل تحميل الدوال (Function Overloading) وتحميل العوامل (Operator Overloading).

تعدد الأشكال الديناميكي

يحدث أثناء وقت التنفيذ (Runtime). يعتمد على الوراثة والدوال الافتراضية (Virtual Functions).

أولاً: تعدد الأشكال الثابت (Compile-time Polymorphism)

1. تحميل الدوال (Function Overloading)

إمكانية تعريف عدة دوال بنفس الاسم ولكن بمعاملات مختلفة (عدداً أو نوعاً). المترجم يختار الدالة المناسبة بناءً على المعاملات المرسلة.

#include <iostream>
using namespace std;

class Calculator {
public:
    // نفس الاسم ولكن معاملات مختلفة
    int add(int a, int b) {
        return a + b;
    }

    double add(double a, double b) {
        return a + b;
    }

    int add(int a, int b, int c) {
        return a + b + c;
    }

    string add(string a, string b) {
        return a + " " + b;
    }
};

int main() {
    Calculator calc;
    cout << calc.add(5, 10) << endl;          // 15
    cout << calc.add(3.5, 2.7) << endl;      // 6.2
    cout << calc.add(1, 2, 3) << endl;       // 6
    cout << calc.add("Hello", "World") << endl; // Hello World
    return 0;
}

قواعد تحميل الدوال: يتم التمييز بناءً على عدد المعاملات، أنواعها، أو ترتيبها. لا يمكن التحميل بناءً على نوع الإرجاع فقط.

2. تحميل العوامل (Operator Overloading)

إعطاء معنى إضافي لعامل (مثل +, -, *, etc.) ليعمل مع الكائنات المعرفة من قبل المستخدم.

class Complex {
private:
    double real, imag;
public:
    Complex(double r = 0, double i = 0) : real(r), imag(i) {}

    // تحميل عامل +
    Complex operator+(const Complex& other) const {
        return Complex(real + other.real, imag + other.imag);
    }

    // تحميل عامل << (كصديق)
    friend ostream& operator<<(ostream& os, const Complex& c) {
        os << c.real << (c.imag >= 0 ? " + " : " - ") << abs(c.imag) << "i";
        return os;
    }
};

int main() {
    Complex c1(3, 4), c2(1, 2);
    Complex c3 = c1 + c2;  // يستخدم العامل المحمّل
    cout << c1 << " + " << c2 << " = " << c3 << endl;
    // الإخراج: 3 + 4i + 1 + 2i = 4 + 6i
    return 0;
}

ثانياً: تعدد الأشكال الديناميكي (Runtime Polymorphism)

يعتمد على الوراثة والدوال الافتراضية (virtual functions). يتم تحديد الدالة المناسبة أثناء تنفيذ البرنامج بناءً على نوع الكائن الفعلي.

1. الدوال الافتراضية (Virtual Functions)

دالة في الصنف الأساسي يُصرح عنها بـ virtual، ويمكن تجاوزها (override) في الأصناف المشتقة. عندما نستدعيها عبر مؤشر أو مرجع للصنف الأساسي، يتم تنفيذ نسخة الكائن الفعلي.

class Animal {
public:
    virtual void sound() {   // دالة افتراضية
        cout << "الحيوان يصدر صوتاً" << endl;
    }
};

class Dog : public Animal {
public:
    void sound() override {  // تجاوز الدالة (override)
        cout << "الكلب ينبح: هوو هوو" << endl;
    }
};

class Cat : public Animal {
public:
    void sound() override {
        cout << "القطة تموء: مياو مياو" << endl;
    }
};

int main() {
    Animal* animals[3];
    animals[0] = new Animal();
    animals[1] = new Dog();
    animals[2] = new Cat();

    for (int i = 0; i < 3; i++) {
        animals[i]->sound();  // استدعاء تعدد الأشكال
    }
    // الإخراج:
    // الحيوان يصدر صوتاً
    // الكلب ينبح: هوو هوو
    // القطة تموء: مياو مياو

    // تنظيف الذاكرة
    for (int i = 0; i < 3; i++) delete animals[i];
    return 0;
}

2. الدوال الافتراضية البحتة (Pure Virtual) والأصناف المجردة

الدالة الافتراضية البحتة هي دالة ليس لها تعريف في الصنف الأساسي (تُسند إليها = 0). تجعل الصنف "مجرداً" (Abstract) لا يمكن إنشاء كائنات منه، وتجبر الأصناف المشتقة على توفير تعريف لها.

class Shape {  // صنف مجرد
public:
    virtual double area() const = 0;  // دالة افتراضية بحتة
    virtual void draw() const = 0;
    
    virtual ~Shape() {}  // مدمر افتراضي (مهم)
};

class Circle : public Shape {
private:
    double radius;
public:
    Circle(double r) : radius(r) {}
    
    double area() const override {
        return 3.14159 * radius * radius;
    }
    
    void draw() const override {
        cout << "رسم دائرة" << endl;
    }
};

class Rectangle : public Shape {
private:
    double width, height;
public:
    Rectangle(double w, double h) : width(w), height(h) {}
    
    double area() const override {
        return width * height;
    }
    
    void draw() const override {
        cout << "رسم مستطيل" << endl;
    }
};

int main() {
    // Shape s; // خطأ! لا يمكن إنشاء كائن من صنف مجرد
    Shape* shapes[2] = {new Circle(5), new Rectangle(4, 6)};
    
    for (int i = 0; i < 2; i++) {
        shapes[i]->draw();
        cout << "المساحة: " << shapes[i]->area() << endl;
    }
    // تنظيف
    for (int i = 0; i < 2; i++) delete shapes[i];
    return 0;
}
مهم جداً: اجعل المدمر في الأصناف الأساسية virtual لضمان استدعاء المدمرات الصحيحة للأصناف المشتقة عند الحذف عبر مؤشر للصنف الأساسي.

مفاهيم متقدمة

كيف يعمل تعدد الأشكال داخلياً؟ (vtable)

عند تعريف دوال افتراضية، يقوم المترجم بإنشاء جدول افتراضي (vtable) لكل صنف يحتوي على عناوين الدوال الافتراضية. كل كائن يحمل مؤشراً (vptr) إلى هذا الجدول. عند استدعاء دالة افتراضية، يتم تتبع المؤشر إلى الجدول واستدعاء الدالة المناسبة.

ملاحظة أداء: استدعاء الدوال الافتراضية أبطأ قليلاً من الدوال العادية بسبب الـ indirection، لكن الفرق ضئيل في معظم التطبيقات.

الكلمات المفتاحية override و final (C++11)

override تؤكد أن الدالة تعيد تعريف دالة افتراضية من الصنف الأساسي (تساعد في اكتشاف الأخطاء). final تمنع إعادة تعريف الدالة أو الوراثة من الصنف.

class Base {
public:
    virtual void func() {}
};

class Derived : public Base {
public:
    void func() override {  // تأكيد أننا نعيد التعريف
        cout << "Derived func" << endl;
    }
};

class FinalClass final { };  // لا يمكن الوراثة من هذا الصنف

// class Test : public FinalClass { }; // خطأ!

class Base2 {
public:
    virtual void show() final { }  // لا يمكن تجاوز هذه الدالة
};

class Derived2 : public Base2 {
    // void show() override { } // خطأ! الدالة final
};

تطبيقات عملية

نظام المدفوعات

class Payment {
public:
    virtual void pay(double amount) = 0;
};

class CreditCard : public Payment {
    void pay(double amount) override {
        cout << "دفع " << amount << " ببطاقة ائتمان";
    }
};

class PayPal : public Payment {
    void pay(double amount) override {
        cout << "دفع " << amount << " عبر PayPal";
    }
};

مشغل وسائط

class MediaPlayer {
public:
    virtual void play() = 0;
    virtual void stop() = 0;
};

class MP3Player : public MediaPlayer {
    void play() override { /* تشغيل MP3 */ }
    void stop() override { /* إيقاف */ }
};

تمرين: صمم نظاماً للعبة يحتوي على صنف أساسي Character مع دوال افتراضية attack() و defend(). اشتق منه Warrior و Mage يعيدان تعريف هذه الدوال.

عرض الحل 
#include <iostream>
#include <vector>
using namespace std;

class Character {
public:
    virtual void attack() = 0;
    virtual void defend() = 0;
    virtual ~Character() {}
};

class Warrior : public Character {
public:
    void attack() override {
        cout << "المحارب يهاجم بالسيف!" << endl;
    }
    void defend() override {
        cout << "المحارب يدافع بالدرع" << endl;
    }
};

class Mage : public Character {
public:
    void attack() override {
        cout << "الساحر يهاجم بكرة نارية!" << endl;
    }
    void defend() override {
        cout << "الساحر يدافع بدرع سحري" << endl;
    }
};

int main() {
    vector<Character*> characters;
    characters.push_back(new Warrior());
    characters.push_back(new Mage());

    for (auto c : characters) {
        c->attack();
        c->defend();
    }
    // تنظيف
    for (auto c : characters) delete c;
    return 0;
}

أفضل الممارسات في تعدد الأشكال

  • استخدم override صراحةً لتجنب الأخطاء.
  • اجعل المدمرات virtual في الأصناف الأساسية.
  • استخدم الأصناف المجردة لتحديد واجهات واضحة.
  • تذكر أن تحميل العوامل يزيد من وضوح الكود.
  • تجنب الإفراط في استخدام الدوال الافتراضية إذا لم تكن بحاجة حقيقية لتعدد الأشكال.
  • استخدم المؤشرات الذكية (smart pointers) لإدارة الذاكرة مع الكائنات متعددة الأشكال.
