مقدمة عن البرمجة الكائنية (OOP)

البرمجة الكائنية (Object-Oriented Programming) هي نمط برمجي يعتمد على مفهوم "الكائنات" التي تحتوي على بيانات (خصائص) ودوال (سلوكيات).

الفئة (Class)

قالب أو مخطط لإنشاء الكائنات. تحدد الخصائص والسلوكيات المشتركة.

الكائن (Object)

نسخة حقيقية من الفئة. له وجود فعلي في الذاكرة ويمكن التعامل معه.

التغليف (Encapsulation)

إخفاء البيانات الداخلية وحمايتها من الوصول المباشر.

- model: string
- year: int

تعريف الفئة (Class Definition)

لتعريف فئة في C++ نستخدم الكلمة المفتاحية class متبوعة باسم الفئة.

مثال 1: تعريف فئة بسيطة 
#include <iostream>
#include <string>
using namespace std;

// تعريف فئة Person
class Person {
    // الخصائص (Attributes)
    string name;
    int age;
    string address;
    
public:
    // الدوال (Methods)
    void setName(string n) {
        name = n;
    }
    
    string getName() {
        return name;
    }
    
    void setAge(int a) {
        if(a > 0) {
            age = a;
        }
    }
    
    int getAge() {
        return age;
    }
    
    void introduce() {
        cout << "مرحباً، اسمي " << name 
             << " وعمري " << age << " سنة" << endl;
    }
};  // ملاحظة: تنتهي الفئة بفاصلة منقوطة

int main() {
    // سنقوم بإنشاء كائنات هنا لاحقاً
    return 0;
}

شرح الأجزاء:

class Person: تعريف فئة باسم Person

الخصائص: name, age, address - بيانات خاصة بالفئة

public: معدل وصول يسمح بالوصول للدوال من خارج الفئة

الدوال: وظائف للتعامل مع خصائص الفئة

إنشاء الكائنات (Creating Objects)

بعد تعريف الفئة، يمكننا إنشاء كائنات (نسخ) منها بطرق مختلفة.

مثال 2: إنشاء واستخدام الكائنات 
#include <iostream>
#include <string>
using namespace std;

class Student {
    string name;
    int id;
    double gpa;
    
public:
    void setData(string n, int i, double g) {
        name = n;
        id = i;
        gpa = g;
    }
    
    void display() {
        cout << "الاسم: " << name << endl;
        cout << "الرقم الجامعي: " << id << endl;
        cout << "المعدل: " << gpa << endl;
        cout << "------------------------" << endl;
    }
    
    bool isExcellent() {
        return gpa >= 3.5;
    }
};

int main() {
    // إنشاء كائنات من الفئة Student
    Student student1;
    Student student2;
    
    // تعيين بيانات للكائنات
    student1.setData("أحمد محمد", 2021001, 3.7);
    student2.setData("سارة علي", 2021002, 3.2);
    
    // عرض بيانات الطلاب
    cout << "بيانات الطلاب:\n";
    student1.display();
    student2.display();
    
    // التحقق من التفوق
    cout << "حالة التفوق:\n";
    cout << "أحمد: " << (student1.isExcellent() ? "متفوق" : "غير متفوق") << endl;
    cout << "سارة: " << (student2.isExcellent() ? "متفوق" : "غير متفوق") << endl;
    
    return 0;
}

المخرجات:

بيانات الطلاب:
الاسم: أحمد محمد
الرقم الجامعي: 2021001
المعدل: 3.7
------------------------
الاسم: سارة علي
الرقم الجامعي: 2021002
المعدل: 3.2
------------------------
حالة التفوق:
أحمد: متفوق
سارة: غير متفوق

معدّلات الوصول (Access Specifiers)

تستخدم معدّلات الوصول لتحديد مدى إمكانية الوصول إلى أعضاء الفئة (خصائص ودوال).

المعدّل الوصف الوصول من داخل الفئة الوصول من خارج الفئة
public عام - يمكن الوصول من أي مكان ✅ نعم ✅ نعم
private خاص - الوصول فقط داخل الفئة ✅ نعم ❌ لا
protected محمي - الوصول داخل الفئة والفئات المشتقة ✅ نعم ❌ لا
مثال 3: توضيح معدّلات الوصول 
#include <iostream>
using namespace std;

class Example {
private:
    // لا يمكن الوصول إليها إلا من داخل الفئة
    int privateVar;
    
protected:
    // يمكن الوصول إليها داخل الفئة والفئات المشتقة
    int protectedVar;
    
public:
    // يمكن الوصول إليها من أي مكان
    int publicVar;
    
    void setValues(int priv, int prot, int pub) {
        privateVar = priv;
        protectedVar = prot;
        publicVar = pub;
    }
    
    void display() {
        cout << "privateVar = " << privateVar << endl;
        cout << "protectedVar = " << protectedVar << endl;
        cout << "publicVar = " << publicVar << endl;
    }
};

int main() {
    Example ex;
    
    // ex.privateVar = 10;  // خطأ! لا يمكن الوصول
    // ex.protectedVar = 20; // خطأ! لا يمكن الوصول
    
    ex.publicVar = 30;      // صحيح! public يمكن الوصول
    ex.setValues(10, 20, 30); // صحيح! دالة عامة
    
    ex.display();
    
    return 0;
}

دوال الفئة (Class Methods)

يمكن تعريف دوال الفئة بطريقتين: داخل الفئة أو خارجها باستخدام عامل النطاق ::.

مثال 4: تعريف الدوال خارج الفئة 
#include <iostream>
#include <cmath>
using namespace std;

class Rectangle {
private:
    double width;
    double height;
    
public:
    // إعلان الدوال فقط داخل الفئة
    void setDimensions(double w, double h);
    double getArea();
    double getPerimeter();
    double getDiagonal();
    void display();
};

// تعريف الدوال خارج الفئة باستخدام ::
void Rectangle::setDimensions(double w, double h) {
    if(w > 0 && h > 0) {
        width = w;
        height = h;
    }
}

double Rectangle::getArea() {
    return width * height;
}

double Rectangle::getPerimeter() {
    return 2 * (width + height);
}

double Rectangle::getDiagonal() {
    return sqrt(width * width + height * height);
}

void Rectangle::display() {
    cout << "المستطيل:" << endl;
    cout << "العرض: " << width << endl;
    cout << "الارتفاع: " << height << endl;
    cout << "المساحة: " << getArea() << endl;
    cout << "المحيط: " << getPerimeter() << endl;
    cout << "القطر: " << getDiagonal() << endl;
}

int main() {
    Rectangle rect;
    rect.setDimensions(5, 3);
    rect.display();
    
    return 0;
}

المُنشئات (Constructors)

المُنشئ هو دالة خاصة يتم استدعاؤها تلقائياً عند إنشاء كائن جديد. له نفس اسم الفئة ولا يُرجع قيمة.

مثال 5: أنواع المُنشئات 
#include <iostream>
#include <string>
using namespace std;

class Book {
private:
    string title;
    string author;
    int pages;
    double price;
    
public:
    // 1. المُنشئ الافتراضي (Default Constructor)
    Book() {
        title = "غير محدد";
        author = "غير محدد";
        pages = 0;
        price = 0.0;
        cout << "تم إنشاء كائن باستخدام المُنشئ الافتراضي" << endl;
    }
    
    // 2. مُنشئ بمعاملات (Parameterized Constructor)
    Book(string t, string a, int p, double pr) {
        title = t;
        author = a;
        pages = p;
        price = pr;
        cout << "تم إنشاء كائن باستخدام المُنشئ المُعطى" << endl;
    }
    
    // 3. مُنشئ النسخ (Copy Constructor)
    Book(const Book &b) {
        title = b.title;
        author = b.author;
        pages = b.pages;
        price = b.price;
        cout << "تم إنشاء كائن باستخدام مُنشئ النسخ" << endl;
    }
    
    void display() {
        cout << "عنوان الكتاب: " << title << endl;
        cout << "المؤلف: " << author << endl;
        cout << "عدد الصفحات: " << pages << endl;
        cout << "السعر: $" << price << endl;
        cout << "------------------------" << endl;
    }
    
    // المُدمّر (Destructor)
    ~Book() {
        cout << "تم تدمير كائن الكتاب: " << title << endl;
    }
};

int main() {
    // استخدام المُنشئ الافتراضي
    Book book1;
    book1.display();
    
    // استخدام المُنشئ بمعاملات
    Book book2("C++ Programming", "Ahmed Ali", 500, 49.99);
    book2.display();
    
    // استخدام مُنشئ النسخ
    Book book3(book2);
    book3.display();
    
    return 0;
}

أنواع المُنشئات:

1. المُنشئ الافتراضي: لا يأخذ معاملات، يهيئ القيم الافتراضية

2. مُنشئ بمعاملات: يأخذ معاملات لتهيئة الكائن بقيم محددة

3. مُنشئ النسخ: ينشئ كائن جديد كنسخة من كائن موجود

4. المُدمّر (Destructor): ~ يتم استدعاؤه عند حذف الكائن

التمرين 1: فئة BankAccount

قم بإنشاء فئة BankAccount تمثل حساب بنكي تحتوي على:

  • رقم الحساب، اسم صاحب الحساب، الرصيد
  • دوال للإيداع والسحب والاستعلام عن الرصيد
  • منع السحب إذا كان الرصيد غير كافٍ
الحل 
#include <iostream>
#include <string>
using namespace std;

class BankAccount {
private:
    string accountNumber;
    string ownerName;
    double balance;
    
public:
    // المُنشئ
    BankAccount(string accNum, string name, double initialBalance) {
        accountNumber = accNum;
        ownerName = name;
        balance = (initialBalance >= 0) ? initialBalance : 0;
        cout << "تم إنشاء حساب جديد بنجاح" << endl;
    }
    
    // الإيداع
    void deposit(double amount) {
        if(amount > 0) {
            balance += amount;
            cout << "تم إيداع $" << amount << " بنجاح" << endl;
        } else {
            cout << "خطأ: المبلغ يجب أن يكون أكبر من صفر" << endl;
        }
    }
    
    // السحب
    void withdraw(double amount) {
        if(amount <= 0) {
            cout << "خطأ: المبلغ يجب أن يكون أكبر من صفر" << endl;
        } else if(amount > balance) {
            cout << "خطأ: رصيد غير كافٍ" << endl;
        } else {
            balance -= amount;
            cout << "تم سحب $" << amount << " بنجاح" << endl;
        }
    }
    
    // الاستعلام عن الرصيد
    double getBalance() {
        return balance;
    }
    
    // عرض معلومات الحساب
    void displayAccountInfo() {
        cout << "\nمعلومات الحساب:" << endl;
        cout << "رقم الحساب: " << accountNumber << endl;
        cout << "اسم صاحب الحساب: " << ownerName << endl;
        cout << "الرصيد الحالي: $" << balance << endl;
    }
};

int main() {
    // إنشاء حساب جديد
    BankAccount acc1("123456789", "أحمد محمد", 1000);
    
    // عرض معلومات الحساب
    acc1.displayAccountInfo();
    
    // إيداع مبلغ
    cout << "\nمحاولة إيداع $500:" << endl;
    acc1.deposit(500);
    cout << "الرصيد الحالي: $" << acc1.getBalance() << endl;
    
    // سحب مبلغ
    cout << "\nمحاولة سحب $200:" << endl;
    acc1.withdraw(200);
    cout << "الرصيد الحالي: $" << acc1.getBalance() << endl;
    
    // محاولة سحب مبلغ أكبر من الرصيد
    cout << "\nمحاولة سحب $2000:" << endl;
    acc1.withdraw(2000);
    
    return 0;
}

التمرين 2: فئة Employee

قم بإنشاء فئة Employee تحتوي على خصائص الموظف ودوال لحساب الراتب والضرائب.

الحل 
#include <iostream>
#include <string>
#include <iomanip>
using namespace std;

class Employee {
private:
    string name;
    int id;
    double hourlyRate;
    int hoursWorked;
    double taxRate;  // نسبة الضريبة (مثلاً 0.15 تعني 15%)
    
public:
    Employee(string n, int i, double rate, double tax = 0.15) {
        name = n;
        id = i;
        hourlyRate = rate;
        hoursWorked = 0;
        taxRate = tax;
    }
    
    void setHoursWorked(int hours) {
        if(hours >= 0 && hours <= 744) { // max hours in a month
            hoursWorked = hours;
        } else {
            cout << "عدد الساعات غير صحيح" << endl;
        }
    }
    
    double calculateGrossSalary() {
        return hourlyRate * hoursWorked;
    }
    
    double calculateTax() {
        return calculateGrossSalary() * taxRate;
    }
    
    double calculateNetSalary() {
        return calculateGrossSalary() - calculateTax();
    }
    
    void displaySalaryDetails() {
        cout << fixed << setprecision(2);
        cout << "\nتفاصيل راتب " << name << " (ID: " << id << ")" << endl;
        cout << "------------------------" << endl;
        cout << "ساعات العمل: " << hoursWorked << " ساعة" << endl;
        cout << "السعر للساعة: $" << hourlyRate << endl;
        cout << "الراتب الإجمالي: $" << calculateGrossSalary() << endl;
        cout << "الضريبة (15%): $" << calculateTax() << endl;
        cout << "الراتب الصافي: $" << calculateNetSalary() << endl;
    }
};

int main() {
    // إنشاء كائنات موظفين
    Employee emp1("أحمد علي", 1001, 50.0);
    Employee emp2("سارة محمد", 1002, 45.5, 0.12); // ضريبة 12%
    
    // تعيين ساعات العمل
    emp1.setHoursWorked(160);
    emp2.setHoursWorked(180);
    
    // عرض تفاصيل الرواتب
    emp1.displaySalaryDetails();
    emp2.displaySalaryDetails();
    
    return 0;
}
