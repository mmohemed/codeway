OOP في ++C

البرمجة كائنية التوجه – المبادئ الأربعة

البرمجة كائنية التوجه (Object-Oriented Programming) هي نمط برمجة يعتمد على مفهوم "الكائنات" التي تحتوي على بيانات (خصائص) ودوال (سلوكيات). تتيح OOP تنظيم البرامج المعقدة وجعلها أكثر قابلية للفهم والصيانة.

الركائز الأربع لـ OOP

التغليف

Encapsulation: جمع البيانات والدوال معًا وإخفاء التفاصيل الداخلية.

الوراثة

Inheritance: إنشاء أصناف جديدة بناءً على أصناف موجودة.

تعدد الأشكال

Polymorphism: إمكانية التعامل مع كائنات مختلفة بنفس الواجهة.

التجريد

Abstraction: إخفاء التعقيد وإظهار الأساسيات فقط.

1. التغليف (Encapsulation)

هو تجميع البيانات (الخصائص) والدوال (الطرق) التي تعالجها داخل وحدة واحدة (الصنف)، وتقييد الوصول المباشر لبعض مكونات الكائن. يتم باستخدام معدلات الوصول: private, protected, public.

class BankAccount {
private:
    double balance;  // لا يمكن الوصول إليها مباشرة خارج الصنف

public:
    // واجهة عامة للتعامل مع الرصيد
    void deposit(double amount) {
        if (amount > 0) balance += amount;
    }

    double getBalance() const {
        return balance;
    }
};

int main() {
    BankAccount acc;
    // acc.balance = 1000;  // خطأ! balance خاصة
    acc.deposit(500);       // صحيح: استخدام الواجهة العامة
    cout << acc.getBalance(); // 500
    return 0;
}

فوائد التغليف: حماية البيانات من التعديل العشوائي، إمكانية تغيير التنفيذ الداخلي دون التأثير على الكود الخارجي، زيادة نمطية البرنامج.

2. الوراثة (Inheritance)

آلية تسمح لصنف (فئة فرعية) باشتقاق خصائص وسلوكيات من صنف آخر (فئة أساسية). تعزز إعادة استخدام الكود وتمثل علاقة "is-a".

// الصنف الأساسي (Parent)
class Animal {
public:
    void eat() {
        cout << "الحيوان يأكل" << endl;
    }
};

// الصنف المشتق (Child) يرث من Animal
class Dog : public Animal {
public:
    void bark() {
        cout << "الكلب ينبح" << endl;
    }
};

int main() {
    Dog d;
    d.eat();  // موروثة من Animal
    d.bark(); // خاصة بـ Dog
    return 0;
}
نوع الوراثةوصف
publicالأعضاء العامة في الأساس تبقى عامة في المشتق، والمحمية تبقى محمية.
protectedالأعضاء العامة والمحمية في الأساس تصبح محمية في المشتق.
privateجميع الأعضاء في الأساس تصبح خاصة في المشتق (لا يمكن الوصول إليها).

3. تعدد الأشكال (Polymorphism)

القدرة على استخدام واجهة واحدة لتمثيل أنواع مختلفة من الكائنات. يتحقق نوعان: تعدد الأشكال أثناء الترجمة (تحميل الدوال) وأثناء التنفيذ (الدوال الافتراضية).

أ. تعدد الأشكال أثناء التنفيذ (الدوال الافتراضية)

class Shape {
public:
    virtual void draw() {  // دالة افتراضية
        cout << "رسم شكل عام" << endl;
    }
};

class Circle : public Shape {
public:
    void draw() override {  // تجاوز الدالة
        cout << "رسم دائرة" << endl;
    }
};

class Rectangle : public Shape {
public:
    void draw() override {
        cout << "رسم مستطيل" << endl;
    }
};

int main() {
    Shape* shapes[2];
    shapes[0] = new Circle();
    shapes[1] = new Rectangle();

    for(int i = 0; i < 2; i++) {
        shapes[i]->draw();  // ينادي الدالة المناسبة حسب نوع الكائن
    }
    // الإخراج: رسم دائرة / رسم مستطيل
    return 0;
}

ب. تعدد الأشكال أثناء الترجمة (تحميل الدوال)

class Printer {
public:
    void print(int i) {
        cout << "طباعة عدد: " << i << endl;
    }
    void print(double d) {
        cout << "طباعة عدد عشري: " << d << endl;
    }
    void print(string s) {
        cout << "طباعة نص: " << s << endl;
    }
};

4. التجريد (Abstraction)

إخفاء تفاصيل التنفيذ المعقدة وإظهار الوظائف الأساسية فقط للمستخدم. في ++C، يتحقق باستخدام الأصناف المجردة (Abstract classes) التي تحتوي على دوال افتراضية بحتة (pure virtual).

// صنف مجرد (لا يمكن إنشاء كائنات منه)
class Vehicle {
public:
    virtual void start() = 0;  // دالة افتراضية بحتة
    virtual void stop() = 0;

    void fuel() {  // دالة عادية
        cout << "تزويد بالوقود" << endl;
    }
};

class Car : public Vehicle {
public:
    void start() override {
        cout << "السيارة تبدأ" << endl;
    }
    void stop() override {
        cout << "السيارة تتوقف" << endl;
    }
};

int main() {
    // Vehicle v; // خطأ! صنف مجرد
    Car myCar;
    myCar.start();   // السيارة تبدأ
    myCar.fuel();    // تزويد بالوقود
    return 0;
}
ملاحظة: الأصناف المجردة تستخدم لتحديد واجهة موحدة لمجموعة من الأصناف المشتقة، ولا يمكن إنشاء كائنات منها مباشرة.

المنشئ (Constructor) والمدمر (Destructor)

المنشئ هو دالة خاصة تُستدعى تلقائياً عند إنشاء كائن. المدمر يُستدعى عند تدمير الكائن.

class Person {
private:
    string name;
    int age;
public:
    // المنشئ
    Person(string n, int a) {
        name = n;
        age = a;
        cout << "تم إنشاء كائن Person" << endl;
    }

    // المدمر
    ~Person() {
        cout << "تم تدمير كائن Person" << endl;
    }

    void display() {
        cout << "الاسم: " << name << ", العمر: " << age << endl;
    }
};

int main() {
    Person p1("أحمد", 25);
    p1.display();
    return 0; // عند انتهاء الدالة، يُستدعى المدمر تلقائياً
}

تمرين: صمم صنف Shape مع دوال افتراضية، ثم اشتق منه صنفين Circle و Rectangle. أضف دالة لحساب المساحة.

عرض الحل 
#include <iostream>
using namespace std;

class Shape {
public:
    virtual double area() const = 0; // pure virtual
};

class Circle : public Shape {
private:
    double radius;
public:
    Circle(double r) : radius(r) {}
    double area() const override {
        return 3.14159 * radius * radius;
    }
};

class Rectangle : public Shape {
private:
    double width, height;
public:
    Rectangle(double w, double h) : width(w), height(h) {}
    double area() const override {
        return width * height;
    }
};

int main() {
    Circle c(5);
    Rectangle r(4, 6);
    cout << "مساحة الدائرة: " << c.area() << endl;
    cout << "مساحة المستطيل: " << r.area() << endl;
    return 0;
}

أفضل الممارسات في OOP

  • اتبع مبدأ Single Responsibility: كل صنف له مسؤولية واحدة.
  • استخدم التغليف لحماية البيانات (اجعل الخصائص private).
  • فضل التركيب (composition) على الوراثة عندما يكون مناسباً.
  • استخدم الدوال الافتراضية (virtual) عندما تتوقع تعدد الأشكال.
  • اجعل المدمرات virtual في الأصناف الأساسية.
  • استخدم const مع الدوال التي لا تعدل الكائن.
