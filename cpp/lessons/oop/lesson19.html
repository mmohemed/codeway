ما هو التغليف (Encapsulation)؟

التغليف هو أحد المبادئ الأساسية في البرمجة الكائنية (OOP) والذي يهدف إلى:

إخفاء البيانات

إخفاء التفاصيل الداخلية للكائن ومنع الوصول المباشر للبيانات

تجميع البيانات

تجميع البيانات (الخصائص) والدوال (السلوكيات) في وحدة واحدة

التحكم بالوصول

تحديد كيفية وأين يمكن الوصول للبيانات

تمثيل التغليف

-balance: double
-accountNumber: string
+deposit(amount: double): void
+withdraw(amount: double): bool
+getBalance(): double

الخصائص خاصة (private) والدوال عامة (public)

معدّلات الوصول (Access Specifiers)

تستخدم معدّلات الوصول لتحقيق التغليف وتحديد مدى إمكانية الوصول لأعضاء الفئة.

المعدّل الرمز الوصف مثال
public + يمكن الوصول من أي مكان في البرنامج public: int x;
private - يمكن الوصول فقط داخل نفس الفئة private: int y;
protected # يمكن الوصول داخل الفئة والفئات المشتقة protected: int z;
مثال 1: تطبيق التغليف 
#include <iostream>
#include <string>
using namespace std;

class BankAccount {
private:
    // بيانات خاصة - لا يمكن الوصول المباشر من خارج الفئة
    string accountNumber;
    string ownerName;
    double balance;
    string pinCode;
    
public:
    // مُنشئ لتهيئة الحساب
    BankAccount(string accNum, string name, double initialBalance, string pin) {
        accountNumber = accNum;
        ownerName = name;
        if(initialBalance >= 0) {
            balance = initialBalance;
        } else {
            balance = 0;
        }
        pinCode = pin;
    }
    
    // واجهة عامة للتعامل مع البيانات
    bool deposit(double amount) {
        if(amount > 0) {
            balance += amount;
            return true;
        }
        return false;
    }
    
    bool withdraw(double amount, string pin) {
        if(pin == pinCode && amount > 0 && amount <= balance) {
            balance -= amount;
            return true;
        }
        return false;
    }
    
    double getBalance(string pin) {
        if(pin == pinCode) {
            return balance;
        }
        return -1; // رمز للدلالة على خطأ
    }
    
    string getAccountNumber() {
        return accountNumber;
    }
    
    string getOwnerName() {
        return ownerName;
    }
};

int main() {
    BankAccount myAccount("123456", "Ahmed Ali", 5000, "1234");
    
    // myAccount.balance = 10000; // خطأ! لا يمكن الوصول المباشر
    
    // استخدام الواجهة العامة
    cout << "Owner: " << myAccount.getOwnerName() << endl;
    cout << "Account: " << myAccount.getAccountNumber() << endl;
    
    myAccount.deposit(1000);
    cout << "Balance: $" << myAccount.getBalance("1234") << endl;
    
    if(myAccount.withdraw(500, "1234")) {
        cout << "Withdrawal successful" << endl;
    }
    
    cout << "Final balance: $" << myAccount.getBalance("1234") << endl;
    
    return 0;
}

شرح المثال:

private: البيانات (accountNumber, balance, pinCode) مخفية

public: الدوال العامة هي الواجهة للتعامل مع الكائن

التحقق: نتحقق من صحة المدخلات قبل تعديل البيانات

الأمان: الرقم السري محمي ولا يمكن الوصول إليه مباشرة

Getters و Setters

الدوال التي تسمح بالوصول الآمن للخصائص الخاصة (private) وتعديلها.

مثال 2: استخدام Getters و Setters 
#include <iostream>
#include <string>
using namespace std;

class Student {
private:
    string name;
    int age;
    double gpa;
    string studentId;
    
public:
    // Constructor
    Student(string id) {
        studentId = id;
        name = "غير محدد";
        age = 0;
        gpa = 0.0;
    }
    
    // Setters مع التحقق من صحة البيانات
    void setName(string n) {
        if(!n.empty()) {
            name = n;
        } else {
            cout << "خطأ: الاسم لا يمكن أن يكون فارغاً" << endl;
        }
    }
    
    void setAge(int a) {
        if(a >= 5 && a <= 100) {
            age = a;
        } else {
            cout << "خطأ: العمر يجب أن يكون بين 5 و 100" << endl;
        }
    }
    
    void setGPA(double g) {
        if(g >= 0.0 && g <= 4.0) {
            gpa = g;
        } else {
            cout << "خطأ: المعدل يجب أن يكون بين 0 و 4" << endl;
        }
    }
    
    // Getters
    string getName() {
        return name;
    }
    
    int getAge() {
        return age;
    }
    
    double getGPA() {
        return gpa;
    }
    
    string getStudentId() {
        return studentId;
    }
    
    // دوال إضافية
    bool isExcellent() {
        return gpa >= 3.5;
    }
    
    void displayInfo() {
        cout << "=== معلومات الطالب ===" << endl;
        cout << "الرقم الجامعي: " << studentId << endl;
        cout << "الاسم: " << name << endl;
        cout << "العمر: " << age << " سنة" << endl;
        cout << "المعدل: " << gpa << endl;
        cout << "الحالة: " << (isExcellent() ? "متفوق" : "عادي") << endl;
    }
};

int main() {
    Student s1("2021001");
    
    // استخدام setters
    s1.setName("Ahmed Mohamed");
    s1.setAge(20);
    s1.setGPA(3.7);
    
    // استخدام getters
    cout << "Student Name: " << s1.getName() << endl;
    
    s1.displayInfo();
    
    // محاولة إدخال قيم غير صحيحة
    cout << "\nمحاولة إدخال قيم غير صحيحة:" << endl;
    s1.setAge(150);  // سيتم رفضها
    s1.setGPA(5.0);  // سيتم رفضها
    
    return 0;
}

فوائد التغليف (Benefits of Encapsulation)

حماية البيانات

تمنع التعديل غير المصرح به أو غير الصحيح للبيانات

التحقق من الصحة

يمكن التحقق من صحة البيانات قبل تعيينها

سهولة الصيانة

يمكن تعديل التنفيذ الداخلي دون التأثير على الكود الخارجي

إعادة الاستخدام

تسهل إعادة استخدام الكود في أجزاء مختلفة من البرنامج

مثال 3: بدون تغليف ومع تغليف 
#include <iostream>
using namespace std;

// بدون تغليف - مشاكل محتملة
class BadRectangle {
public:
    double width;
    double height;
};

// مع تغليف - آمن ومنظم
class GoodRectangle {
private:
    double width;
    double height;
    
public:
    void setDimensions(double w, double h) {
        if(w > 0 && h > 0) {
            width = w;
            height = h;
        } else {
            cout << "خطأ: الأبعاد يجب أن تكون موجبة" << endl;
        }
    }
    
    double getWidth() {
        return width;
    }
    
    double getHeight() {
        return height;
    }
    
    double getArea() {
        return width * height;
    }
    
    double getPerimeter() {
        return 2 * (width + height);
    }
};

int main() {
    // الطريقة السيئة
    BadRectangle badRect;
    badRect.width = -5;  // يمكن تعيين قيمة سالبة!
    badRect.height = 10;
    cout << "Bad Rectangle Area: " << badRect.width * badRect.height << endl;
    
    // الطريقة الجيدة
    GoodRectangle goodRect;
    goodRect.setDimensions(-5, 10);  // سيتم رفضها
    goodRect.setDimensions(5, 10);   // قيمة صحيحة
    cout << "Good Rectangle Area: " << goodRect.getArea() << endl;
    
    return 0;
}

مثال 4: نظام إدارة الموظفين 
#include <iostream>
#include <string>
#include <vector>
using namespace std;

class Employee {
private:
    int id;
    string name;
    double baseSalary;
    double bonus;
    string department;
    vector<string> skills;
    
    // دوال مساعدة خاصة
    double calculateTax() {
        double salary = getTotalSalary();
        if(salary < 5000) return 0;
        if(salary < 10000) return salary * 0.1;
        return salary * 0.2;
    }
    
public:
    Employee(int empId, string empName, double salary, string dept) {
        id = empId;
        name = empName;
        baseSalary = salary;
        bonus = 0;
        department = dept;
    }
    
    // إدارة الراتب
    void setBonus(double b) {
        if(b >= 0) {
            bonus = b;
        }
    }
    
    double getTotalSalary() {
        return baseSalary + bonus;
    }
    
    double getNetSalary() {
        return getTotalSalary() - calculateTax();
    }
    
    // إدارة المهارات
    void addSkill(string skill) {
        skills.push_back(skill);
    }
    
    bool hasSkill(string skill) {
        for(const string& s : skills) {
            if(s == skill) return true;
        }
        return false;
    }
    
    void displaySkills() {
        cout << "مهارات " << name << ": ";
        for(const string& s : skills) {
            cout << s << " ";
        }
        cout << endl;
    }
    
    // Getters
    int getId() { return id; }
    string getName() { return name; }
    string getDepartment() { return department; }
    
    // عرض كامل المعلومات
    void displayInfo() {
        cout << "\n=== موظف: " << name << " ===" << endl;
        cout << "ID: " << id << endl;
        cout << "القسم: " << department << endl;
        cout << "الراتب الأساسي: $" << baseSalary << endl;
        cout << "المكافأة: $" << bonus << endl;
        cout << "إجمالي الراتب: $" << getTotalSalary() << endl;
        cout << "صافي الراتب: $" << getNetSalary() << endl;
        displaySkills();
    }
};

int main() {
    Employee emp1(1001, "Ahmed Ali", 8000, "IT");
    
    emp1.setBonus(1000);
    emp1.addSkill("C++");
    emp1.addSkill("Python");
    emp1.addSkill("SQL");
    
    emp1.displayInfo();
    
    if(emp1.hasSkill("C++")) {
        cout << emp1.getName() << " لديه مهارة C++" << endl;
    }
    
    return 0;
}

التمرين 1: نظام إدارة مكتبة

قم بإنشاء فئة Book مع تطبيق مبدأ التغليف:

الحل 
#include <iostream>
#include <string>
using namespace std;

class Book {
private:
    string title;
    string author;
    string isbn;
    int publicationYear;
    bool isBorrowed;
    string borrowedBy;
    
public:
    Book(string t, string a, string i, int year) {
        title = t;
        author = a;
        isbn = i;
        publicationYear = year;
        isBorrowed = false;
        borrowedBy = "";
    }
    
    bool borrowBook(string borrower) {
        if(!isBorrowed) {
            isBorrowed = true;
            borrowedBy = borrower;
            return true;
        }
        return false;
    }
    
    bool returnBook() {
        if(isBorrowed) {
            isBorrowed = false;
            borrowedBy = "";
            return true;
        }
        return false;
    }
    
    string getTitle() { return title; }
    string getAuthor() { return author; }
    string getISBN() { return isbn; }
    int getYear() { return publicationYear; }
    bool getStatus() { return isBorrowed; }
    string getBorrower() { return borrowedBy; }
    
    void displayInfo() {
        cout << "\n=== معلومات الكتاب ===" << endl;
        cout << "العنوان: " << title << endl;
        cout << "المؤلف: " << author << endl;
        cout << "ISBN: " << isbn << endl;
        cout << "سنة النشر: " << publicationYear << endl;
        cout << "الحالة: " << (isBorrowed ? "معار" : "متاح") << endl;
        if(isBorrowed) {
            cout << "مستعار من: " << borrowedBy << endl;
        }
    }
};

int main() {
    Book book1("C++ Programming", "Bjarne Stroustrup", "978-0-321-99278-9", 2013);
    
    book1.displayInfo();
    
    if(book1.borrowBook("Ahmed")) {
        cout << "\nتم استعارة الكتاب بنجاح" << endl;
    }
    
    book1.displayInfo();
    
    book1.returnBook();
    cout << "\nتم إعادة الكتاب" << endl;
    
    book1.displayInfo();
    
    return 0;
}

التمرين 2: نظام إدارة المنتجات

قم بإنشاء فئة Product مع تطبيق مبدأ التغليف:

الحل 
#include <iostream>
#include <string>
#include <iomanip>
using namespace std;

class Product {
private:
    int id;
    string name;
    double price;
    int quantity;
    string category;
    static double discountRate;  // نسبة خصم عامة
    
    double calculateDiscount() {
        return price * discountRate;
    }
    
public:
    Product(int i, string n, double p, int q, string c) {
        id = i;
        name = n;
        price = p;
        quantity = q;
        category = c;
    }
    
    // تحديث المخزون
    bool sell(int amount) {
        if(amount > 0 && amount <= quantity) {
            quantity -= amount;
            return true;
        }
        return false;
    }
    
    void restock(int amount) {
        if(amount > 0) {
            quantity += amount;
        }
    }
    
    // تعديل السعر مع التحقق
    void setPrice(double newPrice) {
        if(newPrice > 0) {
            price = newPrice;
        }
    }
    
    double getPrice() {
        return price;
    }
    
    double getPriceAfterDiscount() {
        return price - calculateDiscount();
    }
    
    // Getters
    int getId() { return id; }
    string getName() { return name; }
    int getQuantity() { return quantity; }
    string getCategory() { return category; }
    
    // عرض المعلومات
    void displayInfo() {
        cout << fixed << setprecision(2);
        cout << "\n=== " << name << " ===" << endl;
        cout << "ID: " << id << endl;
        cout << "التصنيف: " << category << endl;
        cout << "السعر الأصلي: $" << price << endl;
        cout << "السعر بعد الخصم: $" << getPriceAfterDiscount() << endl;
        cout << "الكمية المتوفرة: " << quantity << endl;
    }
};

double Product::discountRate = 0.1;  // 10% خصم عام

int main() {
    Product laptop(1001, "Laptop Dell XPS", 1200, 10, "Electronics");
    
    laptop.displayInfo();
    
    if(laptop.sell(2)) {
        cout << "\nتم بيع المنتج" << endl;
    }
    
    laptop.displayInfo();
    
    laptop.restock(5);
    cout << "\nتم إضافة 5 وحدات" << endl;
    
    laptop.displayInfo();
    
    return 0;
}
